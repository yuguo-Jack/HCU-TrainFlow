# [Bugfix][Pooling] Handle engine-originated abort outputs

[Upstream PR](https://github.com/vllm-project/vllm/pull/61062) · state: open · merged: False

## Original PR description

## Overview

Finish pooling requests cleanly when the engine aborts them, such as during `pause_generation(mode="abort")`. A tensor-less ABORT currently falls into the text-output path and kills the output handler.

## Claims

- Engine-aborted embedding requests finish with an empty pooling result, matching the existing frontend-abort behavior.
- Normal embedding requests continue working after resume.

## Validation

Qwen3-Embedding-0.6B: main and revert hit the detokenizer assertion and requests time out. With the fix, all 64 requests finish, including 11 cancelled requests, and an embedding request after resume succeeds. Applicable manual pre-commit hooks pass.

The six targeted abort tests pass with a public Qwen tokenizer substituted in the external test harness. The unmodified full output-processor suite requires a gated Llama tokenizer unavailable in this environment; it is not reported as passing.

```python
#!/usr/bin/env python3
import asyncio
import os
import traceback
from contextlib import suppress

from vllm.engine.arg_utils import AsyncEngineArgs
from vllm.pooling_params import PoolingParams
from vllm.utils.torch_utils import set_default_torch_num_threads
from vllm.v1.engine.async_llm import AsyncLLM


MODEL = os.environ.get(
    "MODEL", "Qwen/Qwen3-Embedding-0.6B"
)
NUM_REQUESTS = int(os.environ.get("NUM_REQUESTS", "64"))
PAUSE_DELAY = float(os.environ.get("PAUSE_DELAY", "0.01"))
TIMEOUT = float(os.environ.get("TIMEOUT", "30"))


def make_prompt(i: int) -> str:
    return (
        f"query {i}: "
        + " ".join(
            [
                "vllm pooling abort lifecycle correctness probe"
            ]
            * 120
        )
    )


async def collect_encode(engine: AsyncLLM, request_id: str) -> dict:
    outputs = []
    try:
        async for out in engine.encode(
            make_prompt(int(request_id.rsplit("-", 1)[-1])),
            PoolingParams(task="embed"),
            request_id=request_id,
        ):
            outputs.append(out)
        final = outputs[-1] if outputs else None
        shape = None
        if final is not None:
            shape = tuple(final.outputs.data.shape)
        return {
            "request_id": request_id,
            "ok": True,
            "num_outputs": len(outputs),
            "finished": None if final is None else final.finished,
            "shape": shape,
        }
    except BaseException as exc:
        return {
            "request_id": request_id,
            "ok": False,
            "exc_type": type(exc).__name__,
            "exc": repr(exc),
            "traceback": traceback.format_exc(limit=8),
        }


async def run_normal_control(engine: AsyncLLM) -> None:
    result = await asyncio.wait_for(collect_encode(engine, "normal-0"), TIMEOUT)
    print("NORMAL_CONTROL", result, flush=True)
    if not result["ok"] or not result["finished"]:
        raise RuntimeError(f"normal encode control failed: {result}")


async def run_abort_control(engine: AsyncLLM) -> None:
    task = asyncio.create_task(collect_encode(engine, "external-abort-0"))
    await asyncio.sleep(PAUSE_DELAY)
    await engine.abort("external-abort-0", internal=False)
    result = await asyncio.wait_for(task, TIMEOUT)
    print("EXTERNAL_ABORT_CONTROL", result, flush=True)
    if not result["ok"] or not result["finished"]:
        raise RuntimeError(f"external abort control failed: {result}")


async def run_pause_abort_probe(engine: AsyncLLM) -> int:
    tasks = [
        asyncio.create_task(collect_encode(engine, f"pause-abort-{i}"))
        for i in range(NUM_REQUESTS)
    ]
    await asyncio.sleep(PAUSE_DELAY)
    print(
        f"PAUSE_ABORT start num_requests={NUM_REQUESTS} delay={PAUSE_DELAY}",
        flush=True,
    )
    await asyncio.wait_for(
        engine.pause_generation(mode="abort", clear_cache=False), 30
    )
    await engine.resume_generation()
    results = await asyncio.wait_for(asyncio.gather(*tasks), TIMEOUT)
    failures = [result for result in results if not result["ok"]]
    finished = sum(1 for result in results if result.get("finished"))
    empty_shapes = sum(1 for result in results if result.get("shape") == (0,))
    print(
        f"PAUSE_ABORT summary finished={finished} empty_shapes={empty_shapes} "
        f"failures={len(failures)}",
        flush=True,
    )
    for result in results[:8]:
        print("PAUSE_ABORT sample", result, flush=True)
    for failure in failures[:4]:
        print("PAUSE_ABORT failure", failure, flush=True)
    if not failures:
        await run_normal_control(engine)
        assert empty_shapes > 0, "no request was aborted"
        print("POOLING_ABORT_PASSED", flush=True)
    return 2 if failures else 0


async def main() -> int:
    print(f"MODEL={MODEL}", flush=True)
    args = AsyncEngineArgs(
        model=MODEL,
        runner="pooling",
        enforce_eager=True,
        max_model_len=1024,
        max_num_seqs=max(NUM_REQUESTS, 8),
        max_num_batched_tokens=1024,
        gpu_memory_utilization=0.55,
        enable_log_requests=True,
    )
    with set_default_torch_num_threads(1):
        engine = AsyncLLM.from_engine_args(args)
    try:
        await run_normal_control(engine)
        await run_abort_control(engine)
        return await run_pause_abort_probe(engine)
    finally:
        with suppress(Exception):
            engine.shutdown()


if __name__ == "__main__":
    raise SystemExit(asyncio.run(main()))
```

## Details

Use the existing `EMPTY_CPU_TENSOR` sentinel only for a pooling request with ABORT and no pooling tensor. This preserves the distinction between an aborted pooling request and a generated-token response without changing scheduler policy or adding a device operation.

Duplicate check: #54430 handles callers abandoning `encode()`; #55940 records frontend-abort metrics. #51488 adds rejection of new requests while paused and a sentinel for its new PAUSED reason. This fix handles existing requests receiving ABORT from the engine.
