# [HiSparse] Own GPU residency instead of editing core block-table rows

[Upstream PR](https://github.com/vllm-project/vllm/pull/60647) · state: closed · merged: True

## Original PR description

## Purpose

Fixes #60109. This is an alternative to #60117 and #60123.

HiSparse stored GPU residency by editing core block-table rows after sending them: it adopted GPU copies of host prefix pages, and nulled pages the pool reused. The edits reached the worker through `SchedulerOutput.block_table_updates`, and #60109 was a new request's row arriving twice in one step. With this PR, HiSparse owns residency, and the worker only ever sees core rows grow (the scheduler fills an imported prefix's null tail entry before the row is first sent):

- **Scheduler:** `HiSparseResidentManager` records adopted and lost pages as per-request `residency_changes`. For each scheduled request, the coordinator sends the block ids of only the pages the worker lacks (new pages, plus pages adopted or lost since the last send; a new or re-admitted request gets its full row once), in HiSparse's connector metadata. At batch scale some request gains a page nearly every decode step.
- **Worker:** HiSparse keeps one persistent device table `[request state rows, resident groups, pages]`, written with one scatter per step that carries updates. It adds to core's per-group resident block tables, which still drive KV writes; HiSparse reads residency from its own table. The decode resolver kernel indexes it by state row, so nothing is copied per step; prefill staging gathers its rows once per batch.
- **Core:** `block_table_updates` is removed and `update_requests` is back to its pre-HiSparse form. `add_requests` is unchanged.

**Why this over #60117/#60123:** those keep HiSparse's row edits in core (in `update_requests`, and in `add_requests` for #60123). This removes them entirely. Cost: about 230 added lines versus about 10, plus one int32 table: about 18 MB at 256 seqs × 18k context × 61 groups, and about 0.25 GB at 512 seqs × 128k. It is allocated after memory profiling, like MRv2's block tables.

```python
"""Multi-needle retrieval at 16k-20k tokens against a vLLM completions server."""
import argparse, asyncio, json, random, re, time
import aiohttp
from transformers import AutoTokenizer

ADJ = "quiet amber distant hollow brisk silver crooked gentle rusty vivid narrow ancient humid pale restless sturdy".split()
NOUN = "river lantern orchard engine meadow harbor violin ledger tower canyon market glacier compass furnace garden archive".split()
VERB = "follows shelters outlines remembers crosses lifts measures echoes guards circles borrows mirrors".split()
PLACE = "the old mill|the northern ridge|the city wall|the train yard|the salt flats|the copper mine|the library steps|the dry lake".split("|")
KEYS = [a + b for a in "Bri Cal Dun Elm Fen Gal Har Ing".split()
        for b in "ton dor wick mere ford holm by stead".split()]
NEEDLES, ASKED = 32, 32

def sentence(rng):
    return (f"The {rng.choice(ADJ)} {rng.choice(NOUN)} {rng.choice(VERB)} the "
            f"{rng.choice(ADJ)} {rng.choice(NOUN)} near {rng.choice(PLACE)}.")

def build(tok, n, seed):
    rng = random.Random(seed)
    prompts = []
    for i in range(n):
        target = 16384 if i % 2 == 0 else 20480
        keys = rng.sample(KEYS, NEEDLES)
        values = [str(rng.randrange(1_000_000, 9_999_999)) for _ in keys]
        depths = [rng.random() for _ in keys]
        asked = rng.sample(range(NEEDLES), ASKED)
        filler = [sentence(rng) for _ in range(target // 11)]
        body_tokens = len(tok.encode(" ".join(filler), add_special_tokens=False))
        filler = filler[: int(len(filler) * (target - 200) / body_tokens)]
        for d, k, v in sorted(zip(depths, keys, values), reverse=True):
            filler.insert(int(d * len(filler)), f"The secret code for {k} is {v}.")
        names = [keys[q] for q in asked]
        prompt = ("Read the document and answer the question.\n\nDocument:\n"
                  + " ".join(filler)
                  + "\n\nQuestion: What are the secret codes for "
                  + ", ".join(names[:-1]) + f" and {names[-1]}? Reply with one "
                  + "line per name in the form 'name: code'.\nAnswer:\n"
                  + f"{names[0]}:")
        prompts.append(dict(id=i, prompt=prompt, names=names,
                            answers=[values[q] for q in asked],
                            ntok=len(tok.encode(prompt, add_special_tokens=False))))
    return prompts

async def run_wave(url, model, prompts, conc):
    sem = asyncio.Semaphore(conc)
    timeout = aiohttp.ClientTimeout(total=3600)
    async with aiohttp.ClientSession(timeout=timeout) as s:
        async def one(p):
            async with sem:
                body = dict(model=model, prompt=p["prompt"], max_tokens=512, temperature=0)
                async with s.post(f"{url}/v1/completions", json=body) as r:
                    js = await r.json()
                    if "choices" not in js:
                        return p["id"], None, str(js)[:300]
                    return p["id"], js["choices"][0]["text"], None
        return await asyncio.gather(*(one(p) for p in prompts))

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--model", default="RedHatAI/GLM-5.3-NVFP4")
    ap.add_argument("--url", default="http://localhost:8123")
    ap.add_argument("--n", type=int, default=192)
    ap.add_argument("--waves", type=int, default=2)
    ap.add_argument("--concurrency", type=int, default=64)
    ap.add_argument("--seed", type=int, default=0)
    ap.add_argument("--out", required=True)
    a = ap.parse_args()
    tok = AutoTokenizer.from_pretrained(a.model)
    prompts = build(tok, a.n, a.seed)
    print(f"prompts={len(prompts)} tokens min={min(p['ntok'] for p in prompts)} "
          f"max={max(p['ntok'] for p in prompts)}", flush=True)
    results = []
    for w in range(a.waves):
        t = time.time()
        res = asyncio.run(run_wave(a.url, a.model, prompts, a.concurrency))
        errors = sum(e is not None for _, _, e in res)
        def score(i, text):
            if text is None:
                return 0
            full = f"{prompts[i]['names'][0]}:" + text
            got = 0
            for name, answer in zip(prompts[i]["names"], prompts[i]["answers"]):
                m = re.search(rf"{name}:\s*(\d+)", full)
                got += m is not None and m.group(1) == answer
            return got
        correct = sum(score(i, text) for i, text, _ in res)
        print(f"wave {w}: correct={correct}/{ASKED * len(prompts)} errors={errors} "
              f"time={time.time()-t:.0f}s", flush=True)
        results.append([dict(id=i, text=text, error=e, answers=prompts[i]["answers"],
                             names=prompts[i]["names"]) for i, text, e in res])
    json.dump(results, open(a.out, "w"))

if __name__ == "__main__":
    main()
```
