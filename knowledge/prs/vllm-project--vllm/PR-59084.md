# [GLM5.3 Perf] Fuse Q projection with `fused_q` kernel, 1.27 ~1.64x kernel performance improvement

[Upstream PR](https://github.com/vllm-project/vllm/pull/59084) · state: closed · merged: False

## Original PR description

## Purpose

Fuse Q projection with `fused_q` kernel, removing one separate `torch.bmm` launch per attention layer.

All 78 layers are eligible: **78 fewer GPU kernel launches per TP rank**

## Test

Acc covered in current unit test

Perf can be seen in this AI generated script

```py
import torch
import triton

from vllm.models.deepseek_v32.common.kernels import fused_q


@torch.inference_mode()
def benchmark(num_tokens: int) -> None:
    q = torch.randn(num_tokens, 16, 256, device="cuda", dtype=torch.bfloat16)
    q_nope, q_pe = q.split([192, 64], dim=-1)
    w_uk_t = torch.randn(512, 16, 192, device="cuda", dtype=torch.bfloat16).permute(
        1, 2, 0
    )
    index_q = torch.randn(num_tokens, 32, 128, device="cuda", dtype=torch.bfloat16)
    index_weights = torch.randn(num_tokens, 32, device="cuda")
    positions = torch.arange(num_tokens, device="cuda")
    rope_cache = torch.randn(max(128, num_tokens), 64, device="cuda")
    q_scale = torch.tensor([0.37], device="cuda")

    def prepare(ql_nope, *, fuse_projection: bool):
        return fused_q(
            positions,
            q_pe,
            rope_cache,
            index_q,
            rope_cache,
            ql_nope,
            q_scale,
            index_weights,
            128**-0.5,
            32**-0.5,
            index_rope_interleave=True,
            quantize_mqa=False,
            q_nope=q_nope if fuse_projection else None,
            w_uk_t=w_uk_t if fuse_projection else None,
        )

    def baseline():
        ql_nope = torch.bmm(q_nope.transpose(0, 1), w_uk_t).transpose(0, 1)
        return ql_nope, *prepare(ql_nope, fuse_projection=False)

    def fused():
        ql_nope = torch.empty(num_tokens, 16, 512, device="cuda", dtype=q.dtype)
        return ql_nope, *prepare(ql_nope, fuse_projection=True)

    reference, result = baseline(), fused()
    torch.testing.assert_close(result[0], reference[0], rtol=0.02, atol=0.05)
    for actual, expected in zip(result[1:], reference[1:]):
        torch.testing.assert_close(
            actual.float(), expected.float(), rtol=0.08, atol=0.25
        )

    baseline_us = triton.testing.do_bench_cudagraph(baseline, rep=40) * 1000
    fused_us = triton.testing.do_bench_cudagraph(fused, rep=40) * 1000
    print(
        f"{num_tokens:>3}  {baseline_us:>10.2f}  {fused_us:>8.2f}  "
        f"{baseline_us / fused_us:>7.2f}x",
        flush=True,
    )


if __name__ == "__main__":
    torch.manual_seed(0)
    print("  T  baseline_us  fused_us  speedup")
    for num_tokens in (
        1, 4, 6, 8, 16, 24, 32, 48, 64, 96, 128, 256, 512, 1024, 2048, 4096
    ):
        benchmark(num_tokens)

```

I haven't checked all the changes carefully but having `NUM_TOKENS` being a constexpr is definitely a blocking issue i.e. it will recompile for every M values from 1 to 1024 based on the current code.

Can you also sweep M until we see the cross-over between the fused_q and unfused baseline? This will justify the dispatch threshold. For example your microbenchmark results already show the fused_q is better up to M=4096, but you only dispatch for M <= 1024

I wonder if torch.bmm baseline can be better too e.g. LL-GEMM from FlashInfer or vLLM in-tree kernels.

`use_pdl` is also disabled when `fuse_q_proj` is enabled, so benchmarking should be careful. Ideally we should have e2e perf results too.

```bash
"""Compare fused Q against BMM + fused_q with warm and cold L2 timing."""
import sys

# CUPTI is isolated from the user environment.
sys.path.insert(0, "/tmp/vllm-cupti-deps")

import importlib.metadata
import importlib.util
import statistics
import warnings

import torch
import triton.testing
from flashinfer.testing import bench_gpu_time_with_cupti
from cupti import cupti  # Verify that FlashInfer will use CUPTI, not its fallback.

# Load only the PR kernel; the temporary worktree lacks vLLM CUDA extensions.
KERNEL_PATH = "/tmp/vllm-fuse-q-cupti/vllm/models/deepseek_v32/common/kernels.py"
spec = importlib.util.spec_from_file_location("review_fused_q_kernels", KERNEL_PATH)
assert spec is not None and spec.loader is not None
kernels = importlib.util.module_from_spec(spec)
spec.loader.exec_module(kernels)
fused_q = kernels.fused_q

TOKENS = (1, 4, 16, 32, 128, 512, 1024, 4096)
HEADS = 16
INDEX_HEADS = 32


def bench_cupti(fn, cold):
    with warnings.catch_warnings():
        warnings.filterwarnings('error', message='.*Falling back to CUDA events.*')
        times = bench_gpu_time_with_cupti(
            fn,
            dry_run_iters=5,
            repeat_iters=50,
            use_cuda_graph=True,
            cold_l2_cache=cold,
        )
    return statistics.median(times) * 1000


def main():
    torch.manual_seed(42)
    print('GPU:', torch.cuda.get_device_name())
    print('torch:', torch.__version__, 'CUDA:', torch.version.cuda)
    print('cupti-python:', importlib.metadata.version('cupti-python'))
    print('T  mode        baseline_us  fused_us  speedup')
    for t in TOKENS:
        pos = torch.arange(t, dtype=torch.int64, device='cuda')
        q_nope = torch.randn(t, HEADS, 192, dtype=torch.bfloat16, device='cuda')
        q_pe = torch.randn(t, HEADS, 64, dtype=torch.bfloat16, device='cuda')
        w_uk_t = torch.randn(512, HEADS, 192, dtype=torch.bfloat16, device='cuda').permute(1, 2, 0)
        cos_sin = torch.randn(max(4096, t), 64, dtype=torch.float32, device='cuda')
        index_q = torch.randn(t, INDEX_HEADS, 128, dtype=torch.bfloat16, device='cuda')
        index_weights = torch.rand(t, INDEX_HEADS, dtype=torch.float32, device='cuda')
        q_scale = torch.ones(1, dtype=torch.float32, device='cuda')
        ql_out = torch.empty(t, HEADS, 512, dtype=torch.bfloat16, device='cuda')

        def run_baseline():
            ql = torch.bmm(q_nope.transpose(0, 1), w_uk_t).transpose(0, 1)
            return ql, fused_q(
                pos, q_pe, cos_sin, index_q, cos_sin, ql, q_scale,
                index_weights, 128**-0.5, INDEX_HEADS**-0.5,
                has_indexer=True, index_rope_interleave=False,
                quantize_mqa=False,
            )

        def run_fused():
            return ql_out, fused_q(
                pos, q_pe, cos_sin, index_q, cos_sin, ql_out, q_scale,
                index_weights, 128**-0.5, INDEX_HEADS**-0.5,
                has_indexer=True, index_rope_interleave=False,
                quantize_mqa=False, q_nope=q_nope, w_uk_t=w_uk_t,
            )

        for _ in range(3):
            ref = run_baseline()
            actual = run_fused()
        torch.cuda.synchronize()
        torch.testing.assert_close(actual[0], ref[0], atol=1e-2, rtol=1e-2)
        torch.testing.assert_close(actual[1][2], ref[1][2], atol=1e-2, rtol=1e-2)
        torch.testing.assert_close(actual[1][1], ref[1][1], atol=1e-3, rtol=1e-3)

        methods = (
            ('triton_warm', lambda fn: triton.testing.do_bench_cudagraph(fn, rep=10) * 1000),
            ('cupti_warm', lambda fn: bench_cupti(fn, cold=False)),
            ('cupti_cold', lambda fn: bench_cupti(fn, cold=True)),
        )
        for mode, bench in methods:
            baseline = bench(run_baseline)
            fused = bench(run_fused)
            print(f'{t:<4d} {mode:<12s} {baseline:11.2f} {fused:9.2f} {baseline/fused:7.2f}x', flush=True)


if __name__ == '__main__':
    main()

```
