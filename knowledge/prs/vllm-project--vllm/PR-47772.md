# [Bugfix][Pooling] Align CrossEncoder token type ids after truncation

[Upstream PR](https://github.com/vllm-project/vllm/pull/47772) · state: closed · merged: True

## Original PR description


## Purpose

CrossEncoder scoring compresses `token_type_ids` into a query/document split point before post-tokenization truncation runs. When a request uses `truncate_prompt_tokens` with `truncation_side="left"`, the kept prompt tokens can all come from the document segment, but the stale split still marks them as query tokens. `LLM.score()`, `/score`, and `/rerank` can then return HTTP 200 with silently wrong scores for BERT-style cross-encoder models such as `cross-encoder/ms-marco-MiniLM-L6-v2`.

This change applies the same post-tokenization truncation to `token_type_ids` before compressing the split, while preserving existing pooling extras such as `cache_salt`. Requests without token type ids keep the old path, and requests that do not change the prompt length keep the same compressed boundary.

I did not find an existing issue or PR for this token type boundary bug. The nearest merged PR is #47082, which preserves cross-encoder `extra_kwargs` but does not update the compressed boundary after truncation.

## Test Plan

Run a real `LLM.score()` E2E with `cross-encoder/ms-marco-MiniLM-L6-v2`, comparing vLLM raw scores against HF references using the correct left-truncated segment ids and the stale all-query segment ids. Run the same script with the fix, after reverting the fix, and after restoring the fix. Run the full cross-encoder offline scoring test file plus scoped ruff, format, and diff checks.

## Test Result

The E2E flips in the expected direction. On main/reverted, vLLM is closer to the stale all-query HF score. With the fix, vLLM is closer to the correct left-truncated HF score.

Repro script:

```python
import os

import torch
from transformers import AutoModelForSequenceClassification, AutoTokenizer

from vllm import LLM, PoolingParams
from vllm.entrypoints.pooling.scoring.utils import compress_token_type_ids


MODEL = os.environ.get("MODEL", "cross-encoder/ms-marco-MiniLM-L6-v2")
QUERY = os.environ.get("QUERY", "query " * 30)
DOCUMENT = os.environ.get("DOCUMENT", "document " * 30)
TRUNCATE_TO = int(os.environ.get("TRUNCATE_TO", "16"))


def score_hf(
    model: AutoModelForSequenceClassification,
    input_ids: list[int],
    token_type_ids: list[int],
) -> float:
    with torch.inference_mode():
        outputs = model(
            input_ids=torch.tensor([input_ids], device=model.device),
            token_type_ids=torch.tensor([token_type_ids], device=model.device),
            attention_mask=torch.ones((1, len(input_ids)), device=model.device),
        )
    return float(outputs.logits[0, 0].detach().cpu())


def main() -> None:
    tokenizer = AutoTokenizer.from_pretrained(MODEL)
    encoded = tokenizer(
        text=QUERY,
        text_pair=DOCUMENT,
        truncation=False,
        add_special_tokens=True,
    )
    input_ids = list(encoded["input_ids"])
    token_type_ids = list(encoded["token_type_ids"])

    kept_input_ids = input_ids[-TRUNCATE_TO:]
    correct_token_type_ids = token_type_ids[-TRUNCATE_TO:]

    stale_boundary = compress_token_type_ids(token_type_ids)
    wrong_token_type_ids = [
        int(i >= stale_boundary) for i in range(len(kept_input_ids))
    ]

    hf_model = AutoModelForSequenceClassification.from_pretrained(MODEL).cuda().eval()
    hf_correct = score_hf(hf_model, kept_input_ids, correct_token_type_ids)
    hf_wrong = score_hf(hf_model, kept_input_ids, wrong_token_type_ids)

    llm = LLM(
        model=MODEL,
        runner="pooling",
        dtype="float32",
        enforce_eager=True,
        gpu_memory_utilization=0.25,
        max_model_len=128,
    )
    outputs = llm.score(
        QUERY,
        DOCUMENT,
        pooling_params=PoolingParams(use_activation=False),
        tokenization_kwargs={
            "truncate_prompt_tokens": TRUNCATE_TO,
            "truncation_side": "left",
        },
        use_tqdm=False,
    )
    vllm_score = float(outputs[0].outputs.score)

    print(
        {
            "full_len": len(input_ids),
            "kept_len": len(kept_input_ids),
            "stale_boundary": stale_boundary,
            "correct_boundary_after_left": compress_token_type_ids(
                correct_token_type_ids
            ),
            "hf_correct_raw": hf_correct,
            "hf_wrong_raw": hf_wrong,
            "vllm_raw": vllm_score,
            "abs_vllm_minus_correct": abs(vllm_score - hf_correct),
            "abs_vllm_minus_wrong": abs(vllm_score - hf_wrong),
            "vllm_prompt_len": len(outputs[0].prompt_token_ids),
        }
    )


if __name__ == "__main__":
    main()
```
