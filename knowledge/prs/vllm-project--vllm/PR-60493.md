# [Bugfix] Reserve sparse MLA indexer metadata before sizing the KV cache

[Upstream PR](https://github.com/vllm-project/vllm/pull/60493) · state: open · merged: False

## Original PR description

## Overview

Reserve the sparse MLA indexer's expanded block table before automatic KV-cache sizing (#60450). The table is allocated after profiling and can otherwise push startup past the estimated budget.

## Claims

- Match runtime builder groups and table widths, including microbatches and draft-model builders, without changing buffer dimensions or lifetime.
- Include the reservation in context auto-fit. Preserve explicit KV capacity and avoid charging it again when replaying a startup plan.

## Validation

Against `19c0bed`: 324 tests passed; two multi-GPU configuration tests failed on this single-GPU host, also on unmodified main. PR-scoped pre-commit passed.

On an RTX 5070 Ti, an eager two-layer dummy-weight DSA model produced the same 32 tokens and returned logprobs. KV plus table allocation went from 2,052,710 bytes over budget to 65,434 bytes below it. This measures capacity and output parity; full B200/GLM, CUDA graphs, HiSparse and distributed execution remain untested.

## Details

Start with the [indexer reservation](https://github.com/vllm-project/vllm/blob/3e13609f050959132d930b8eccf606178b881097/vllm/v1/attention/backends/mla/indexer.py#L925) and [KV planner](https://github.com/vllm-project/vllm/blob/3e13609f050959132d930b8eccf606178b881097/vllm/v1/core/kv_cache_utils.py#L2813). A default-zero builder hook reports this late allocation; the planner caches per-worker estimates during context fitting. Both runners allocate the table, so both need the accounting fix. The hook covers this table only and adds no steady-state inference synchronization.

Related #46884, #46794 and #43970 address table shape/alignment; this PR accounts for its memory.

The worker extension only reads allocation metadata through a named RPC; it does not alter inference or memory planning.

`metadata_probe_worker.py`:

```python
"""Read-only worker diagnostic, exposed through a named local test RPC."""
class MetadataProbeExtension:
    def inspect_metadata_budget(self):
        import torch
        runner = self.model_runner
        tables = []
        for groups in runner.attn_groups:
            for group in groups:
                for builder in group.metadata_builders:
                    table = getattr(builder, 'expanded_block_table_buffer', None)
                    if table is not None:
                        tables.append(dict(shape=list(table.shape),
                            storage_bytes=table.untyped_storage().nbytes(),
                            builder=type(builder).__name__, device=str(table.device)))
        return dict(tables=tables, num_gpu_blocks=self.cache_config.num_gpu_blocks,
            reserved_metadata_bytes=getattr(self, 'metadata_memory_bytes', 0),
            kv_allocation_bytes=max(t.size for t in runner.kv_cache_config.kv_cache_tensors
                if not t.host_resident),
            raw_available_kv_budget=self.available_kv_cache_memory_bytes,
            max_model_len=self.model_config.max_model_len,
            runner=type(runner).__module__, gpu=torch.cuda.get_device_name(),
            torch=str(torch.__version__), cuda=torch.version.cuda)
```

```python
"""Full engine A/B with a small randomly initialized DeepSeek V3.2 model.

This is a systems regression fixture, not an accuracy evaluation of trained
DeepSeek/GLM weights. Production model, attention and indexer code are unchanged.
"""
import faulthandler
faulthandler.enable()
faulthandler.dump_traceback_later(180, repeat=True)
import argparse
import json
from pathlib import Path
import time

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--label', required=True)
    parser.add_argument('--output', required=True)
    parser.add_argument('--backend', default='FLASHINFER_MLA_SPARSE_SM120')
    parser.add_argument('--kv-cache-dtype', default='fp8')
    args = parser.parse_args()
    out = Path(args.output)
    assert not out.exists(), 'Preserve previous result'
    root = Path(__file__).resolve().parent
    model = root/'tiny_dsa'
    model.mkdir(exist_ok=True)
    config = dict(
        model_type='deepseek_v3', architectures=['DeepseekV32ForCausalLM'],
        vocab_size=256, hidden_size=512, intermediate_size=1024,
        moe_intermediate_size=256, num_hidden_layers=2, num_attention_heads=32,
        num_key_value_heads=32, q_lora_rank=256, kv_lora_rank=512,
        qk_nope_head_dim=128, qk_rope_head_dim=64, v_head_dim=128,
        index_topk=2048, index_n_heads=64, index_head_dim=128,
        n_routed_experts=4, n_shared_experts=1, num_experts_per_tok=2,
        first_k_dense_replace=2, max_position_embeddings=65536,
        rope_theta=10000.0, rms_norm_eps=1e-6, hidden_act='silu',
        torch_dtype='bfloat16', tie_word_embeddings=False,
        bos_token_id=1, eos_token_id=2, attention_bias=False,
    )
    (model/'config.json').write_text(json.dumps(config,indent=2)+'\n')
    print('Creating full vLLM engine with local dummy-weight DSA fixture', flush=True)
    from vllm import LLM, SamplingParams
    import vllm

    start = time.monotonic()
    llm = LLM(model=str(model), skip_tokenizer_init=True, load_format='dummy',
        dtype='bfloat16', enforce_eager=True, seed=7,
        max_model_len=65536, max_num_batched_tokens=512, max_num_seqs=4,
        gpu_memory_utilization=0.35, enable_prefix_caching=False,
        kv_cache_dtype=args.kv_cache_dtype, block_size=64,
        attention_config={'backend':args.backend},
        worker_extension_cls='metadata_probe_worker.MetadataProbeExtension')
    initialized = time.monotonic()
    worker_info = llm.collective_rpc('inspect_metadata_budget')
    print(json.dumps(worker_info,indent=2), flush=True)
    prompts = [{'prompt_token_ids': [3+(i%253) for i in range(length)]}
        for length in (8, 256, 1024, 4096)]
    outputs = llm.generate(prompts, SamplingParams(temperature=0, max_tokens=8,
        ignore_eos=True, logprobs=5), use_tqdm=False)
    result = dict(label=args.label, source=vllm.__file__, version=vllm.__version__,
        init_seconds=initialized-start, generation_seconds=time.monotonic()-initialized,
        config=config, workers=worker_info, backend=args.backend,
        kv_cache_dtype=args.kv_cache_dtype,
        outputs=[dict(prompt_tokens=len(r.prompt_token_ids),
            token_ids=list(r.outputs[0].token_ids),
            logprobs=[{str(k):v.logprob for k,v in entry.items()}
                for entry in r.outputs[0].logprobs]) for r in outputs],
        scope='Real end-to-end engine and native GPU sparse attention, randomly initialized two-layer DSA fixture. This is not a trained-model quality benchmark or reproduction of the full B200 GLM deployment.')
    out.parent.mkdir(parents=True,exist_ok=True)
    out.write_text(json.dumps(result,indent=2)+'\n')
    print(f'Completed and wrote {out}', flush=True)


if __name__ == '__main__':
    main()
```
