# [Bugfix][Frontend] Preserve token IDs in chat derender responses

[Upstream PR](https://github.com/vllm-project/vllm/pull/61065) · state: closed · merged: True

## Original PR description

## Overview

Preserve requested `token_ids` in chat derender responses, including plain and parsed streaming. Follow normal chat behavior when reasoning is hidden.

## Claims

- The render → generate → derender flow returns the generated token IDs when requested.
- Requests without `return_token_ids` still omit them; parsed responses with `include_reasoning=False` also omit them.

## Validation

Real Qwen3-0.6B HTTP before/fix/revert confirms batch and streaming IDs are lost on main, retained by the fix, and lost again after revert. False/absent flag controls and a real reasoning-parser privacy check pass. The existing DeepSeek-R1-Distill-Qwen-1.5B parity suite passes 10 tests with 6 native skips; the hidden-reasoning unit test passes. Applicable manual pre-commit hooks pass.

```python
import json
import sys
import time
from typing import Any

import requests


def post_json(base_url: str, path: str, payload: dict[str, Any]) -> dict[str, Any]:
    response = requests.post(f"{base_url}{path}", json=payload, timeout=120)
    try:
        data = response.json()
    except Exception:
        data = {"raw_text": response.text}
    if response.status_code != 200:
        raise RuntimeError(
            f"{path} returned {response.status_code}: "
            f"{json.dumps(data, ensure_ascii=False)}"
        )
    return data


def wait_ready(base_url: str, model: str) -> None:
    deadline = time.time() + 180
    last_error = None
    while time.time() < deadline:
        try:
            response = requests.get(f"{base_url}/v1/models", timeout=5)
            if response.status_code == 200:
                models = response.json().get("data", [])
                if not models or any(item.get("id") == model for item in models):
                    return
        except Exception as exc:
            last_error = exc
        time.sleep(2)
    raise RuntimeError(f"server did not become ready: {last_error!r}")


def main() -> int:
    if len(sys.argv) != 3:
        print("usage: probe.py PORT MODEL", file=sys.stderr)
        return 2

    port = int(sys.argv[1])
    model = sys.argv[2]
    base_url = f"http://127.0.0.1:{port}"
    wait_ready(base_url, model)

    chat_request = {
        "model": model,
        "messages": [{"role": "user", "content": "Say alpha beta gamma."}],
        "max_tokens": 8,
        "temperature": 0,
        "return_token_ids": True,
    }

    render = post_json(base_url, "/v1/chat/completions/render", chat_request)
    generate = post_json(base_url, "/inference/v1/generate", render)
    derender = post_json(
        base_url,
        "/v1/chat/completions/derender",
        {
            "model": model,
            "generate_response": generate,
            "prompt_tokens": len(render["token_ids"]),
            "chat_request": chat_request,
        },
    )
    for flag in (False, None):
        control_request = dict(chat_request)
        if flag is None:
            control_request.pop("return_token_ids")
        else:
            control_request["return_token_ids"] = flag
        control = post_json(base_url, "/v1/chat/completions/derender", {
            "model": model, "generate_response": generate,
            "prompt_tokens": len(render["token_ids"]), "chat_request": control_request})
        assert control["choices"][0].get("token_ids") is None, control
    stream_results = []
    state = None
    ids = generate["choices"][0]["token_ids"]
    for i, tid in enumerate(ids):
        streamed = post_json(base_url, "/v1/chat/completions/derender", {
            "stream": True, "model": model,
            "generate_chunk": {"request_id": generate["request_id"], "choices": [{
                "index": 0, "token_ids": [tid], "finish_reason": "length" if i == len(ids)-1 else None}]},
            "stream_state": state, "prompt_tokens": len(render["token_ids"]),
            "prompt_token_ids": render["token_ids"], "chat_request": chat_request})
        state = streamed["stream_state"]
        stream_results.append(streamed["chunk"]["choices"][0].get("token_ids"))
    print("STREAM_TOKEN_IDS", json.dumps(stream_results), flush=True)
    normal = post_json(base_url, "/v1/chat/completions", chat_request)

    generated_token_ids = generate["choices"][0]["token_ids"]
    derender_choice = derender["choices"][0]
    normal_choice = normal["choices"][0]
    derender_token_ids = derender_choice.get("token_ids")
    normal_token_ids = normal_choice.get("token_ids")

    result = {
        "render_request_id": render["request_id"],
        "generate_request_id": generate["request_id"],
        "generated_token_ids": generated_token_ids,
        "derender_token_ids": derender_token_ids,
        "normal_token_ids_present": isinstance(normal_token_ids, list),
        "normal_prompt_token_ids_present": isinstance(
            normal.get("prompt_token_ids"), list
        ),
        "derender_prompt_token_ids": derender.get("prompt_token_ids"),
        "derender_matches_generate": derender_token_ids == generated_token_ids,
        "usage": derender.get("usage"),
    }
    print("RESULT", json.dumps(result, ensure_ascii=False))

    if not isinstance(normal_token_ids, list):
        print("CONTROL_FAILED_NORMAL_CHAT_DID_NOT_RETURN_TOKEN_IDS")
        return 3
    if derender_token_ids == generated_token_ids and stream_results == [[tid] for tid in ids]:
        print("NO_BUG_REPRODUCED")
        return 0

    print("BUG_REPRODUCED_DERENDER_CHAT_DROPPED_TOKEN_IDS")
    return 1


if __name__ == "__main__":
    raise SystemExit(main())
```

## Details

The generate response already contains IDs; the chat derender response constructors omitted them. Forward that existing field at each response constructor, gated by the request and the same hidden-reasoning policy used by ordinary chat. Extend existing parity checks rather than adding a separate suite.

Duplicate check: #60962 and #60437 address logprob resolution in parsed streaming, not forwarding `token_ids`. The changes touch adjacent response construction but preserve separate fields.
