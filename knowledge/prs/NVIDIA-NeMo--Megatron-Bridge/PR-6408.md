# fix(ckpt): remove stale weights after HF export

[Upstream PR](https://github.com/NVIDIA-NeMo/Megatron-Bridge/pull/6408) · state: open · merged: False

## Original PR description

# What does this PR do ?

Fix old weights being loaded after `AutoBridge.save_hf_weights` exports a new checkpoint into an existing directory.

A single-file export leaves the previous shard index behind. The next load follows that index and reads the old shards instead of the new `model.safetensors`. This change removes obsolete weight files after the replacement export succeeds.

# Changelog

- Track files written by both source-backed and config-only exports.
- Remove previous standard weight files and index-referenced shards that were not rewritten. Preserve unrelated files and symlink targets.
- Test layout changes, partial exports, symlinks, write failures and recovery from an incomplete index.

## Validation

139 tests passed; all repository pre-commit hooks passed.

Supplemental validation used a real AutoBridge load/export/reload on one H200 with a one-layer BF16 Qwen2 model (hidden size 64, vocabulary 128, TP1/PP1/CP1). Replacing an eight-shard checkpoint with a single file left 9 of 15 tensors incorrect on the baseline and with cleanup reverted (maximum error 0.75); this patch reloaded all tensors exactly. These runs used PyTorch 26.02 and Transformers 5.12.1, not the CI image. The complete harness and recorded results are included below.

No downloaded checkpoint or training data is required.

Recorded versions: baseline `4f0e30eeef287fcc605d02e8e32c87b591ca155c`, candidate `571dfe51acc4fab838442a6ccc89c9b593280bf4`, Megatron-Core `8949def81eb744b9219e2d794c80265928c40a5b`, PyTorch 26.02 container, Transformers 5.12.1, one H200. These were not runs of the upstream CI image.

The script creates two small Qwen2 models. It fills the old parameters with `-0.5` and saves them with a deliberately small `max_shard_size="5KB"`, producing eight shards. It loads parameters filled with `0.25` through AutoBridge, then exports through the config-only path to the old directory. That export fits in one file. Finally, it reloads through AutoBridge and compares every tensor in the Megatron model state.

The 5KB limit forces a layout change; ordinary repeated exports do not necessarily change layout. This demonstrates correctness under directory reuse, not production frequency.

Run the following from a clean checkout of this PR. This is the full Python harness used for the recorded runs; the commands select only its `roundtrip` case. Its other cases are not part of the results below. Each arm runs in a separate process and a fresh output directory. `revert` disables only collection of previous filenames, leaving the writer unchanged. No source monkeypatch is used for the baseline or fixed arm.

#!/usr/bin/env python3
"""Independent public AutoBridge load-save-load and symlink-boundary probe."""

from __future__ import annotations

import argparse
import datetime
import json
import os
from pathlib import Path


triton_root = os.environ.get("REVIEW_TRITON_ROOT")
if triton_root:
    os.environ["TRITON_CACHE_DIR"] = str(Path(triton_root) / f"rank-{os.environ.get('LOCAL_RANK', '0')}")

import megatron.core.parallel_state as parallel_state
import torch
import torch.distributed as dist
from transformers import Qwen2Config, Qwen2ForCausalLM

import megatron.bridge.models.conversion.auto_bridge as auto_bridge_module
from megatron.bridge.models.conversion.auto_bridge import AutoBridge


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    parser.add_argument("--mode", choices=("current", "fix", "revert"), required=True)
    parser.add_argument(
        "--case",
        choices=("roundtrip", "leaf-valid", "leaf-dangling", "indexed-nested-symlink"),
        required=True,
    )
    parser.add_argument("--path-mode", choices=("direct", "symlink"), default="direct")
    parser.add_argument("--run-dir", type=Path, required=True)
    parser.add_argument("--result", type=Path, required=True)
    parser.add_argument("--port", required=True)
    return parser.parse_args()


def initialize_distributed(port: str) -> None:
    os.environ.update(
        MASTER_ADDR="127.0.0.1",
        MASTER_PORT=port,
        RANK="0",
        LOCAL_RANK="0",
        WORLD_SIZE="1",
    )
    torch.cuda.set_device(0)
    dist.init_process_group(
        backend="nccl",
        world_size=1,
        rank=0,
        timeout=datetime.timedelta(minutes=10),
    )
    parallel_state.initialize_model_parallel(
        tensor_model_parallel_size=1,
        pipeline_model_parallel_size=1,
        context_parallel_size=1,
    )


def make_config() -> Qwen2Config:
    config = Qwen2Config(
        vocab_size=128,
        hidden_size=64,
        intermediate_size=128,
        num_hidden_layers=1,
        num_attention_heads=4,
        num_key_value_heads=2,
        max_position_embeddings=128,
        tie_word_embeddings=True,
        torch_dtype=torch.bfloat16,
    )
    config.torch_dtype = torch.bfloat16
    return config


def fill_model(model: Qwen2ForCausalLM, value: float) -> None:
    with torch.no_grad():
        for parameter in model.parameters():
            parameter.fill_(value)


def provide_model(bridge: AutoBridge):
    provider = bridge.to_megatron_provider()
    provider.tensor_model_parallel_size = 1
    provider.pipeline_model_parallel_size = 1
    provider.finalize()
    return provider.provide_distributed_model(wrap_with_ddp=False)


def compare_models(expected, actual) -> dict[str, object]:
    expected_state = expected[0].state_dict()
    actual_state = actual[0].state_dict()
    missing = sorted(set(expected_state) - set(actual_state))
    unexpected = sorted(set(actual_state) - set(expected_state))
    unequal: list[str] = []
    maximum_absolute_error = 0.0
    compared = 0
    for key in sorted(set(expected_state) & set(actual_state)):
        expected_tensor = expected_state[key]
        actual_tensor = actual_state[key]
        if not isinstance(expected_tensor, torch.Tensor) or not isinstance(actual_tensor, torch.Tensor):
            continue
        compared += 1
        if expected_tensor.shape != actual_tensor.shape or not torch.equal(expected_tensor, actual_tensor):
            unequal.append(key)
            if expected_tensor.shape == actual_tensor.shape:
                maximum_absolute_error = max(
                    maximum_absolute_error,
                    (expected_tensor.float() - actual_tensor.float()).abs().max().item(),
                )
    return {
        "compared_tensor_count": compared,
        "missing_keys": missing,
        "unexpected_keys": unexpected,
        "unequal_keys": unequal,
        "maximum_absolute_error": maximum_absolute_error,
    }


def list_weight_entries(path: Path) -> list[dict[str, object]]:
    return [
        {
            "name": entry.name,
            "is_symlink": entry.is_symlink(),
            "exists_following_symlink": entry.exists(),
        }
        for entry in sorted(path.iterdir(), key=lambda item: item.name)
        if entry.name == "model.safetensors.index.json"
        or (entry.name.startswith("model") and entry.name.endswith(".safetensors"))
    ]


def main() -> None:
    args = parse_args()
    if args.result.exists() or args.run_dir.exists():
        raise FileExistsError(args.result if args.result.exists() else args.run_dir)
    args.run_dir.mkdir(parents=True)

    if args.mode == "revert":
        auto_bridge_module._existing_safetensors_files = lambda path: set()

    initialize_distributed(args.port)
    try:
        config = make_config()
        source_path = args.run_dir / "new-source"
        new_hf_model = Qwen2ForCausalLM(config).bfloat16()
        fill_model(new_hf_model, 0.25)
        new_hf_model.save_pretrained(source_path, safe_serialization=True, max_shard_size="5KB")

        real_export_path = args.run_dir / "real-export"
        real_export_path.mkdir()
        if args.path_mode == "symlink":
            export_path = args.run_dir / "export"
            export_path.symlink_to(real_export_path, target_is_directory=True)
        else:
            export_path = real_export_path

        config.save_pretrained(real_export_path)
        external_path = args.run_dir / "external-weight.safetensors"
        nested_target = real_export_path / "legacy-storage.safetensors"
        nested_link = real_export_path / "legacy" / "segment.safetensors"
        external_before: bytes | None = None
        if args.case == "roundtrip":
            stale_hf_model = Qwen2ForCausalLM(config).bfloat16()
            fill_model(stale_hf_model, -0.5)
            stale_hf_model.save_pretrained(real_export_path, safe_serialization=True, max_shard_size="5KB")
        elif args.case == "leaf-valid":
            from safetensors.torch import save_file

            save_file({"external.weight": torch.ones(2)}, external_path)
            external_before = external_path.read_bytes()
            (real_export_path / "model-00001-of-00001.safetensors").symlink_to(external_path)
        elif args.case == "leaf-dangling":
            (real_export_path / "model-00001-of-00001.safetensors").symlink_to(external_path)
        elif args.case == "indexed-nested-symlink":
            from safetensors.torch import save_file

            stale_hf_model = Qwen2ForCausalLM(config).bfloat16()
            fill_model(stale_hf_model, -0.5)
            stale_state = {
                name: tensor.detach().cpu().contiguous().clone()
                for name, tensor in stale_hf_model.state_dict().items()
            }
            save_file(stale_state, nested_target)
            nested_link.parent.mkdir()
            nested_link.symlink_to(nested_target)
            (real_export_path / "model.safetensors.index.json").write_text(
                json.dumps({"weight_map": dict.fromkeys(stale_state, "legacy/segment.safetensors")}),
                encoding="utf-8",
            )

        before_entries = list_weight_entries(real_export_path)

        source_bridge = AutoBridge.from_hf_pretrained(source_path)
        expected_megatron = provide_model(source_bridge)
        source_bridge.load_hf_weights(expected_megatron)

        export_bridge = AutoBridge.from_hf_config(config)
        export_bridge.save_hf_weights(expected_megatron, export_path, show_progress=False)
        after_entries = list_weight_entries(real_export_path)

        reloaded_bridge = AutoBridge.from_hf_pretrained(export_path)
        actual_megatron = provide_model(reloaded_bridge)
        reloaded_bridge.load_hf_weights(actual_megatron)

        stale_link = real_export_path / "model-00001-of-00001.safetensors"
        result = {
            "mode": args.mode,
            "case": args.case,
            "path_mode": args.path_mode,
            "entry": "AutoBridge.load_hf_weights -> AutoBridge.save_hf_weights -> AutoBridge.load_hf_weights",
            "model": {
                "architecture": "Qwen2ForCausalLM",
                "hidden_size": config.hidden_size,
                "intermediate_size": config.intermediate_size,
                "num_hidden_layers": config.num_hidden_layers,
                "vocab_size": config.vocab_size,
                "dtype": "bfloat16",
            },
            "topology": {"tp": 1, "pp": 1, "cp": 1, "world_size": 1, "backend": "nccl"},
            "before_entries": before_entries,
            "after_entries": after_entries,
            "stale_leaf_is_symlink": stale_link.is_symlink(),
            "stale_leaf_target_exists": stale_link.exists(),
            "external_target_exists": external_path.exists(),
            "external_target_unchanged": (
                external_path.read_bytes() == external_before if external_before is not None and external_path.exists() else None
            ),
            "nested_leaf_is_symlink": nested_link.is_symlink(),
            "nested_leaf_target_exists": nested_link.exists(),
            "nested_target_exists": nested_target.exists(),
            "index_exists": (real_export_path / "model.safetensors.index.json").exists(),
            "comparison": compare_models(expected_megatron, actual_megatron),
        }
        args.result.write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8")
        print(json.dumps(result, indent=2))
    finally:
        parallel_state.destroy_model_parallel()
        dist.destroy_process_group()


if __name__ == "__main__":
    main()
PY
