# [Fix] nixl: release transfer worker when room is aborted mid-wait

[Upstream PR](https://github.com/sgl-project/sglang/pull/44570) · state: open · merged: False

## Original PR description

## Motivation

Fixes #44369.

In PD disaggregation with the NIXL backend, aborting a request after its KV
transfer handles were submitted did not bound the prefill transfer worker's
wait when a handle stayed healthy but non-terminal (`PROC`):
`NixlKVManager._await_handles()` only arms its settle deadline after an `ERR`
is observed, so with every handle `PROC` the deadline stays `None` and the
wait is unbounded. The abort notification marks the room `Failed` and
registers the deferred ACK target, but it neither wakes a worker already
inside `_await_handles()` nor drains the room's `_staging_outstanding`
count — so the deferred abort ACK never fires and the worker thread never
returns to its queue. With the default `SGLANG_DISAGGREGATION_QUEUE_SIZE=4`,
four such aborted rooms can exhaust all prefill transfer workers.

## Modifications

- `python/sglang/srt/disaggregation/nixl/conn.py`
  - `_await_handles()` now accepts an optional `room` and returns
    `(settled, any_failed, aborted)`. When the room is marked `Failed` while
    polling (e.g. a decode-side abort notification landed after the batch
    was submitted), the wait ends immediately with `aborted=True` instead of
    spinning forever on healthy non-terminal handles. The exception-path
    call site passes no room, so its already-bounded (`failure_seen=True`)
    behavior is unchanged.
  - `transfer_worker()` handles the aborted wait: it releases the chunk's
    staging count (guarded by the existing `staging_counted` flag, mirroring
    the failure path) and fires the deferred abort ACK via
    `_maybe_ack_drained_abort()`, then continues with the next chunk. The
    room stays `Failed`, exactly as the abort notification left it.
- `test/registered/unit/disaggregation/test_nixl_deferred_kv_release.py`
  - Two new regression tests (see below); one existing mock updated for the
    new 3-tuple return.

Design note: the ACK fires once the worker's accounting drains, matching the
deferred-ACK protocol's intent ("send the deferred ack once an aborted
room's chunks have drained"). The decode side keeps its release timeout as a
backstop, as before.

## Reproducer

Deterministic and GPU-free: a fake NIXL agent whose handle stays `PROC`,
driving the real `_handle_abort_notification()` and (Phase B) the real
`transfer_worker()`. Save as `repro_44369.py` and run with
`PYTHONPATH=python python repro_44369.py` from a checkout.

```python
"""Reproducer for sgl-project/sglang issue #44369 (GPU-free, no RDMA)."""
import inspect
from collections import defaultdict
from threading import Event, Thread
from time import sleep, time
from types import SimpleNamespace

import numpy as np

from sglang.srt.disaggregation.base.conn import KVPoll
from sglang.srt.disaggregation.common.conn import AbortNotification
from sglang.srt.disaggregation.common.utils import TransferKVChunk
from sglang.srt.disaggregation.nixl.conn import NixlKVManager

ROOM = 90210
HANDLE = object()


class FakeAgent:
    def __init__(self):
        self.state = "PROC"
        self.reads = 0

    def check_xfer_state(self, handle):
        self.reads += 1
        return self.state


def make_manager(agent):
    manager = NixlKVManager.__new__(NixlKVManager)
    manager.agent = agent
    manager.request_status = {ROOM: KVPoll.WaitingForInput}
    manager.req_to_decode_prefix_len = {}
    manager.transfer_infos = {}
    manager.exceptions = {}
    manager._staging_outstanding = defaultdict(int)
    manager._deferred_ack_targets = {}
    manager._deferred_ack_poisoned_rooms = set()
    manager.enable_deferred_decode_kv_release = True
    manager.enable_staging = False
    manager._staging_ctx = None
    manager.sent_acks = []
    manager.update_status = lambda room, status: manager.request_status.__setitem__(
        room, status
    )
    manager.record_failure = lambda room, reason: None
    manager._send_abort_ack = lambda room, target: manager.sent_acks.append(
        (room, target)
    )
    manager._reload_invalidated_peers = lambda infos: None
    return manager


def abort_message():
    return AbortNotification(
        room=ROOM, decode_ip="127.0.0.1", decode_port=54321, generation=7
    ).to_zmq()


def await_kwargs(manager):
    if "room" in inspect.signature(manager._await_handles).parameters:
        return {"room": ROOM}
    return {}


def wait_until_polling(agent):
    deadline = time() + 10
    while agent.reads == 0 and time() < deadline:
        sleep(0.001)
    assert agent.reads > 0


def phase_a():
    print("=== Phase A: direct _await_handles vs abort ===")
    agent = FakeAgent()
    manager = make_manager(agent)
    manager._staging_outstanding[ROOM] = 1
    entered, returned = Event(), Event()
    result = {}

    def wait_for_handle():
        entered.set()
        result["value"] = manager._await_handles(
            [HANDLE], failure_seen=False, **await_kwargs(manager)
        )
        returned.set()

    thread = Thread(target=wait_for_handle, daemon=True)
    thread.start()
    assert entered.wait(5)
    wait_until_polling(agent)
    reads_before_abort = agent.reads
    assert manager._handle_abort_notification(abort_message())
    assert manager.request_status[ROOM] == KVPoll.Failed
    woke = returned.wait(2.0)
    sleep(0.1)
    print(
        {
            "waiter_returned_within_2s_after_abort": woke,
            "waiter_alive_after_abort": thread.is_alive(),
            "return_value": result.get("value"),
            "reads_before_abort": reads_before_abort,
            "reads_after_abort": agent.reads,
        }
    )
    agent.state = "DONE"
    returned.wait(3)
    thread.join(2)


def phase_b():
    print("=== Phase B: transfer_worker vs abort (accounting + ACK) ===")
    agent = FakeAgent()
    manager = make_manager(agent)
    manager.is_mla_backend = False
    manager.is_hybrid_mla_backend = False
    manager.attn_tp_size = 1
    manager.transfer_source_rank = 0
    manager.kv_args = SimpleNamespace(kv_data_ptrs=[0x1000])
    manager.decode_kv_args_table = {
        "agent0": SimpleNamespace(
            decode_tp_size=1,
            requires_dcp_relayout=False,
            kv_xfer_segments=None,
            dst_homogeneous_mem_kind="VRAM",
            dst_kv_ptrs=[0x2000],
            gpu_id=0,
        )
    }
    manager.transfer_infos = {
        ROOM: {
            "req0": SimpleNamespace(
                room=ROOM,
                is_dummy=False,
                agent_name="agent0",
                dst_kv_indices=np.array([0], dtype=np.int64),
                dst_aux_index=np.array([0], dtype=np.int64),
                decode_prefix_len=0,
            )
        }
    }
    manager.send_kvcache = lambda *a, **k: HANDLE
    chunk = TransferKVChunk(
        room=ROOM,
        prefill_kv_indices=np.array([1], dtype=np.int32),
        index_slice=slice(0, 1),
        is_last_chunk=False,
        chunk_id=0,
        prefill_aux_index=None,
        state_indices=None,
    )
    state = {"n": 0}

    def fake_get():
        state["n"] += 1
        if state["n"] == 1:
            return chunk
        raise KeyboardInterrupt()

    worker_done = Event()

    def run_worker():
        try:
            manager.transfer_worker(SimpleNamespace(get=fake_get))
        except KeyboardInterrupt:
            pass
        finally:
            worker_done.set()

    t = Thread(target=run_worker, daemon=True)
    t.start()
    wait_until_polling(agent)
    assert manager._handle_abort_notification(abort_message())
    finished = worker_done.wait(5.0)
    sleep(0.1)
    print(
        {
            "worker_finished_within_5s_after_abort": finished,
            "outstanding_after_abort": manager._staging_outstanding.get(ROOM, 0),
            "abort_ack_sent_after_abort": bool(manager.sent_acks),
            "acks": [(r, str(tgt)) for r, tgt in manager.sent_acks],
            "room_status": str(manager.request_status.get(ROOM)),
        }
    )
    agent.state = "DONE"
    worker_done.wait(3)
    t.join(2)


if __name__ == "__main__":
    phase_a()
    phase_b()
```
