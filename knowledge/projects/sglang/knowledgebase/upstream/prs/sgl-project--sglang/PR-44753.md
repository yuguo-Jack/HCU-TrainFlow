# [Refactor] RoPE: Support RotaryEmbedding class usage for DiT models

[Upstream PR](https://github.com/sgl-project/sglang/pull/44753) · state: open · merged: False

## Original PR description

## Motivation

Each DiT implements its own RoPE: the same flashinfer / AITER / eager cascade is written out four times across `wanvideo.py` and `lingbot_world.py`, and Z-Image has eleven RoPE call sites. `RotaryEmbedding` already exists to unify this. This PR moves Wan-Video, LingBot-World, Causal-Wan/LongLive2 and Z-Image onto the class and hoists each model's frequency table into its top-level `forward`, which also gets them the class's complex-table path off CUDA. CUDA keeps the
same flashinfer kernel.

## Modifications

1. `runtime/models/dits/wanvideo.py`
Removed the hand-written `CUDA`/`AITER` cascade from `WanTransformerBlock` and the `VSA` variant; both now hold a `RotaryEmbedding` and receive a `complex_freqs` table built once per `WanTransformer3DModel.forward`. The per-forward `rope_cos_sin_cache` plumbing is gone, forward_cuda rebuilds the same cache internally and calls the same `apply_flashinfer_rope_qk_inplace`.

2. `runtime/models/dits/lingbot_world.py`
Same change for `LingBotWorldCausalSelfAttention` and` LingBotWorldTransformerBlock`: the `_is_cuda` branch and the duplicated `cat([cos, sin])` cache build are replaced by one class call that takes cos/sin, the existing `cos_sin_cache` and `complex_freqs`. The `AITER` branch is untouched.

3. `runtime/models/dits/causal_wanvideo.py`
`CausalWanSelfAttention` gets a `RotaryEmbedding`; `complex_freqs` is built once in `CausalWanTransformer3DModel.forward` and threaded through the block as a keyword-only argument.

4. runtime/models/dits/longlive2.py
Forwards `complex_freqs` through the overridden block so the `LongLive2` subclass keeps the parent's new signature.

5. runtime/models/dits/zimage.py
Collapsed five `apply_qk_norm_with_optional_rope` call sites and six direct `RoPE` calls into two branches: fused QK-norm or no QK-norm goes to the class, otherwise `apply_qk_norm_with_optional_rope with` the cache it already had. `_get_rope_cache` lost its CUDA-only gate and now returns (cos_sin_cache, positions, freqs_complex) for all three groups (x / cap / unified), built once per forward by a new pure `_build_rope_cache`.

6. `runtime/models/dits/ming_image.py`
Builds the same three tables via `_build_rope_cache` and passes them down. Required: Ming reuses `Z-Image's` attention but has its own forward, so without this it would reach `apply_qk_norm_with_optional_rope(cos_sin_cache=None)` and get QK-norm without `RoPE`.

7. `runtime/layers/layernorm.py, runtime/layers/elementwise.py`
Fix per-frame [B, F, 1, C] modulation on NPU, which the branch hits before any `RoPE` code: the fused kernel flattens the modulation to one row, so `MulAdd.forward_npu` falls back to eager when it is wider, and the new `_expand_npu_modulation` broadcasts shift/scale to [B, L, C] instead of letting a plain multiply produce [B, F, L, C].

<summary><b>Measurement script</b> </summary>

Reuses the helpers in `multimodal_gen/test/server/realtime_consistency.py`.
`run` writes an mp4, a lossless frame dump and `chunk_stats`; `compare` prints
PSNR between two dumps.

```python
"""Headless LingBot-World realtime client: scripted camera, mp4 + frames + chunk_stats.

Run:
  python lingbot_rt_client.py run --tag main --image ref.jpg
  python lingbot_rt_client.py compare lingbot_out/main_frames.npz lingbot_out/branch_frames.npz
"""

import argparse
import asyncio
import json
from pathlib import Path

import numpy as np

from sglang.multimodal_gen.test.server.realtime_consistency import (
    collect_realtime_output,
    encode_realtime_frames_to_mp4,
    prepare_realtime_first_frame,
    summarize_realtime_perf_stats,
)


def parse_actions(spec: str) -> list[list[str]]:
    # "w:24,d:12,:12" -> 24 frames of ["w"], 12 of ["d"], 12 of [] (idle).
    actions: list[list[str]] = []
    for part in filter(None, spec.split(",")):
        keys, count = part.split(":")
        actions += [list(keys)] * int(count)
    return actions


def run(args) -> None:
    init = {
        "type": "init",
        "prompt": args.prompt,
        "first_frame": prepare_realtime_first_frame(args.image),
        "size": args.size,
        "fps": args.fps,
        "num_frames": args.num_frames,
        "seed": args.seed,
        "num_inference_steps": args.steps,
        "guidance_scale": 1.0,
        "max_chunks": args.num_chunks,
        # No realtime_output_format -> lossless delta-gzip RGB, needed for frame diff.
        "condition_inputs": {"camera_actions": parse_actions(args.actions)},
    }
    result = asyncio.run(
        collect_realtime_output(
            ws_url=args.url,
            init_payload=init,
            events=[],
            num_chunks=args.num_chunks,
            require_chunk_stats=True,
        )
    )

    out = Path(args.out_dir)
    out.mkdir(parents=True, exist_ok=True)
    (out / f"{args.tag}.mp4").write_bytes(
        encode_realtime_frames_to_mp4(result.frames, fps=args.fps)
    )
    np.savez_compressed(out / f"{args.tag}_frames.npz", frames=np.stack(result.frames))
    summary = summarize_realtime_perf_stats(
        result.chunk_stats, ignore_initial_chunks=args.ignore_initial_chunks
    )
    per_chunk = [
        {
            "chunk": s.chunk_index,
            "forward_ms": s.scheduler_forward_ms,
            "total_ms": s.chunk_total_ms,
        }
        for s in result.chunk_stats
    ]
    (out / f"{args.tag}_stats.json").write_text(
        json.dumps({"summary": summary, "per_chunk": per_chunk}, indent=2)
    )
    print(f"[{args.tag}] frames={len(result.frames)}")
    for key in (
        "avg_scheduler_forward_ms",
        "p95_scheduler_forward_ms",
        "avg_chunk_total_ms",
    ):
        print(f"  {key:28s} {summary[key]:9.1f}")
    fps_gen = (
        summary["total_frames"]
        / sum(s.chunk_total_ms for s in result.chunk_stats)
        * 1e3
    )
    print(f"  {'generated_fps (all chunks)':28s} {fps_gen:9.2f}")


def compare(args) -> None:
    a = np.load(args.a)["frames"].astype(np.float32)
    b = np.load(args.b)["frames"].astype(np.float32)
    n = min(len(a), len(b))
    if len(a) != len(b):
        print(f"frame count differs: {len(a)} vs {len(b)}, comparing first {n}")
    diff = a[:n] - b[:n]
    mse = (diff**2).reshape(n, -1).mean(axis=1)
    psnr = 10 * np.log10(255.0**2 / np.maximum(mse, 1e-10))
    print(
        f"max_abs_diff={np.abs(diff).max():.0f}  "
        f"mean_abs_diff={np.abs(diff).mean():.3f}"
    )
    print(f"PSNR dB: min={psnr.min():.2f}  mean={psnr.mean():.2f}  (100 = identical)")
    print("PSNR per frame:", " ".join(f"{p:.1f}" for p in psnr))


def main() -> None:
    parser = argparse.ArgumentParser()
    sub = parser.add_subparsers(dest="cmd", required=True)

    r = sub.add_parser("run")
    r.add_argument("--tag", required=True)
    r.add_argument("--url", default="ws://127.0.0.1:30000/v1/realtime_video/generate")
    r.add_argument("--image", required=True, help="first frame: local path or URL")
    r.add_argument(
        "--prompt",
        default="A slow aerial orbit around a pastel floating island hotel in the open "
        "ocean, hazy sunlight, turquoise water, clean horizon, cinematic but playful.",
    )
    r.add_argument("--size", default="832x480")
    r.add_argument("--fps", type=int, default=16)
    r.add_argument("--num-frames", type=int, default=9, help="frames per chunk")
    r.add_argument("--num-chunks", type=int, default=12)
    r.add_argument("--steps", type=int, default=4)
    r.add_argument("--seed", type=int, default=42)
    r.add_argument("--actions", default="w:48,d:24,:36", help="per-frame camera script")
    r.add_argument("--ignore-initial-chunks", type=int, default=2)
    r.add_argument("--out-dir", default="lingbot_out")
    r.set_defaults(func=run)

    c = sub.add_parser("compare")
    c.add_argument("a")
    c.add_argument("b")
    c.set_defaults(func=compare)

    args = parser.parse_args()
    args.func(args)


if __name__ == "__main__":
    main()
```
