# [Perf] tokenizer_manager: Reduce incremental streaming metadata scans

[Upstream PR](https://github.com/sgl-project/sglang/pull/46372) · state: open · merged: False

## Original PR description

## Motivation

When incremental streaming has multiple queued chunks, `_coalesce_streaming_chunks` checks each optional metadata key with `any(...)`, then scans the chunk metadata again to flatten that key. The flattening pass also constructs a default empty list for each `dict.get` call.

This addresses the backlog metadata traversal described by [SimBuddy](https://github.com/SimBuddy) in #36226, item 5. It is a small implementation against current main; the other items in that issue are outside this PR.

## Modifications

Collect the lists for each present key once, then flatten those lists. Keep the existing absent-versus-empty distinction: `values = [[]]` must still produce an empty output field. Preserve token order, sparse custom fields, last-chunk cumulative metadata, and the input containers. Single-chunk and cumulative-stream dispatch are unchanged.

The production diff is 3 added lines and 4 removed lines, with no new helper, import, setting, or dependency. Add CPU behavior guards through the real `_wait_one_response` / `_stream_one_response` path.

```python
"""Compare complete, normally imported production modules with bounded payloads.

This local acceptance benchmark is deliberately not a timing-sensitive CI test.
The baseline is an unchanged copy of tokenizer_manager.py from the recorded base.
"""

import argparse
import asyncio
import copy
import hashlib
import importlib.util
import inspect
import json
import math
import os
import platform
import statistics
import sys
import time
from pathlib import Path

from sglang.srt.managers import tokenizer_manager as candidate
from sglang.srt.managers.io_struct import GenerateReqInput
from sglang.srt.observability.req_time_stats import APIServerReqTimeStats
from sglang.srt.utils.json_response import dumps_json

parser = argparse.ArgumentParser()
parser.add_argument("--output", required=True)
parser.add_argument("--rounds", type=int, default=9)
parser.add_argument("--iterations", type=int, default=1500)
parser.add_argument("--require-speedup", action="store_true")
args = parser.parse_args()

baseline_path = Path(__file__).with_name("tokenizer_manager_baseline.py")
spec = importlib.util.spec_from_file_location("coalesce_original_production", baseline_path)
baseline = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = baseline
spec.loader.exec_module(baseline)

TOKEN_FIELDS = {
    "output_token_logprobs": [[-0.5, 7, "x"]],
    "output_top_logprobs": [[[-0.5, 7, "x"], [-1.5, 8, "y"]]],
    "output_token_ids_logprobs": [[[-0.5, 7, "x"]]],
    "output_token_sampling_mask": [[7, 8]],
    "output_token_sampling_logprobs": [-0.5],
}


def make_payload(count, mode, tokens=1):
    chunks = []
    for i in range(count):
        meta = {
            "id": "bounded-coalesce-check", "finish_reason": None,
            "prompt_tokens": 12, "completion_tokens": (i + 1) * tokens,
            "cached_tokens": 0, "e2e_latency": 0.01,
            "input_token_logprobs": [[None, 2, "hello"]],
        }
        if mode == "empty":
            meta.update({key: [] for key in TOKEN_FIELDS})
        elif mode == "full" or (mode == "sparse" and i == count - 1):
            meta.update({key: value * tokens for key, value in TOKEN_FIELDS.items()})
        elif mode == "custom" and i in (0, count - 1):
            meta["custom_scores"] = [i]
        chunks.append({"text": "x" * tokens, "output_ids": [7] * tokens, "meta_info": meta})
    return chunks


def setup(module, custom_keys):
    manager = module.TokenizerManager.__new__(module.TokenizerManager)
    manager.incremental_streaming_output = True
    obj = GenerateReqInput(text="hello", rid="bounded-coalesce-check", stream=True)
    state = module.ReqState(
        out_list=[], finished=False, event=asyncio.Event(), obj=obj,
        time_stats=APIServerReqTimeStats(),
    )
    state.time_stats.set_response_sent_to_client_time()
    state.customized_info_accumulated = {key: [] for key in custom_keys}
    return manager, obj, state


async def consume(setup_items, chunks, serialize=False):
    manager, obj, state = setup_items
    state.out_list = chunks
    state.event.set()
    stream = manager._stream_one_response(obj, state)
    out = await anext(stream)
    await stream.aclose()
    if serialize:
        return b"data: " + dumps_json(out) + b"\n\n"
    return out


async def measure(setup_items, chunks, custom_keys, path):
    manager, obj, _ = setup_items
    start = time.perf_counter_ns()
    if path == "direct":
        for _ in range(args.iterations):
            manager._coalesce_streaming_chunks(chunks, obj.rid, custom_keys)
    else:
        for _ in range(args.iterations):
            await consume(setup_items, chunks, serialize=path == "consumer_json")
    return (time.perf_counter_ns() - start) / args.iterations / 1000


async def main():
    if hasattr(os, "sched_getaffinity"):
        os.sched_setaffinity(0, {min(os.sched_getaffinity(0))})
    result = {"python": platform.python_version(), "platform": platform.platform(),
        "baseline_source": str(baseline_path),
        "candidate_source": inspect.getfile(candidate.TokenizerManager),
        "baseline_sha256": hashlib.sha256(baseline_path.read_bytes()).hexdigest(),
        "candidate_sha256": hashlib.sha256(Path(inspect.getfile(candidate.TokenizerManager)).read_bytes()).hexdigest(),
        "rounds": args.rounds, "iterations": args.iterations, "cases": []}
    cases = [(n, mode, 1) for n in (2, 8, 16)
             for mode in ("absent", "empty", "sparse", "full", "custom")]
    cases += [(8, "full", 8), (16, "full", 8)]
    for count, mode, tokens in cases:
        chunks = make_payload(count, mode, tokens)
        original = copy.deepcopy(chunks)
        custom_keys = ("custom_scores", "hidden_scores") if mode == "custom" else ()
        old, new = setup(baseline, custom_keys), setup(candidate, custom_keys)
        old_out = await consume(old, chunks)
        new_out = await consume(new, chunks)
        assert new_out == old_out
        assert new_out["text"] == "x" * (count * tokens)
        assert new_out["output_ids"] == [7] * (count * tokens)
        assert new_out["meta_info"]["completion_tokens"] == count * tokens
        assert new_out["meta_info"]["input_token_logprobs"] == [[None, 2, "hello"]]
        assert "hidden_scores" not in new_out["meta_info"]
        if mode == "absent":
            assert not set(TOKEN_FIELDS).intersection(new_out["meta_info"])
        if mode == "empty":
            assert all(new_out["meta_info"][key] == [] for key in TOKEN_FIELDS)
        if mode == "custom":
            assert new_out["meta_info"]["custom_scores"] == [0, count - 1]
        assert chunks == original
        assert not new[2].event.is_set() and new[2].out_list == []
        old_wire, new_wire = await consume(old, chunks, True), await consume(new, chunks, True)
        # JSON object member order is not an API contract.
        assert json.loads(old_wire[6:-2]) == json.loads(new_wire[6:-2])
        row = {"chunks": count, "mode": mode, "tokens_per_chunk": tokens, "checks": "passed", "paths": {}}
        for path in ("direct", "consumer", "consumer_json"):
            await measure(old, chunks, custom_keys, path)
            await measure(new, chunks, custom_keys, path)
            old_times, new_times = [], []
            for repeat in range(args.rounds):
                for label in (("old", "new") if repeat % 2 == 0 else ("new", "old")):
                    elapsed = await measure(old if label == "old" else new, chunks, custom_keys, path)
                    (old_times if label == "old" else new_times).append(elapsed)
            old_median, new_median = statistics.median(old_times), statistics.median(new_times)
            row["paths"][path] = {"baseline_us": old_median, "candidate_us": new_median,
                "speedup": old_median / new_median,
                "baseline_samples_us": old_times, "candidate_samples_us": new_times}
        result["cases"].append(row)
        p = row["paths"]
        print(f"n={count:2} {mode:6} tokens={tokens} direct={p['direct']['speedup']:.3f}x "
              f"consumer={p['consumer']['speedup']:.3f}x json={p['consumer_json']['speedup']:.3f}x", flush=True)
    result["geomean_speedup"] = {path: math.exp(statistics.mean(
        math.log(case["paths"][path]["speedup"]) for case in result["cases"]))
        for path in ("direct", "consumer", "consumer_json")}
    result["acceptance"] = {
        "direct_at_least_10_percent_faster": result["geomean_speedup"]["direct"] >= 1.10,
        "serialized_consumer_at_least_5_percent_faster": result["geomean_speedup"]["consumer_json"] >= 1.05,
        "no_case_over_5_percent_slower": all(case["paths"][path]["speedup"] >= 0.95
            for case in result["cases"] for path in ("direct", "consumer", "consumer_json")),
    }
    Path(args.output).write_text(json.dumps(result, indent=2))
    print(json.dumps({key: result[key] for key in ("geomean_speedup", "acceptance")}, indent=2))
    if args.require_speedup:
        assert all(result["acceptance"].values()), "Performance target not met; do not claim a verified optimization"


asyncio.run(main())
```
