## Summary

- add a regression test so Responses API background jobs do not silently fall back to foreground disconnect handling after slicing

    def test_getitem_preserves_background(self):
        """Batch subrequests must keep the background response flag and other request-level flags."""
        req = GenerateReqInput(
            text=["Hello", "World"],
            sampling_params=[{}, {}],
            rid=["id1", "id2"],
            background=True,
            require_reasoning=True,
            use_audio_in_video=True,
            routing_key="test_key",
        )
        req.normalize_batch_and_arguments()

        self.assertTrue(req[0].background)
        self.assertTrue(req[1].background)
        self.assertTrue(req[0].require_reasoning)
        self.assertTrue(req[1].require_reasoning)
        self.assertTrue(req[0].use_audio_in_video)
        self.assertTrue(req[1].use_audio_in_video)
        self.assertEqual(req[0].routing_key, "test_key")
