# [Fix] benchmark: Compensate cumulative request arrival drift

[Upstream PR](https://github.com/sgl-project/sglang/pull/44162) · state: open · merged: False

## Original PR description

## Motivation

Fixes #43930. Finite-rate, non-trace request generation currently starts each exponential sleep after the consumer resumes the generator, so consumer work and late wake-ups accumulate into a lower offered load.

## Modifications

- Accumulate monotonic absolute arrival deadlines and sleep only for the remaining time before yielding each request. Overdue arrivals catch up without resetting the schedule.
- Keep the first request immediate, request order, and post-yield exponential draws, including the final unused draw. This preserves seeded effective gaps, interleaving with consumer NumPy draws (such as skewed LoRA selection), and the RNG state after full consumption.
- Remove the terminal sleep because it schedules no request and unnecessarily extends benchmark duration. Infinity and trace replay retain their behavior.
- Add registered CPU regressions for oversleep, consumer delays, catch-up, seeded RNG interleaving, empty/singleton, infinity, and trace slowdown.

```python
"""Measure generator yield and consumer-created task dispatch, never socket sends."""
import argparse
import asyncio
import json
import platform
import sys
import time
from unittest.mock import patch

import numpy as np
from sglang.benchmark.serving import get_request


async def measure(count, rate, seed, consumer_delay):
    np.random.seed(seed)
    gaps, yielded, dispatched, tasks = [], [], [], []
    exponential = np.random.exponential

    def draw(scale):
        gap = exponential(scale)
        gaps.append(gap)
        return gap

    async def dispatch():
        dispatched.append(time.perf_counter())

    with patch("numpy.random.exponential", draw):
        async for _ in get_request(list(range(count)), rate):
            yielded.append(time.perf_counter())
            tasks.append(asyncio.create_task(dispatch()))
            if consumer_delay:
                time.sleep(consumer_delay)
    exhausted = time.perf_counter()
    await asyncio.gather(*tasks)
    assert len(gaps) == count
    assert len(yielded) == len(dispatched) == count
    planned = np.r_[0.0, np.cumsum(gaps[:-1])]
    observed = np.asarray(yielded) - yielded[0]
    dispatch_times = np.asarray(dispatched) - yielded[0]
    lag = (observed - planned) * 1000
    dispatch_lag = (dispatch_times - planned) * 1000
    return {
        "rate": rate, "count": count, "seed": seed,
        "consumer_delay_s": consumer_delay,
        "scheduled_rate": (count - 1) / planned[-1],
        "yield_rate": (count - 1) / observed[-1],
        "dispatch_rate": (count - 1) / (dispatch_times[-1] - dispatch_times[0]),
        "last_yield_lag_ms": float(lag[-1]),
        "p95_yield_lag_ms": float(np.percentile(lag, 95)),
        "last_dispatch_lag_ms": float(dispatch_lag[-1]),
        "p95_dispatch_lag_ms": float(np.percentile(dispatch_lag, 95)),
        "exhaustion_after_last_yield_ms": (exhausted - yielded[-1]) * 1000,
        "gaps": gaps,
    }


async def main(args):
    loop = asyncio.get_running_loop()
    output = {
        "source": get_request.__code__.co_filename,
        "os": platform.platform(), "machine": platform.machine(),
        "python": sys.version, "numpy": np.__version__,
        "loop": type(loop).__name__,
        "selector": type(loop._selector).__name__,
        "runs": [],
    }
    for rate in args.rates:
        for _ in range(args.repeats):
            output["runs"].append(await measure(args.count, rate, 0, args.consumer_delay))
    print(json.dumps(output, indent=2))


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--count", type=int, default=500)
    parser.add_argument("--rates", type=float, nargs="+", default=[100.0, 400.0])
    parser.add_argument("--repeats", type=int, default=5)
    parser.add_argument("--consumer-delay", type=float, default=0.0005)
    asyncio.run(main(parser.parse_args()))
```
