```python
"""Reproduce every stack tree from its pre-rename tree and compare all bytes.

Usage: python3 reproduce.py --repo /path/to/sglang 
Requires git and the repository's pre-commit environments. Fetch the base and
target SHAs embedded below into the source clone before running this script.
"""
import argparse
import json
import os
import pathlib
import subprocess
import tempfile

"""Rename the two Muse Spark kernel trees and all references, with no code edits."""
import pathlib
import subprocess

OLD = b"muse_spark_v12"
NEW = b"muse_spark"
DIRECTORIES = ("python/sglang/kernels/jit/csrc/", "python/sglang/kernels/ops/")


def transform(root):
    root = pathlib.Path(root)
    paths = subprocess.check_output(["git", "ls-files", "-z"], cwd=root).decode().split("\0")
    changed = []
    for relative in filter(None, paths):
        path = root / relative
        if path.is_symlink() or not path.is_file():
            continue
        source = path.read_bytes()
        target = source.replace(OLD, NEW)
        if target != source:
            path.write_bytes(target)
            changed.append(relative)
    moved = []
    for prefix in DIRECTORIES:
        source = root / (prefix + OLD.decode())
        target = root / (prefix + NEW.decode())
        assert source.is_dir() and not target.exists(), (source, target)
        moved.extend(str(path.relative_to(root)) for path in source.rglob("*") if path.is_file())
        source.rename(target)
    return dict(references=changed, moved=moved)





def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--repo', required=True)
    args = parser.parse_args()
    records = [{'pr': 44665, 'base': '359030dbac3aeab909533411583646790c3ebcb9', 'target': 'a2ac2718795b16a10a6e221eefa38e234abcc9e0'}, {'pr': 44666, 'base': 'f60333c7412ef5001b4e7fbab13dcb7aa1b87993', 'target': '90785dc87625d82e18c148826f99e01fefdbb503'}, {'pr': 44667, 'base': 'caf6b36a44e01a91a1f21524ca086b04af4b3933', 'target': 'cbd25ea6497f6dcec269af385fa374e75b9e4e13'}, {'pr': 44668, 'base': '51abebf433801bd6bcfadedf398a5b9a3c27dbf3', 'target': 'e297cab65cafffc94f3224d49858820affac7019'}]
    env = dict(os.environ, SKIP='no-commit-to-branch')
    with tempfile.TemporaryDirectory(prefix='muse-spark-rename-proof-') as temporary:
        checkout = pathlib.Path(temporary) / 'repo'
        subprocess.run(['git', 'clone', '--shared', '--no-checkout', '--quiet',
                        str(pathlib.Path(args.repo).resolve()), str(checkout)], check=True)

        def git(*argv):
            return subprocess.check_output(['git', *argv], cwd=checkout, env=env).decode().strip()

        for record in records:
            git('checkout', '--detach', '--quiet', record['base'])
            transform(checkout)
            git('add', '--all')
            changed = git('diff', '--cached', '--name-only', '--diff-filter=ACMR').splitlines()
            for attempt in range(2):
                result = subprocess.run(['pre-commit', 'run', '--files', *changed],
                                        cwd=checkout, env=env, capture_output=True, text=True)
                if result.returncode == 0:
                    break
                if attempt == 1:
                    raise RuntimeError(result.stdout + result.stderr)
                git('add', '--all')
            git('add', '--all')
            actual = git('write-tree')
            expected = git('rev-parse', record['target'] + '^{tree}')
            if actual != expected:
                raise AssertionError(git('diff', expected, actual))
            print(f"PASS #{record['pr']}: full tree reproduced exactly ({actual})", flush=True)
            git('reset', '--hard', record['target'])


if __name__ == '__main__':
    main()

```
