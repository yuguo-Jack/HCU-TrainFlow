# [Fix] diffusion: Unmerge LoRA layers the next adapter does not cover

[Upstream PR](https://github.com/sgl-project/sglang/pull/44770) · state: closed · merged: True

## Original PR description

## Problem
If adapter A is merged into a layer and `set_lora` switches to adapter B that does not cover that layer, the layer is only marked `disable_lora`. Its base weight still holds A's merged delta, so `forward` keeps applying A. `unmerge_lora_weights` also returned early on `disable_lora`, so deactivate/unmerge could not remove it either.

## Fix
- `_apply_lora_to_layers`: a layer no new adapter covers is unmerged (if merged) and cleared before being disabled. Clearing matters: `merge_lora_weights` re-enables any layer that still has `lora_A` set.
- `unmerge_lora_weights` guards on `merged` only. All callers already check `merged` first.
- The existing clearing block in `set_lora_weights` moved into `clear_lora_weights()` so both paths share it.

On main, A's delta survives both the switch and the unmerge. With this PR the output matches base exactly.

## Note
When an adapter does not cover every layer, A -> B -> disable -> B now takes the full re-apply path instead of `_reactivate_cached_dynamic_lora_layers`. Output is the same; only that narrow case is slower.

```python
"""GPU e2e: does a merged LoRA leak into layers the next adapter skips?

Runs the real DiffGenerator (Wan2.1-T2V-1.3B) once on origin/main, then again
with the patched linear.py / pipeline.py copied over the install.
"""

import modal

COMMIT = "ed2ebad342"
REPO = "/sglang"
RT = f"{REPO}/python/sglang/multimodal_gen/runtime"
PATCH = "<checkout of this PR>/python/sglang/multimodal_gen/runtime"

image = (
    modal.Image.from_registry("nvidia/cuda:12.8.1-devel-ubuntu22.04", add_python="3.11")
    .apt_install("git", "ffmpeg", "libgl1", "libglib2.0-0")
    .run_commands(
        f"git clone https://github.com/sgl-project/sglang.git {REPO}",
        f"cd {REPO} && git checkout {COMMIT}",
        f"cd {REPO} && SGLANG_BUILD_RUST_EXTS=none pip install -e 'python[diffusion]'",
    )
    .env({"HF_HOME": "/hf", "PYTORCH_CUDA_ALLOC_CONF": "expandable_segments:True"})
    .add_local_file(f"{PATCH}/layers/lora/linear.py", "/patch/linear.py")
    .add_local_file(f"{PATCH}/pipelines_core/lora/pipeline.py", "/patch/pipeline.py")
)
hf = modal.Volume.from_name("hf-cache-lora-leak", create_if_missing=True)
app = modal.App("sglang-lora-leak", image=image)

WORKER = r'''
import json, sys, numpy as np, torch
from safetensors.torch import load_file, save_file
from sglang.multimodal_gen import DiffGenerator

# Synthetic adapter A sized for Wan2.1-1.3B (dim 1536, 30 blocks).
g = torch.Generator().manual_seed(0)
sd = {}
for i in range(30):
    for proj in ("q", "k", "v", "o"):
        base = f"diffusion_model.blocks.{i}.self_attn.{proj}"
        sd[base + ".lora_A.weight"] = (torch.randn(4, 1536, generator=g) * 0.05).to(torch.bfloat16)
        sd[base + ".lora_B.weight"] = (torch.randn(1536, 4, generator=g) * 0.05).to(torch.bfloat16)
a_path = "/tmp/lora_a.safetensors"
save_file(sd, a_path)
# Adapter B covers one layer of A with a zero up-projection: no delta of its
# own, and every other A layer is left uncovered.
down = "diffusion_model.blocks.0.self_attn.q.lora_A.weight"
up = down.replace("lora_A", "lora_B")
b = {down: sd[down].clone(), up: torch.zeros_like(sd[up])}
b_path = "/tmp/lora_b.safetensors"
save_file(b, b_path)
print("A keys", len(sd), "B keys", list(b), flush=True)

gen = DiffGenerator.from_pretrained(
    model_path="Wan-AI/Wan2.1-T2V-1.3B-Diffusers", num_gpus=1, layerwise_offload_components=["text_encoder", "vae"],
    pin_cpu_memory=False
)
kw = dict(prompt="a red ball on a table", height=256, width=256, num_frames=5,
          num_inference_steps=4, seed=0, save_output=False)

def frames():
    r = gen.generate(sampling_params_kwargs=dict(kw))
    r = r[0] if isinstance(r, list) else r
    return np.asarray(r.frames, dtype=np.float32)

def drift(x, y):
    return float(np.abs(x - y).max())

base = frames()
gen.set_lora("a", a_path, merge_mode="merge")
with_a = frames()
gen.set_lora("b", b_path, merge_mode="merge")
after_switch = frames()
gen.unmerge_lora_weights()
after_unmerge = frames()
out = {
    "a_vs_base": drift(with_a, base),
    "switch_to_b_vs_base": drift(after_switch, base),
    "unmerge_vs_base": drift(after_unmerge, base),
}
print("RESULT", json.dumps(out), flush=True)
'''


@app.function(gpu="L4", timeout=3600, volumes={"/hf": hf}, memory=32768)
def run():
    import json, shutil, subprocess

    def once(label):
        import textwrap
        with open("/tmp/worker.py", "w") as f:
            f.write("if __name__ == '__main__':\n" + textwrap.indent(WORKER, "    "))
        p = subprocess.run(["python", "/tmp/worker.py"], capture_output=True, text=True)
        hf.commit()
        tail = "\n".join(l for l in (p.stdout + p.stderr).splitlines() if "Error" in l or "error" in l or "Traceback" in l)[-4000:]
        res = [l for l in p.stdout.splitlines() if l.startswith("RESULT ")]
        print(f"===== {label} rc={p.returncode}\n{tail if not res else res[0]}")
        return json.loads(res[0][7:]) if res else None

    subprocess.run(["nvidia-smi", "--query-gpu=name,memory.total", "--format=csv"])
    main = once("origin/main")
    shutil.copy("/patch/linear.py", f"{RT}/layers/lora/linear.py")
    shutil.copy("/patch/pipeline.py", f"{RT}/pipelines_core/lora/pipeline.py")
    print(subprocess.run(["git", "-C", REPO, "diff", "--stat"], capture_output=True, text=True).stdout)
    fixed = once("patched")
    return {"main": main, "patched": fixed}


@app.local_entrypoint()
def main():
    print(run.remote())
```
