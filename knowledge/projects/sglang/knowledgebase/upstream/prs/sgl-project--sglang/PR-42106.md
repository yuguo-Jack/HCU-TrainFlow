```
#!/usr/bin/env bash
# WildSongBench generation driver: 192 prompts x 2 candidates via the sglang curl API.
# Resumable: finalized candidates are skipped.
set -euo pipefail

SERVER="${SERVER:-http://127.0.0.1:30065}"
WSB_ROOT="$(cd "$(dirname "$0")" && pwd)"
MANIFEST="$WSB_ROOT/reproduction_manifest.jsonl"
RUNS="${RUNS:-$WSB_ROOT/runs/wsb}"
SONGS="$RUNS/songs"
CANDIDATES="${CANDIDATES:-2}"
mkdir -p "$SONGS"

# NOTE (yiakwy) : please download the original repo https://github.com/multimodal-art-projection/YuE
EXAMPLES=/home/yiakwy/workspace/Github/YuE/examples

MODEL_DIR=/raid/yiakwy/YuE2-3B

VAE_DIR=/root/.cache/huggingface/hub/models--m-a-p--YuE2-Vae-legacy/snapshots/b54118f0fc462f08999d1ec07e88817f4ee3f770

RUNTIME_SHA=$(jq -r .runtime_sha256 /tmp/opencode/wsb_runtime.json)

if ! curl -s --max-time 5 "$SERVER/health" > /dev/null; then
    echo "ERROR: no reachable server at $SERVER" >&2
    exit 1
fi

# static evaluation_inputs.jsonl (format expected by wsb_eval.py score)
python3 - "$MANIFEST" "$CANDIDATES" "$RUNS" <<'EOF'
import json, sys
from pathlib import Path
rows = [json.loads(l) for l in Path(sys.argv[1]).read_text().splitlines() if l.strip()]

cands = int(sys.argv[2]); runs = Path(sys.argv[3])

out = []
for row in rows:
    for j, seed in enumerate(row["candidate_ar_seeds"][:cands]):
        out.append({"prompt_index": row["prompt_index"], "candidate_index": j,
                    "seed": seed, "path": f"songs/wsb_{row['prompt_index']:03d}_r{j}_s{seed}"})
(runs / "evaluation_inputs.jsonl").write_text("".join(json.dumps(r) + "\n" for r in out))
print(f"evaluation_inputs.jsonl: {len(out)} rows")
EOF

total=0; done_n=0
while read -r row; do
    idx=$(jq -r .prompt_index <<<"$row")

    style=$(jq -Rs . <<<"$row" | jq -r .style 2>/dev/null || true)
    style=$(jq -r .style <<<"$row")

    lyrics=$(jq -r .lyrics <<<"$row")

    for j in 0 1; do
        [ "$j" -ge "$CANDIDATES" ] && continue
        seed=$(jq -r ".candidate_ar_seeds[$j]" <<<"$row")
        ident=$(printf "wsb_%03d_r%d_s%d" "$idx" "$j" "$seed")
        total=$((total + 1))
        cand="$SONGS/$ident"
        if [ -f "$cand/result.json" ]; then
            done_n=$((done_n + 1))
            continue
        fi

        mkdir -p "$cand"

        payload="$(jq -n --arg style "$style" --arg lyrics "$lyrics" \
            --arg cot "full" --arg seed "$seed" --arg out "$cand" \
            --arg id "$ident" \
            '{style:$style, lyrics:$lyrics, cot:$cot, abc:null,
              cfg_scale:1.0, seed:($seed|tonumber),
              ode_steps:32, semantic_max_tokens:9000, abc_max_tokens:4096,
              vae_core_frames:1024, vae_halo_frames:16,
              output_path:$out, output_file_name:"audio.wav",
              artifacts_dir:$out, id:$id}')"

        if ! curl -s --max-time 900 -X POST "$SERVER/v1/audio/generations" \
                -H 'Content-Type: application/json' -d "$payload" \
                -o "/tmp/opencode/wsb_resp.json"; then
            echo "GENERATION FAILED: $ident" >&2
            continue
        fi

        sleep 2  # NFS: let server-side writes become visible to this process

        python3 "$WSB_ROOT/finalize_candidate.py" "$cand" "$cand/audio.wav" \
            "$MODEL_DIR" "$VAE_DIR" "$RUNTIME_SHA" false \
            || { echo "FINALIZE FAILED: $ident" >&2; continue; }
        done_n=$((done_n + 1))
        echo "[$done_n/$total] $ident ok"
    done
done < "$MANIFEST"

echo "generation complete: $done_n/$total candidates finalized"
```

```bash
#!/usr/bin/env bash
# YuE2 task-set curl benchmark: the four full-music manual tasks.
set -euo pipefail
SERVER="${SERVER:-http://127.0.0.1:30065}"
OUT="${OUT:-/tmp/opencode/curl_tasks}"

BENCH_ROOT="$(cd "$(dirname "$0")/.." && pwd)"
BENCH_OUT="${BENCH_OUT:-$BENCH_ROOT/outputs/bench}"
mkdir -p "$BENCH_OUT" "$OUT"

if ! curl -s --max-time 5 "$SERVER/health" > /dev/null; then
    echo "ERROR: no reachable server at $SERVER" >&2
    echo "  start one:  bash yue2_tests/serve_yue2.sh            (port 30063)" >&2
    echo "  or point at an existing one:  SERVER=http://127.0.0.1:<port> bash $0" >&2
    exit 1
fi
echo "server: $SERVER (health OK)"

# NOTE (yiakwy) : please download YuE
EXAMPLES=/home/yiakwy/workspace/Github/YuE/examples

STYLE="$(jq -Rs . <<<"$(jq -r .style "$EXAMPLES/song.json")")"
LYRICS="$(jq -Rs . <<<"$(jq -r .lyrics "$EXAMPLES/song.json")")"

task_curl() {  # name cot abc_file n_parallel reps
    local name="$1" cot="$2" abc_file="$3" npar="$4" reps="$5"
    local abc_json="null"
    [[ -n "$abc_file" && -f "$abc_file" ]] && abc_json="$(jq -Rs . < "$abc_file")"
    local payload
    local out_dir="$BENCH_OUT"
    payload="$(jq -n --arg style "$STYLE" --arg lyrics "$LYRICS" --arg cot "$cot" \
        --argjson abc "$abc_json" --arg out "$out_dir" --arg fname "bench_$name.wav" \
        '{style:$style, lyrics:$lyrics, cot:$cot, abc:$abc, seed:831001,
          ode_steps:32, semantic_max_tokens:9000, abc_max_tokens:4096,
          vae_core_frames:1024, vae_halo_frames:16,
          output_path:$out, output_file_name:$fname}')"
    : > "$OUT/${name}_lat.txt"
    local wall_begin wall_end
    wall_begin=$(date +%s.%N)
    seq 1 "$reps" | xargs -P "$npar" -I{} curl -s -o "$OUT/${name}_resp_{}.json" \
        -w '%{time_total}\n' --max-time 600 -X POST "$SERVER/v1/audio/generations" \
        -H 'Content-Type: application/json' -d "$payload" >> "$OUT/${name}_lat.txt"
    wall_end=$(date +%s.%N)
    echo "$npar $(echo "$wall_end $wall_begin" | awk '{printf "%.3f", $1-$2}')" > "$OUT/${name}_wall.txt"

    for f in "$OUT/${name}_resp_"*.json; do
        jq -r '[.inference_time_s, .sample_rate, .audio] | @tsv' "$f" >> "$OUT/${name}_meta.txt" 2>/dev/null || true
    done
    rm -f "$OUT/${name}_resp_"*.json
}

stats() {
    local file="$1" label="$2"
    sort -n "$file" | awk -v label="$label" '
    {v[NR]=$1} END {
        p50=v[int((NR+1)/2)]; mean=0
        for(i=1;i<=NR;i++) mean+=v[i]; mean/=NR
        printf "%-22s n=%d  p50=%.3fs  mean=%.3fs  min=%.3fs  max=%.3fs\n",
               label, NR, p50, mean, v[1], v[NR]
    }'
}

echo "== warmup =="
task_curl warmup full null 1 2

echo "== batch=1 (5 reps/task) =="
task_curl full_song full null 1 5
task_curl melody melody "$EXAMPLES/melody.abc" 1 5
task_curl harmony_original full "$EXAMPLES/score.abc" 1 5
task_curl harmony_edited full "$EXAMPLES/score-jazz.abc" 1 5

echo "== concurrency on full_song =="
task_curl full_song_c2 full null 2 6
task_curl full_song_c4 full null 4 8

echo "== stats =="
stats "$OUT/full_song_lat.txt"  "full_song(c=1)"
stats "$OUT/melody_lat.txt"      "melody(c=1)"
stats "$OUT/harmony_original_lat.txt" "harmony_orig(c=1)"
stats "$OUT/harmony_edited_lat.txt"   "harmony_edit(c=1)"
stats "$OUT/full_song_c2_lat.txt" "full_song(c=2)"
stats "$OUT/full_song_c4_lat.txt" "full_song(c=4)"
for t in full_song melody harmony_original harmony_edited; do
    wall=$(cat "$OUT/${t}_wall.txt" | awk '{print $2}')
    rps=$(echo "$wall" | awk -v n=5 '{printf "%.2f", n/$1}')
    echo "$t(c=1): 5 req in ${wall}s -> ${rps} req/s"
done
for t in full_song_c2 full_song_c4; do
    wall=$(cat "$OUT/${t}_wall.txt" | awk '{print $2}')
    n=$(wc -l < "$OUT/${t}_lat.txt")
    rps=$(echo "$wall" | awk -v n="$n" '{printf "%.2f", n/$1}')
    echo "$t: $n req in ${wall}s -> ${rps} req/s"
done

echo "== spot-check rendered songs ($BENCH_OUT) =="
bash "$(dirname "$0")/spot_check.sh" \
    "$BENCH_OUT/bench_full_song.wav" \
    "$BENCH_OUT/bench_melody.wav" \
    "$BENCH_OUT/bench_harmony_original.wav" \
    "$BENCH_OUT/bench_harmony_edited.wav"
```

```
#!/usr/bin/env bash
# Maximum-concurrency scan: boots one server with a large batch cap and sweeps
# N=1,2,4,6,8,12,16 to find the throughput ceiling.
#
# Usage:
#   bash yue2_tests/bench/bench_max_concurrency.sh [options]
#     --gpu N  --sem N  --port P  --n-list "1 2 4 6 8 12 16"
#     --max-size N  --delay-ms N  --keep
set -eo pipefail

HERE="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "$HERE/_lib.sh"
CONC="$HERE/bench_concurrent.py"

GPU=5; SEM=512; N_LIST="1 2 4 6 8 12 16"; MMAX=8; DELAY=50; PORT=30910; KEEP=0
while [[ $# -gt 0 ]]; do
  case "$1" in
    --gpu) GPU="$2"; shift 2 ;;
    --sem) SEM="$2"; shift 2 ;;
    --port) PORT="$2"; shift 2 ;;
    --n-list) N_LIST="$2"; shift 2 ;;
    --max-size) MMAX="$2"; shift 2 ;;
    --delay-ms) DELAY="$2"; shift 2 ;;
    --keep) KEEP=1; shift ;;
    *) echo "unknown arg: $1" >&2; exit 2 ;;
  esac
done

OUT="${OUT:-$HERE/max_conc_$(date +%Y%m%d_%H%M%S)}"
mkdir -p "$OUT"
PORT_REAL="$(start_yue2_server "$OUT/server.pid" "$OUT/server.log" "$GPU" "$PORT" "$MMAX" "$DELAY" 21620)"

"$PYTHON_BIN" "$CONC" --port "$PORT_REAL" -n 4 --semantic-max-tokens "$SEM" > /dev/null 2>&1 || true
RESULTS=()
for n in $N_LIST; do
  line="$("$PYTHON_BIN" "$CONC" --port "$PORT_REAL" -n "$n" --semantic-max-tokens "$SEM" | tail -1)"
  bench_log "$line"
  RESULTS+=("$line")
done
[[ "$KEEP" == 1 ]] || stop_yue2_server "$OUT/server.pid"

{
  echo
  echo "==== MAX CONCURRENCY (semantic=$SEM, gpu=$GPU, batching=${MMAX}x${DELAY}ms) ===="
  printf "%-6s %-12s %-12s\n" "N" "wall_s" "req_s"
  for line in "${RESULTS[@]}"; do
    printf "%-6s %-12s %-12s\n" \
      "$(sed -E 's/.*N=([0-9]+).*/\1/' <<<"$line")" \
      "$(sed -E 's/.*wall=([0-9.]+)s.*/\1/' <<<"$line")" \
      "$(sed -E 's/.*throughput=([0-9.]+).*/\1/' <<<"$line")"
  done
  echo "grouped batches: $(grouped_batches "$OUT/server.log")"
  echo "module (avg): AR=$(avg_stage "$OUT/server.log" Yue2ARStage)s NAR=$(avg_stage "$OUT/server.log" Yue2NARStage)s VAE=$(avg_stage "$OUT/server.log" Yue2VAEDecodeStage)s"
} | tee "$OUT/final.txt"
bench_log "done. $OUT/final.txt"
```
