# [Fix] sampling: Make top-k/top-p filtering consistent across sampling paths

[Upstream PR](https://github.com/sgl-project/sglang/pull/41916) · state: closed · merged: True

## Original PR description

Supersedes #36173 and addresses its final review findings.

## Motivation

SGLang normally computes the top-p cutoff from the original probability distribution and intersects that support with top-k. Some sampling paths instead applied top-k, renormalized the remaining probabilities, and then applied top-p. That applies top-p to whatever probability mass happens to remain after top-k, making its effect unpredictable and inconsistent with other sampling paths using the same parameters.

The affected paths are the FlashInfer min-p path, the shared EAGLE-family/standalone/n-gram verifier, and the target-distribution builder shared by DFlash, DSpark, and UNO. UNO's separate compact top-k path also needed the same fix; its behavior could otherwise depend on other requests in the batch.

## Modifications

- Apply top-p before top-k in the FlashInfer min-p path.
- Apply the same ordering in the shared EAGLE-family, standalone, and n-gram verifier.
- Correct dense/sparse target distributions shared by DFlash, DSpark, and UNO.
- Preserve the sparse top-k path by translating the top-p cutoff into the renormalized top-k distribution using the top-k support's original probability mass.
- Apply the same correction to UNO's separate compact top-k path.
- Fuse CUDA unseeded top-p rescaling by reusing the retained top-k maximum, avoiding a full-vocabulary maximum reduction and exponential intermediate.
- Preserve logsumexp for seeded/non-CUDA callers; UNO rejects deterministic inference.
- Document that fused FP32 rounding can change cutoff membership, without promising bitwise identity or trainer alignment.
- Reject `min_p > 0` for built-in speculative decoding, where it is not currently applied.
- Add focused CUDA regression tests for each corrected implementation.

## Accuracy Tests

The regressions use probabilities `[0.35, 0.25, 0.20, 0.05, 0.03, ...]`, `top_k=4`, and `top_p=0.82`. The FlashInfer min-p case also uses `min_p=0.1`.

The original top-p nucleus includes token 3 because the probability mass before it is `0.80`. Applying and renormalizing top-k first changes that mass to `0.80 / 0.85`, causing token 3 to be incorrectly removed.

Validated rebased commit `50e6ca6b73cc49214b5e9e28068b187ac3d7de2f` on an NVIDIA H200 with Python 3.12.3, CUDA 13.0, PyTorch 2.13.0, FlashInfer 0.6.18, and Triton 3.7.1.

import json
import os
from pathlib import Path
import statistics

from types import SimpleNamespace

import torch

from sglang.srt.speculative import dflash_utils as dflash
from sglang.srt.speculative import uno_utils as uno

ROOT = Path(__file__).resolve().parent
OUT = ROOT / 'benchmark-results.json'


def source_node(path, name):
    source = Path(path).read_text()
    node = next(n for n in ast.parse(source).body if isinstance(n, ast.FunctionDef) and n.name == name)
    return source, node


def load_base(path, names, module):
    namespace = dict(vars(module))
    source = Path(path).read_text()
    nodes = [n for n in ast.parse(source).body if isinstance(n, ast.FunctionDef) and n.name in names]
    assert len(nodes) == len(names)
    exec(compile(ast.Module(body=nodes, type_ignores=[]), str(path), 'exec'), namespace)
    return namespace


def capture(fn):
    # Compilation and allocator warmup are outside the measured region.
    for _ in range(4):
        fn()
    torch.cuda.synchronize()
    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        output = fn()
    for _ in range(10):
        graph.replay()
    torch.cuda.synchronize()
    tensors = output if isinstance(output, tuple) else (output,)
    assert all(torch.isfinite(t).all().item() for t in tensors)
    probs = tensors[-1]
    assert torch.allclose(probs.sum(-1), torch.ones_like(probs.sum(-1)), atol=2e-5, rtol=2e-5)
    return graph, output


def measure(graph, repetitions=200):
    start, end = torch.cuda.Event(enable_timing=True), torch.cuda.Event(enable_timing=True)
    start.record()
    for _ in range(repetitions):
        graph.replay()
    end.record()
    end.synchronize()
    return start.elapsed_time(end) * 1000 / repetitions

        'source_sha256': {str(p.name): hashlib.sha256(p.read_bytes()).hexdigest() for p in (ROOT/'base_dflash_utils.py', ROOT/'base_uno_utils.py', Path(dflash.__file__), Path(uno.__file__), Path(__file__))},

import os
from pathlib import Path
import signal
import subprocess

import time

import psutil
import requests

ROOT = Path(__file__).resolve().parent
BASE_URL = 'http://127.0.0.1:31916'
RESULT = {'head': 'f0c0a64ad1b70d9a72f9d741b9eaa2b51d088c03', 'model': json.loads((ROOT/'model.json').read_text()), 'checks': [], 'launches': []}


def save():
    (ROOT/'http-results.json').write_text(json.dumps(RESULT, indent=2)+'\n')


def request(min_p, stream, expected_rejection, label):
    payload = {'text': 'The capital of France is', 'sampling_params': {'temperature': .7, 'top_k': 50, 'top_p': .9, 'min_p': min_p, 'max_new_tokens': 8}, 'stream': stream}
    start = time.monotonic()
    response = requests.post(BASE_URL+'/generate', json=payload, timeout=(10, 60))
    duration = time.monotonic()-start
    body = response.text
    if expected_rejection:
        assert 'min_p is not supported with NGRAM speculative decoding' in body, (response.status_code, body)
        assert response.status_code in (200, 400, 422), response.status_code
        if stream:
            assert 'data:' in body or response.status_code != 200, body
    else:
        assert response.status_code == 200, (response.status_code, body)
        parsed = response.json()
        assert isinstance(parsed.get('text'), str), parsed
        assert parsed['meta_info']['completion_tokens'] > 0, parsed
        assert parsed['meta_info']['finish_reason']['type'] != 'abort', parsed
    RESULT['checks'].append({'label': label, 'request': payload, 'status_code': response.status_code, 'elapsed_seconds': duration, 'response': body, 'passed': True})
    save()
    print(json.dumps({'label':label, 'status':response.status_code, 'seconds':duration, 'passed':True}), flush=True)


def stop_server(process):
    try:
        children = psutil.Process(process.pid).children(recursive=True)
    except psutil.NoSuchProcess:
        children = []
    try:
        os.killpg(process.pid, signal.SIGTERM)
    except ProcessLookupError:
        pass
    try:
        process.wait(timeout=45)
    except subprocess.TimeoutExpired:
        os.killpg(process.pid, signal.SIGKILL)
        process.wait(timeout=15)
    for child in children:
        try:
            if child.is_running(): child.terminate()
        except psutil.NoSuchProcess:
            pass
    _, alive = psutil.wait_procs(children, timeout=15)
    for child in alive:
        try: child.kill()
        except psutil.NoSuchProcess: pass
    _, alive = psutil.wait_procs(alive, timeout=15)
    assert not [p.pid for p in alive if p.status() != psutil.STATUS_ZOMBIE], 'server children survived'


def run_server(speculative):
    label = 'ngram' if speculative else 'ordinary'
    cmd = [sys.executable, '-m', 'sglang.launch_server', '--model-path', str(ROOT/'model'), '--host', '127.0.0.1', '--port', '31916', '--attention-backend', 'flashinfer', '--dtype', 'bfloat16', '--mem-fraction-static', '.5', '--max-running-requests', '8', '--cuda-graph-max-bs-decode', '8', '--context-length', '2048']
    if speculative:
        cmd += ['--speculative-algorithm', 'NGRAM', '--speculative-num-draft-tokens', '16']
    env = dict(os.environ, PYTHONPATH=str(ROOT/'source/python'), HF_HOME=str(ROOT/'hf'), HF_HUB_CACHE=str(ROOT/'hf/hub'), TORCHINDUCTOR_CACHE_DIR=str(ROOT/'inductor'), TRITON_CACHE_DIR=str(ROOT/'triton'), SGLANG_JIT_DEEPGEMM_PRECOMPILE='0', SGLANG_RUST_BUILD_MODE='auto', SGLANG_CACHE_DIR=str(ROOT/'sglang-cache'))
    with open(ROOT/f'server-{label}.log', 'w') as log:
        process = subprocess.Popen(cmd, env=env, cwd=ROOT/'source', stdout=log, stderr=subprocess.STDOUT, stdin=subprocess.DEVNULL, start_new_session=True)
        RESULT['launches'].append({'mode':label, 'command':cmd, 'pid':process.pid})
        save()
        try:
            deadline = time.monotonic()+600
            while time.monotonic()<deadline:
                if process.poll() is not None:
                    raise RuntimeError(f'{label} server exited: {process.returncode}; see server-{label}.log')
                try:
                    if requests.get(BASE_URL+'/health',timeout=2).status_code == 200:
                        break
                except requests.RequestException:
                    pass
                time.sleep(2)
            else:
                raise TimeoutError(f'{label} server startup exceeded 600 seconds')
            print(f'SERVER_READY {label}',flush=True)
            if speculative:
                request(0, False, False, 'ngram valid initial control')
                for iteration in range(2):
                    for stream in (False, True):
                        request(.1, stream, True, f'ngram reject min_p=.1 stream={stream} repeat={iteration}')
                        request(0, False, False, f'ngram valid after rejection stream={stream} repeat={iteration}')
            else:
                request(.1, False, False, 'ordinary accepts min_p=.1')
            assert requests.get(BASE_URL+'/health',timeout=10).status_code == 200
            assert process.poll() is None
        finally:
            stop_server(process)
            RESULT['launches'][-1]['stopped'] = True
            save()

    RESULT['complete'] = True
    print('HTTP_CHECK_COMPLETE',flush=True)

    raise
finally:
    save()
