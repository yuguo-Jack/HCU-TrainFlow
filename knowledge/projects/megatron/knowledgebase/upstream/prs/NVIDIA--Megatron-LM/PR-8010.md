# What does this PR do?
<!-- Add a one line overview of what this PR aims to accomplish. -->

#3202 fixed a bug with tied weights + PP + `init_model_with_meta_device`, but the fix was incomplete. That fix broke FSDP under the same scenario. The tied embedding weights are not initialized to the same values, because the all-reduce that would make them match gets skipped.

Running pretraining with `--use-megatron-fsdp --init-model-with-meta-device` and either PP>1 or MTP will trip this bug, unless embeddings are deliberately untied via `--untie-embeddings-and-output-weights`.

Without the fix, the last-stage weight was entirely zero, while the first-stage embedding had norm `127.1424`. With the fix, `torch.equal` confirmed exact equality before and after wrapping.

<summary>Reproduction script</summary>

Save as `/tmp/repro.py` in an environment with Megatron’s dependencies and two CUDA GPUs:

```python
"""Check cross-PP tied-weight equality before and after actual MFSDP v2 wrapping.

Run with torch.distributed.run --standalone --nproc-per-node=2.
The default expectation is equality; --expect-mismatch verifies the broken case.
"""

import argparse
import json
import os
from contextlib import nullcontext
from datetime import timedelta

import torch
import torch.distributed as dist
from torch.distributed.tensor import DTensor

from megatron.core import parallel_state
from megatron.core.distributed.distributed_data_parallel_config import DistributedDataParallelConfig
from megatron.core.distributed.fsdp.mcore_fsdp_adapter import (
    FullyShardedDataParallel,
    FullyShardedDataParallelV2,
)
from megatron.core.models.gpt.gpt_layer_specs import get_gpt_layer_with_transformer_engine_spec
from megatron.core.models.gpt.gpt_model import GPTModel
from megatron.core.process_groups_config import ProcessGroupCollection
from megatron.core.tensor_parallel.random import model_parallel_cuda_manual_seed
from megatron.core.transformer.transformer_config import TransformerConfig


def measure(model, label, group):
    weight = model.shared_embedding_or_output_weight()
    assert not weight.is_meta
    assert not isinstance(weight, DTensor), "Read actual unsharded compute weights"
    weight = weight.detach().clone()
    reference = weight.clone()
    dist.broadcast(reference, src=dist.get_process_group_ranks(group)[0], group=group)
    row = {
        "rank": dist.get_rank(),
        "phase": label,
        "shape": list(weight.shape),
        "nonzero": torch.count_nonzero(weight).item(),
        "norm": weight.float().norm().item(),
        "max_abs_diff_from_first_stage": (weight.float() - reference.float()).abs().max().item(),
        "equal_to_first_stage": torch.equal(weight, reference),
    }
    rows = [None] * dist.get_world_size()
    dist.all_gather_object(rows, row)
    if dist.get_rank() == 0:
        print("MEASUREMENT " + json.dumps(rows, sort_keys=True), flush=True)
    return rows, weight


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--meta", action="store_true")
    parser.add_argument("--expect-mismatch", action="store_true")
    args = parser.parse_args()

    torch.cuda.set_device(int(os.environ["LOCAL_RANK"]))
    dist.init_process_group("nccl", timeout=timedelta(seconds=120))
    assert dist.get_world_size() == 2
    parallel_state.initialize_model_parallel(
        tensor_model_parallel_size=1, pipeline_model_parallel_size=2
    )
    groups = ProcessGroupCollection.use_mpu_process_groups()
    model_parallel_cuda_manual_seed(123)
    config = TransformerConfig(
        num_layers=2,
        hidden_size=128,
        num_attention_heads=4,
        pipeline_model_parallel_size=2,
        pipeline_dtype=torch.bfloat16,
        params_dtype=torch.bfloat16,
        bf16=True,
        embedding_init_method_std=1.0,
        init_model_with_meta_device=args.meta,
        use_cpu_initialization=False,
    )
    with torch.device("meta") if args.meta else nullcontext():
        model = GPTModel(
            config=config,
            transformer_layer_spec=get_gpt_layer_with_transformer_engine_spec(),
            vocab_size=128,
            max_sequence_length=8,
            pre_process=parallel_state.is_pipeline_first_stage(),
            post_process=parallel_state.is_pipeline_last_stage(),
            share_embeddings_and_output_weights=True,
            pg_collection=groups,
        )
    assert any(p.is_meta for p in model.decoder.parameters()) == args.meta
    before, before_weight = measure(model, "before_v2_wrap", groups.embd)
    wrapped = FullyShardedDataParallel(
        config=config,
        ddp_config=DistributedDataParallelConfig(
            use_megatron_fsdp=True,
            megatron_fsdp_version=2,
            data_parallel_sharding_strategy="optim_grads_params",
            nccl_ub=False,
        ),
        module=model,
        device=torch.device("cuda", torch.cuda.current_device()),
        pg_collection=groups,
    )
    assert isinstance(wrapped, FullyShardedDataParallelV2)
    assert not any(p.is_meta for p in wrapped.module.parameters())
    assert isinstance(wrapped.module.shared_embedding_or_output_weight(), DTensor)
    wrapped.module.context.allgather_stream.wait_stream(torch.cuda.current_stream())
    wrapped.module.unshard()
    after, after_weight = measure(wrapped.module, "after_v2_wrap_and_unshard", groups.embd)
    assert torch.equal(before_weight, after_weight), "Wrapping changed local initial weight values"
    for phase in (before, after):
        equal = all(row["equal_to_first_stage"] for row in phase)
        assert equal == (not args.expect_mismatch), phase
        assert phase[0]["nonzero"] > 0, "First-stage reference must be nonzero"
        if args.expect_mismatch:
            assert phase[1]["nonzero"] == 0, "Expected zero last-stage tied output weight"
    if dist.get_rank() == 0:
        print(
            "RESULT "
            + json.dumps(
                {
                    "meta": args.meta,
                    "pp": 2,
                    "dp": 1,
                    "wrapper": type(wrapped).__name__,
                    "expected_mismatch": args.expect_mismatch,
                    "verified": True,
                },
                sort_keys=True,
            ),
            flush=True,
        )
    dist.barrier()
    parallel_state.destroy_model_parallel()
    dist.destroy_process_group()


if __name__ == "__main__":
    main()
```
