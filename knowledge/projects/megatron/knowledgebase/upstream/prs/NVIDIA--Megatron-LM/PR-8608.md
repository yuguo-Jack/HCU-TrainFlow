### Add repeatable tests for the release metadata update

The new [metadata-update step](https://github.com/NVIDIA/Megatron-LM/blob/3bd346eb10c46f24ea01786b213aa3f882e697b4/.github/workflows/release-freeze.yml#L76-L98) adds insertion, existing-version, and rerun behavior without corresponding automated tests. Static tracing supports the intended result, but actionlint and the reported one-off simulation do not preserve these checks against future changes. This is a non-blocking coverage gap, not an observed runtime failure.

**Correction:** Add a CPU-only regression test under [tests/test_utils](https://github.com/NVIDIA/Megatron-LM/tree/3bd346eb10c46f24ea01786b213aa3f882e697b4/tests/test_utils) that executes the actual workflow step against temporary metadata. The existing [workflow-shell test harness](https://github.com/NVIDIA/Megatron-LM/blob/3bd346eb10c46f24ea01786b213aa3f882e697b4/tests/test_utils/test_testmon_workflow.py#L19-L54) demonstrates this approach. Avoid copying the jq implementation into the test or invoking authentication and push steps.

For example, the following proposed test covers both target-version branches and their preservation requirements:

```python
import json
import os
import shutil
import subprocess
from pathlib import Path

import pytest
import yaml

ROOT = Path(__file__).parents[2]
DOCS_ROOT = "https://docs.nvidia.com/megatron-core/developer-guide/"


@pytest.mark.parametrize("target_exists", [False, True])
def test_release_docs_metadata(tmp_path, target_exists):
    if shutil.which("jq") is None:
        pytest.skip("Workflow shell validation requires jq")
    workflow = yaml.safe_load(
        (ROOT / ".github/workflows/release-freeze.yml").read_text()
    )
    step = next(
        s for s in workflow["jobs"]["bump-release-docs"]["steps"]
        if s.get("name") == "Update release documentation metadata"
    )
    docs = tmp_path / "docs"
    docs.mkdir()
    (docs / "conf.py").write_text('release = "nightly"\n')
    (docs / "project.json").write_text(
        '{"name": "megatron-lm", "version": "nightly"}\n'
    )
    nightly = dict(name="nightly", version="nightly", url=DOCS_ROOT + "nightly/")
    historical = dict(name="0.18.2", version="0.18.2", url=DOCS_ROOT + "0.18.2/")
    versions = [nightly, dict(
        name="0.19.0 (latest)", version="0.19.0",
        url=DOCS_ROOT + "latest/", preferred=True,
    ), historical]
    if target_exists:
        versions.insert(1, dict(
            name="0.20.0", version="0.20.0", url=DOCS_ROOT + "0.20.0/"
        ))
    (docs / "versions1.json").write_text(json.dumps(versions))

    def snapshot():
        return {p.name: p.read_bytes() for p in docs.iterdir()}

    def update(branch):
        return subprocess.run(
            ["bash", "-e", "-o", "pipefail", "-c", step["run"]],
            cwd=tmp_path,
            env={**os.environ, "RELEASE_BRANCH": branch},
            capture_output=True, text=True,
        )

    before = snapshot()
    assert update("core_r0.20").returncode != 0
    assert snapshot() == before
    result = update("core_r0.20.0")
    assert result.returncode == 0, result.stderr
    assert (docs / "conf.py").read_text() == 'release = "0.20.0"\n'
    assert json.loads((docs / "project.json").read_text()) == {
        "name": "megatron-lm", "version": "0.20.0"
    }
    actual = json.loads((docs / "versions1.json").read_text())
    target = dict(name="0.20.0 (latest)", version="0.20.0",
                  url=DOCS_ROOT + "latest/", preferred=True)
    assert [v for v in actual if v["version"] == "0.20.0"] == [target]
    assert [v for v in actual if v.get("preferred") is True] == [target]
    assert actual[0] == nightly
    assert next(v for v in actual if v["version"] == "0.19.0") == dict(
        name="0.19.0", version="0.19.0", url=DOCS_ROOT + "0.19.0/"
    )
    assert next(v for v in actual if v["version"] == "0.18.2") == historical
    once = snapshot()
    assert update("core_r0.20.0").returncode == 0
    assert snapshot() == once
```

**Acceptance:** Both parameter cases should pass: inserting a new target and promoting an existing target must produce one target entry and one preferred release, demote the previous release to its numbered URL, preserve nightly and unrelated history, and leave all files byte-identical on rerun. Invalid input must fail without modifying metadata. These are proposed checks; I could not execute them here.

The changed workflow and directly affected documentation consumers were inspected. No concrete runtime defect was established; external reusable workflow/action implementations were not available for inspection.
