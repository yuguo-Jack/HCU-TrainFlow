# fix(inference): preserve low-temperature sampling probabilities

- [x] I, the PR author, have personally reviewed every line of this PR.

# What does this PR do?

Low-temperature BF16 sampling can keep excluded tokens; FP16 scaling can produce NaN probabilities.

Scale and filter a separate FP32 copy for token draws and processed log probabilities, leaving the input unchanged.

## Testing Done

CPU only, macOS arm64, Python 3.12.14, PyTorch 2.15.0, megatron-core installed from this checkout. Triton has no macOS wheel and megatron.core imports it on load, so a placeholder triton module (source below) stands in; this sampling path launches no Triton kernel. The BF16 case is reduced to the two logit values reported in the issue. The trained model, DynamicInferenceEngine and CUDA were not run.

| Scenario | Before | After |
|---|---|---|
| BF16, temperature 0.1, top_p 0.9 | Second token retained; 40 of 256 draws select it | Second token excluded; all 256 draws select the first |
| FP16, temperature 0.00001 | Both probabilities are NaN; all draws select the wrong token | Finite probabilities; all 256 draws select the higher-logit token |
| FP32 control | First token has probability 1 | Unchanged |

Save the sources below as repro_temperature.py and sitecustomize.py in one directory, then run the reproducer from the checkout with that directory on the import path:

```bash
PYTHONPATH=<directory with sitecustomize.py> python - < repro_temperature.py 2>/dev/null
```

import torch
from megatron.core.inference.sampling.torch_sampling import TorchSampling

print('torch=',torch.__version__,'cuda_available=',torch.cuda.is_available())
for dtype, temperature, values, top_p in (
    (torch.bfloat16,0.1,[26.625,26.375],0.9),
    (torch.float16,1e-5,[1.0,2.0],0.0),
    (torch.float32,0.1,[26.625,26.375],0.9),
):
    logits=torch.tensor([values],dtype=dtype)
    filtered=TorchSampling.filter_logits(last_token_logits=logits,temperature=temperature,top_k=0,top_p=top_p)
    sampled=TorchSampling.sample_from_logits(last_token_logits=logits.repeat((256,1)),temperature=temperature,top_k=0,top_p=top_p,generator=torch.Generator().manual_seed(123))
    print({'dtype':str(dtype),'temperature':temperature,'filtered':filtered.tolist(),'probabilities':filtered.softmax(dim=-1,dtype=torch.float32).tolist(),'sample_counts':sampled.bincount(minlength=2).tolist()})
```

"""Local import stand-in for Triton, used only for CPU runs on macOS.

Why this exists: Triton publishes no macOS wheels, but `import megatron.core` imports it at
module load time (for example megatron/core/transformer/moe/ops/paged_stash.py), so the package
cannot be imported on this machine without something named `triton`. The Torch sampling path under
test never launches a Triton kernel.

How it works: Python runs this file at interpreter start (it is named sitecustomize.py and its
directory is put on PYTHONPATH). It imports torch first, then registers a finder that serves
placeholder `triton*` modules only to importers outside torch/functorch, so torch keeps seeing "no
Triton" exactly as on a clean machine. `@triton.jit` returns the plain Python function, so
launching a kernel (`kernel[grid](...)`) raises TypeError instead of silently passing.

This file lives outside the repository and is never committed.
"""

import importlib.abc
import importlib.machinery
import sys
import types

import torch  # noqa: F401  (import torch before the placeholder finder exists)

try:  # torch imports these lazily; load them while Triton is still absent.
    import torch._dynamo  # noqa: F401
    import torch._inductor  # noqa: F401

    # megatron.core.jit decorates functions with torch.compile at import time, which makes
    # torch lazily import functorch/inductor internals. Do that now, before the placeholder
    # exists, so those internals never see it. torch.compile is lazy: nothing is compiled.
    torch.compile(lambda x: x)
except Exception:  # pragma: no cover - best effort only
    pass

_VERSION = "0.0.0+local-import-stub"
_DENY_IMPORTERS = ("torch", "functorch", "torchgen")


class _Stub:
    """Absorbs attribute access, calls and subscripts performed at import time."""

    def __call__(self, *args, **kwargs):
        if len(args) == 1 and not kwargs and callable(args[0]) and not isinstance(args[0], _Stub):
            return args[0]
        return _Stub()

    def __getitem__(self, item):
        return _Stub()

    def __getattr__(self, name):
        if name.startswith("__") and name.endswith("__"):
            raise AttributeError(name)
        return _Stub()


class _StubModule(types.ModuleType):
    def __getattr__(self, name):
        if name.startswith("__") and name.endswith("__"):
            raise AttributeError(name)
        return _Stub()


def _jit(*args, **kwargs):
    if len(args) == 1 and not kwargs and callable(args[0]):
        return args[0]
    return lambda fn: fn


def _decorator_factory(*args, **kwargs):
    return lambda fn: fn


class _OutOfResources(Exception):
    pass


class _Autotuner:
    pass


class _FileCacheManager:
    def __init__(self, *args, **kwargs):
        pass


_PLACEHOLDER_ATTRS = {
    "triton.runtime.errors": {"OutOfResources": _OutOfResources},
    "triton.runtime.autotuner": {"Autotuner": _Autotuner},
    "triton.runtime.cache": {"FileCacheManager": _FileCacheManager},
}


def _importer_module_name():
    """Name of the first module on the stack that is not import machinery."""
    frame = sys._getframe(2)
    while frame is not None:
        name = frame.f_globals.get("__name__", "")
        if not (name.startswith("importlib") or name.startswith("_frozen_importlib")):
            return name
        frame = frame.f_back
    return ""


class _Finder(importlib.abc.MetaPathFinder, importlib.abc.Loader):
    def find_spec(self, fullname, path=None, target=None):
        if fullname != "triton" and not fullname.startswith("triton."):
            return None
        if _importer_module_name().split(".")[0] in _DENY_IMPORTERS:
            return None
        return importlib.machinery.ModuleSpec(fullname, self, is_package=True)

    def create_module(self, spec):
        module = _StubModule(spec.name)
        module.__path__ = []
        module.__version__ = _VERSION
        if spec.name == "triton":
            module.jit = _jit
            module.autotune = _decorator_factory
            module.heuristics = _decorator_factory
            module.cdiv = lambda x, y: (x + y - 1) // y
            module.next_power_of_2 = lambda n: 1 if n <= 1 else 1 << (int(n) - 1).bit_length()
        for attr, value in _PLACEHOLDER_ATTRS.get(spec.name, {}).items():
            setattr(module, attr, value)
        return module

    def exec_module(self, module):
        return None


if "triton" not in sys.modules:
    sys.meta_path.insert(0, _Finder())
```
