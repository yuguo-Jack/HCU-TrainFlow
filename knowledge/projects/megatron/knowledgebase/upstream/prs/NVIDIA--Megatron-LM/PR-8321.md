# Fix default dense TP reduction in batch-invariant mode

[Upstream PR](https://github.com/NVIDIA/Megatron-LM/pull/8321) · state: open · merged: False

## Original PR description

# What does this PR do?

With `batch_invariant_mode=True`, dense TE row-parallel layers still default to NCCL SUM. Its summation order can change with batch layout, so the same prompt can return different log probabilities when run alone or with other requests at fixed TP.

Use TE's existing `two_shot` reduction when batch-invariant mode is enabled, TP > 1, SP is disabled, and no reduction is explicitly selected. This applies only to dense TE row-parallel layers and requires a topology supported by PyTorch symmetric memory.

Existing TE releases support this fix but retain a staging buffer for each input size, which can accumulate in long-running services. [NVIDIA/TransformerEngine#3709](https://github.com/NVIDIA/TransformerEngine/pull/3709) and [#2037](https://github.com/NVIDIA/TransformerEngine/pull/2037) address that cache growth. Should this default change wait for the TE fix?

```python
"""Compare pretrained Qwen3 outputs across batches at fixed TP4 topology."""
import os
os.environ["TRITON_CACHE_DIR"] = os.path.join(os.environ["RESULT_DIR"], "triton-" + os.environ.get("LOCAL_RANK", "0"))
import gc
import json
from pathlib import Path

import torch

from tests.unit_tests.models.test_gpt_model_batch_invariant import (
    DummyTokenizer,
    DynamicInferenceContext,
    DynamicInferenceEngine,
    Float16Module,
    GPTInferenceWrapper,
    InferenceConfig,
    SamplingParams,
    TextGenerationController,
    Utils,
    AttnBackend,
    _configure_flash_attention_env,
    model_parallel_cuda_manual_seed,
    set_batch_invariant_mode,
    te_supports_batch_invariant_attention,
)


def main():
    assert te_supports_batch_invariant_attention()
    Utils.initialize_model_parallel(4, 1)
    _configure_flash_attention_env()
    torch.manual_seed(321)
    model_parallel_cuda_manual_seed(321)
    from megatron.bridge import AutoBridge
    from transformers import AutoTokenizer
    model_path = os.environ['MODEL_PATH']
    bridge = AutoBridge.from_hf_pretrained(model_path)
    provider = bridge.to_megatron_provider(load_weights=True)
    provider.tensor_model_parallel_size = 4
    provider.pipeline_model_parallel_size = 1
    provider.sequence_parallel = False
    provider.params_dtype = torch.bfloat16
    provider.bf16 = True
    provider.fp16 = False
    provider.batch_invariant_mode = True
    provider.batch_invariant_backend = 'te_native'
    provider.attention_backend = AttnBackend.flash
    provider.flash_attention_version = 4
    provider.hidden_dropout = 0.0
    provider.attention_dropout = 0.0
    provider.cuda_graph_impl = 'none'
    assert provider.perform_initialization is False
    assert provider.pre_wrap_hook is not None
    provider.symmetric_ar_type = 'two_shot' if os.environ['SYMMETRIC'] == '1' else None
    provider.finalize()
    models = provider.provide_distributed_model(wrap_with_ddp=False)
    assert len(models) == 1
    model = models[0].cuda().eval()
    base = model.module if isinstance(model, Float16Module) else model

    from safetensors import safe_open
    index = json.loads((Path(model_path) / 'model.safetensors.index.json').read_text())['weight_map']
    for key, actual in [
        ('model.layers.0.self_attn.q_norm.weight', base.decoder.layers[0].self_attention.q_layernorm.weight),
        ('model.norm.weight', base.decoder.final_layernorm.weight),
    ]:
        with safe_open(str(Path(model_path) / index[key]), framework='pt') as weights:
            expected = weights.get_tensor(key)
        assert torch.equal(actual.detach().cpu(), expected.to(actual.dtype)), key

    import importlib
    te_linear = importlib.import_module('transformer_engine.pytorch.module.linear')
    original_symmetric = te_linear.symmetric_all_reduce
    symmetric_calls = []
    def observe_symmetric(tensor, *args, **kwargs):
        result = original_symmetric(tensor, *args, **kwargs)
        symmetric_calls.append(list(tensor.shape))
        return result
    te_linear.symmetric_all_reduce = observe_symmetric
    tokenizer_hf = AutoTokenizer.from_pretrained(model_path)
    tokenizer = DummyTokenizer(vocab_size=tokenizer_hf.vocab_size, eod=tokenizer_hf.eos_token_id)
    texts = [
        'Explain why the sky is blue in simple terms.',
        'Write a Python function that returns the largest number in a list and explain how it handles an empty list.',
        'Compare solar and wind energy. Discuss reliability, installation costs, and how electricity storage can help.',
        'A researcher measures temperature every hour for a week. Explain how to summarize the measurements, find unusual observations, and compare daytime and nighttime temperatures without confusing correlation with causation.',
    ]
    prompts = {i: tokenizer_hf.encode(text, add_special_tokens=False) for i, text in enumerate(texts,1)}
    assert max(map(len, prompts.values())) + 8 <= 256
    assert sum(map(len, prompts.values())) <= 512
    sampling = SamplingParams(
        num_tokens_to_generate=8, temperature=1.0, top_k=1, top_p=0.0,
        add_BOS=False, return_log_probs=True, termination_id=tokenizer.eod,
    )

    def run(ids):
        context = DynamicInferenceContext(
            model_config=base.config,
            inference_config=InferenceConfig(
                max_sequence_length=256, max_tokens=512, max_requests=8,
                buffer_size_gb=1.0, block_size_tokens=16, num_cuda_graphs=None,
                materialize_only_last_token_logits=False,
                use_cuda_graphs_for_non_decode_steps=False, unified_memory_level=0,
            ),
        )
        controller = TextGenerationController(GPTInferenceWrapper(model, context), tokenizer)
        engine = DynamicInferenceEngine(controller=controller, context=context)
        results = {}
        with torch.no_grad(), set_batch_invariant_mode(True, backend="te_native"):
            for rid in ids:
                engine.add_request(rid, prompts[rid], sampling)
            for _ in range(64):
                if not engine.has_unfinished_requests():
                    break
                for req in engine.step_modern()['finished_requests']:
                    results[req.request_id] = {
                        'tokens': req.generated_tokens,
                        'prompt_logprobs': req.prompt_log_probs,
                        'generated_logprobs': req.generated_log_probs,
                    }
            assert not engine.has_unfinished_requests(), 'Probe exceeded step limit'
        assert set(results) == set(ids)
        del engine, controller, context
        gc.collect()
        return results

    solo = {rid: run([rid])[rid] for rid in prompts}
    mixed = run(list(prompts))
    reversed_batch = run(list(reversed(prompts)))
    report = {
        'scope': 'Qwen3-32B pretrained, BF16 te_native TP4 no-SP greedy',
        'model': model_path,
        'prompts': texts,
        'prompt_lengths': {rid: len(tokens) for rid, tokens in prompts.items()},
        'symmetric_calls': symmetric_calls,
        'solo': solo, 'mixed': mixed, 'reversed': reversed_batch,
        'equal': solo == mixed == reversed_batch,
    }
    output = Path(os.environ['RESULT_DIR']) / f"result-{os.environ['SYMMETRIC']}-rank{torch.distributed.get_rank()}.json"
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(json.dumps(report, indent=2) + '\n')
    print('Exact tokens/logprobs equality:', report['equal'])
    Utils.destroy_model_parallel()
    print('Symmetric reductions:', len(symmetric_calls))


if __name__ == '__main__':
    with set_batch_invariant_mode(True, backend="te_native"):
        main()

```
