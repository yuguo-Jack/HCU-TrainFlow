# fix: Keep loss scaling consistent across chained optimizers

[Upstream PR](https://github.com/NVIDIA/Megatron-LM/pull/8249) · state: open · merged: False

## Original PR description

- [x] I, the PR author, have personally reviewed every line of this PR.

# What does this PR do?

Keep FP16 loss scaling consistent across chained optimizers. Backward uses the first optimizer's scale, but each optimizer currently updates its own scaler from its own overflow result. If only the dense or expert gradients overflow, the scales diverge and later updates silently unscale part of the model incorrectly.

Defer scaler updates until every optimizer has checked its gradients, then update each distinct scaler with the combined overflow result. Nested chains follow the same rule. Restored scaler states are aligned with the first scaler, whose scale was used for backward. Standalone optimizers retain their existing behavior and the checkpoint format is unchanged.

## Validation

In a 64-step FP16 EP2 run, the original code reached dense/expert scales of 32768 and 1048576: expert gradients were divided by 32 too much. With the fix, both scales remain equal and the unscaled gradients match the backward scale. The largest AdamW parameter error against an independently calculated update falls from 0.00146269 to 1.19209e-7, the same error as the oracle's calibration against the actual gradients. Reverting the diagnostic fix reproduced the discrepancy; the final implementation was also checked with the same training command.

import json
import os
import runpy
from pathlib import Path

root = Path(__file__).resolve().parent
rank = os.environ.get('LOCAL_RANK', '0')
for key, name in [('TRITON_CACHE_DIR', 'triton'), ('TORCHINDUCTOR_CACHE_DIR', 'inductor'), ('TORCH_EXTENSIONS_DIR', 'torch')]:
    path = root / 'cache' / rank / name
    path.mkdir(parents=True, exist_ok=True)
    os.environ[key] = str(path)

import torch
from megatron.core.optimizer.optimizer import ChainedOptimizer, MixedPrecisionOptimizer

out = open(root / (os.environ.get('VERIFY_RUN', 'baseline') + '-rank' + rank + '.jsonl'), 'w')
original_prepare = ChainedOptimizer.prepare_grads
original_unscale = MixedPrecisionOptimizer._unscale_main_grads_and_check_for_nan
original_step = MixedPrecisionOptimizer.step_with_ready_grads
iteration = 0

def emit(record):
    record.update(iteration=iteration, rank=int(rank))
    out.write(json.dumps(record) + '\n')
    out.flush()

def prepare(self):
    global iteration
    iteration += 1
    scale = self.get_loss_scale().item()
    before = [opt.get_loss_scale().item() for opt in self.chained_optimizers]
    for index, opt in enumerate(self.chained_optimizers):
        opt.verify_scale = scale
        opt.verify_index = index
    skip = original_prepare(self)
    emit(dict(event='chain', before=before, after=[opt.get_loss_scale().item() for opt in self.chained_optimizers], skip=skip))
    return skip

def unscale(self):
    params = [p for group in self.optimizer.param_groups for p in group['params'] if p.grad is not None]
    self.verify_correct = {p: p.grad.detach().clone() / self.verify_scale for p in params}
    own_scale = self.get_loss_scale().item()
    found_inf = original_unscale(self)
    errors = [(p.grad - expected).abs().max().item() for p, expected in self.verify_correct.items() if torch.isfinite(expected).all() and torch.isfinite(p.grad).all()]
    emit(dict(event='unscale', optimizer=self.verify_index, backward_scale=self.verify_scale, own_scale=own_scale, found_inf=found_inf, max_error=max(errors, default=0)))
    return found_inf

def predict(p, group, state, grad):
    beta1, beta2 = group['betas']
    m = state.get('exp_avg', torch.zeros_like(p)) * beta1 + grad * (1 - beta1)
    v = state.get('exp_avg_sq', torch.zeros_like(p)) * beta2 + grad.square() * (1 - beta2)
    step = group.get('step', 0) + 1
    mhat = m / (1 - beta1 ** step) if group.get('bias_correction', True) else m
    vhat = v / (1 - beta2 ** step) if group.get('bias_correction', True) else v
    expected = p.detach() - group['lr'] * (mhat / (vhat.sqrt() + group['eps']) + group['weight_decay'] * p.detach())
    return expected, m

def step(self):
    snapshots = []
    for group in self.optimizer.param_groups:
        for p in group['params']:
            if p.grad is None:
                continue
            state = self.optimizer.state[p]
            actual_prediction, _ = predict(p, group, state, p.grad)
            correct_prediction, correct_m = predict(p, group, state, self.verify_correct[p])
            snapshots.append((p, p.detach().clone(), actual_prediction, correct_prediction, correct_m))
    result = original_step(self)
    emit(dict(event='update', optimizer=self.verify_index,
              oracle_calibration=max((p-a).abs().max().item() for p, old, a, c, m in snapshots),
              parameter_oracle_error=max((p-c).abs().max().item() for p, old, a, c, m in snapshots),
              parameter_delta=max((p-old).abs().max().item() for p, old, a, c, m in snapshots),
              moment_oracle_error=max((self.optimizer.state[p]['exp_avg']-m).abs().max().item() for p, old, a, c, m in snapshots)))
    return result

ChainedOptimizer.prepare_grads = prepare
MixedPrecisionOptimizer._unscale_main_grads_and_check_for_nan = unscale
MixedPrecisionOptimizer.step_with_ready_grads = step
runpy.run_path('pretrain_gpt.py', run_name='__main__')
PY_TRACE

On rank 1, the base unscale error was `0.03257942199707031`; fixed was `0`. Both versions skipped the initial 18 overflowing iterations and performed 46 updates.
