# [Bugfix][MiniMax-M3] Preserve reasoning state while streaming text is buffered

[Upstream PR](https://github.com/vllm-project/vllm/pull/64415) · state: open · merged: False

## Original PR description



## Overview

Keep MiniMax-M3 streaming reasoning detection pending while decoded text is buffered, so an empty text delta with nonempty token IDs cannot prematurely bypass reasoning extraction.

This is separate from the prompt-initialization fixes in #50594 and #63492: it handles generated-token/text timing after streaming extraction starts. Neither of those changes marks an empty text delta pending.

## Claims

- Preserve reasoning/content separation when token IDs arrive before visible text, including split opening markers and a leading closing marker.
- Keep completed reasoning closed and preserve plain content, prefilled thinking, and non-streaming extraction.
- Reuse the existing pending-marker state; do not change prompt-boundary initialization or the engine-core structured-output gate.

`forced_sequence.py`:

```python
import torch

from vllm.v1.sample.logits_processor import AdapterLogitsProcessor


class SequenceMask:
    def __init__(self, token_ids):
        self.token_ids = token_ids

    def __call__(self, output_ids, logits):
        index = min(len(output_ids), len(self.token_ids) - 1)
        logits.fill_(float("-inf"))
        logits[self.token_ids[index]] = 0
        return logits


class ForcedSequence(AdapterLogitsProcessor):
    def is_argmax_invariant(self):
        return False

    def new_req_logits_processor(self, params):
        token_ids = params.extra_args and params.extra_args.get("forced_sequence")
        return SequenceMask(token_ids) if token_ids else None
```

`test_parser_http.py`:

```python
import json
import os
from pathlib import Path

import pytest
from transformers import AutoTokenizer

from tests.utils import RemoteOpenAIServer


@pytest.fixture(scope="module")
def server(tmp_path_factory):
    model = os.environ.get("PARSER_TEST_MODEL", "facebook/opt-125m")
    tokenizer = AutoTokenizer.from_pretrained(model)
    tokenizer.add_special_tokens(
        {"additional_special_tokens": ["<mm:think>", "</mm:think>"]}
    )
    tokenizer_path = tmp_path_factory.mktemp("tokenizer")
    tokenizer.save_pretrained(tokenizer_path)
    template = tokenizer_path / "template.jinja"
    template.write_text(
        "{% for m in messages %}{{ m['role'] }}: {{ m['content'] }}\\n"
        "{% endfor %}assistant: ",
        encoding="utf-8",
    )
    args = [
        "--tokenizer", str(tokenizer_path),
        "--chat-template", str(template),
        "--reasoning-parser", "minimax_m3",
        "--logits-processors", "forced_sequence:ForcedSequence",
        "--served-model-name", "parser-regression",
        "--dtype", "float32",
        "--max-model-len", "128",
        "--max-num-seqs", "1",
        "--enforce-eager",
    ]
    with RemoteOpenAIServer(
        model, args,
        env_dict={"VLLM_USE_V2_MODEL_RUNNER": "0"},
        max_wait_seconds=300,
    ) as remote:
        yield remote, tokenizer


@pytest.mark.parametrize("holdback", [False, True])
@pytest.mark.parametrize("leading_closer", [False, True])
def test_parser_http(server, holdback, leading_closer):
    remote, tokenizer = server
    text = "</mm:think>answer" if leading_closer else "<mm:think>plan</mm:think>answer"
    token_ids = [
        token for char in text
        for token in tokenizer.encode(char, add_special_tokens=False)
    ]
    assert tokenizer.decode(token_ids) == text
    assert max(token_ids) < tokenizer.vocab_size
    kwargs = dict(
        model="parser-regression",
        messages=[{"role": "user", "content": "Continue."}],
        max_tokens=len(token_ids),
        temperature=0,
        extra_body={"vllm_xargs": {"forced_sequence": token_ids}},
    )
    if holdback:
        kwargs["stop"] = ["NEVER_MATCH"]
    client = remote.get_client()
    whole = client.chat.completions.create(**kwargs)
    message = whole.choices[0].message
    expected_reasoning = None if leading_closer else "plan"
    assert (getattr(message, "reasoning", None), message.content) == (
        expected_reasoning, "answer"
    )
    chunks = list(client.chat.completions.create(**kwargs, stream=True))
    reasoning = "".join(
        getattr(c.choices[0].delta, "reasoning", None) or ""
        for c in chunks if c.choices
    ) or None
    content = "".join(
        c.choices[0].delta.content or "" for c in chunks if c.choices
    )
    result = {"holdback": holdback, "leading_closer": leading_closer,
              "reasoning": reasoning, "content": content}
    print(json.dumps(result))
    assert (reasoning, content) == (expected_reasoning, "answer")
```

The token-selection fixture is intentionally deterministic and uses the supported logits-processor extension point. It is not a model-quality benchmark. Run the same fixture before and after the one-line parser change.

## Details

The detokenizer can withhold decoded text while already returning output token IDs, for example while buffering excluded stop strings. `DelegatingParser.parse_delta()` still checks `is_reasoning_end_streaming()` after extraction returns. With split marker IDs and no visible text, the existing predicate can treat the nonempty token history as plain content, latch `reasoning_ended`, and bypass the reasoner for the rest of the stream.

Marking the empty-text interval pending keeps that transition deferred until extraction can classify the visible text. The existing completed-reasoning check retains priority, so this does not reopen a completed reasoning block. The change adds no tokenizer-specific prompt delimiter or new interface.

The new tests are in the existing reasoning suite, selected by the active `cpu-reasoning-renderers` Buildkite step. The step's source dependencies already include `vllm/reasoning/`.
