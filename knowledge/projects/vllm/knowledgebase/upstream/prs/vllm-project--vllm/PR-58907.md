# [Perf][MRV2] Reduce cross-lane reductions in min-p filtering

[Upstream PR](https://github.com/vllm-project/vllm/pull/58907) · state: open · merged: False

## Original PR description

## Overview

Reduce MRV2 min-p scan overhead by accumulating per-lane maxima and reducing across lanes once after the vocabulary scan. Use an 8192-element tile with 16 warps for vocabularies above 16384 and batches up to 32, retaining 1024/4 otherwise.

## Claims

- Preserve one kernel launch, the log-space threshold, request mapping, and int64 row addressing, without a temporary tensor.
- Reduce min-p kernel latency for the measured large-vocabulary batches up to 32 on A100 and RTX 2080 Ti; fixed-parameter comparisons show an independent benefit from per-lane accumulation at small batches.
- Reduce elapsed time for the constructed Qwen3-0.6B application on A100 by 2.72% (batch 8) and 2.43% (batch 32) relative to upstream, with identical outputs in all measured runs.

## Details

This changes `vllm/v1/worker/gpu/sample/min_p.py`. It is distinct from #46539's log-space change in the older runner's logits processor. Current MRV2 sampling PRs and actual module diffs were checked, including #54332, #60566, and #45288; none implements this reduction or launch update.

The filter runs on steps using nonzero min-p. Default `min_p=0` bypasses it. The constructed application uses Qwen3-0.6B with min-p explicitly enabled to exercise this path; Qwen's default recommendation remains min-p=0.

Thanks to @kerry0520 for the [independent measurements and launch-configuration suggestions](https://github.com/vllm-project/vllm/pull/58907#issuecomment-6246069554) that informed this update, including the batch > 8 observation. This PR combines the original per-lane reduction with the launch tuning, correctness tests, and cross-GPU/model validation.

"""Constructed application: briefs -> outlines -> stories -> edited deliverable.

Compare the frozen min-p implementations on identical complete tasks. All text
processing, generation, serialization and output-file fsync are in task time.
Model construction/warmup and output-equality checks are outside task time.
"""
import argparse
import gc
import hashlib
import importlib.util
import json
import os
import time
from pathlib import Path

os.environ['VLLM_ENABLE_V1_MULTIPROCESSING'] = '0'
os.environ['VLLM_USE_V2_MODEL_RUNNER'] = '1'
os.environ['HF_HUB_OFFLINE'] = '1'
os.environ['TRANSFORMERS_OFFLINE'] = '1'
os.environ['OMP_NUM_THREADS'] = '1'

import torch
from vllm import LLM, SamplingParams
import vllm.v1.worker.gpu.sample.states as states

parser = argparse.ArgumentParser()
parser.add_argument('--model', required=True)
parser.add_argument('--output', required=True)
parser.add_argument('--artifact-dir', required=True)
args = parser.parse_args()
root = Path(args.output)
root.mkdir(parents=True, exist_ok=True)
artifact_root = Path(args.artifact_dir)
artifact_root.mkdir(parents=True, exist_ok=True)
briefs = [
    'A retired lighthouse keeper finds a letter inside a clock that stopped thirty years ago.',
    'A city gardener discovers that an abandoned train platform blooms only at midnight.',
    'A young baker must deliver a birthday cake to someone who has forgotten their own birthday.',
    'Two neighbors who have never spoken exchange messages through a misplaced library book.',
    'A village repairs its old wooden bridge and finds a time capsule from an unknown child.',
    'A traveling musician misses the last ferry and performs for the people waiting on the shore.',
    'An apprentice mapmaker notices that the smallest street on every map has disappeared.',
    'A shop owner lends umbrellas without asking for names and receives an unexpected thank-you.',
]
settings = ['a mountain village', 'a rainy coastal town', 'a busy railway district',
            'a quiet university town', 'a desert settlement', 'a snowy northern port',
            'a riverside neighborhood', 'a remote island', 'an old industrial town',
            'a small farming community', 'a crowded capital', 'a lakeside holiday town',
            'a forest settlement', 'a hillside suburb', 'a historic market town',
            'a windswept fishing village']
variants = {}
hashes = {}
for name in ('baseline', 'current', 'knob', 'patched'):
    path = Path(name + '_min_p.py')
    hashes[name] = hashlib.sha256(path.read_bytes()).hexdigest()
    spec = importlib.util.spec_from_file_location('workflow_' + name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    variants[name] = module.apply_min_p

start_init = time.perf_counter()
llm = LLM(model=args.model, dtype='half', max_model_len=2048,
          max_num_seqs=32, max_num_batched_tokens=2048,
          gpu_memory_utilization=0.45, enable_prefix_caching=False,
          seed=0, async_scheduling=True, disable_log_stats=True,
          compilation_config={'cudagraph_capture_sizes': [1, 8, 32]})
init_seconds = time.perf_counter() - start_init
gc_durations = []
gc_start = None


def observe_gc(phase, info):
    global gc_start
    if phase == 'start':
        gc_start = time.perf_counter()
    elif gc_start is not None:
        gc_durations.append(time.perf_counter() - gc_start)


gc.callbacks.append(observe_gc)
records = []
metadata = dict(
    scenario='constructed 128 short stories, groups of 8 or 32; not production traffic',
    model=args.model, gpu=torch.cuda.get_device_name(), torch=torch.__version__,
    source_sha256=hashes, seed=0, torch_threads=torch.get_num_threads(),
    flashinfer_sampler=os.environ.get('VLLM_USE_FLASHINFER_SAMPLER', 'auto'),
    engine_init_seconds=init_seconds, prefix_caching=False,
    temperature=0.8, top_p=1.0, top_k=-1, enable_thinking=False,
    stage_max_tokens=[192, 512, 512], ignore_eos=False,
    timing='warm engine; prompt construction through local-disk output JSON fsync; GC enabled',
    matrix=[dict(batch=8, min_p=.05, pairs=8),
            dict(batch=32, min_p=.05, pairs=8),
            dict(batch=8, min_p=0., pairs=4),
            dict(batch=32, min_p=0., pairs=4)],
    briefs=briefs, settings=settings, total_stories=128, artifact_dir=str(artifact_root),
)


def save_progress():
    (root / 'results.json').write_text(json.dumps(
        dict(metadata=metadata, records=records), indent=2), encoding='utf-8')


def run_task(variant, batch, min_p, label):
    count = 0

    def counted_filter(*inputs):
        nonlocal count
        count += 1
        return variants[variant](*inputs)

    states.apply_min_p = counted_filter
    gc_durations.clear()
    torch.cuda.synchronize()
    start = time.perf_counter()
    stage_outputs = []
    phases = []
    all_briefs = [f'{brief} Set the story in {setting}.'
                  for setting in settings for brief in briefs]
    for start_index in range(0, len(all_briefs), batch):
        group = start_index // batch
        selected = all_briefs[start_index:start_index + batch]
        previous = None
        for stage, limit in zip(('outline', 'draft', 'edit'), (192, 512, 512)):
            stage_start = time.perf_counter()
            calls_before = count
            conversations = []
            for i, brief in enumerate(selected):
                if stage == 'outline':
                    instruction = (
                        f'Plan a short story based on this brief: {brief}\n'
                        'Give exactly four concise plot beats, at most 70 words total. '
                        'Include a clear ending. Return only the outline.')
                elif stage == 'draft':
                    instruction = (
                        f'Brief: {brief}\nOutline:\n{previous[i]}\n'
                        'Write one complete micro-story paragraph. Follow the '
                        'outline, use concrete details, and reach the ending. '
                        'Return only the story, without commentary.')
                else:
                    instruction = (
                        f'Original brief: {brief}\nDraft:\n{previous[i]}\n'
                        'Edit this story for clarity and consistent character names. '
                        'Preserve its events and ending. Return the complete edited '
                        'micro-story as one paragraph, without an explanation.')
                conversations.append([
                    dict(role='system', content='You are a concise fiction writer and editor. Write in English.'),
                    dict(role='user', content=instruction),
                ])
            params = SamplingParams(temperature=.8, top_p=1., top_k=-1,
                                    min_p=min_p, seed=0, max_tokens=limit)
            outputs = llm.chat(conversations, params, use_tqdm=False,
                               chat_template_kwargs={'enable_thinking': False})
            previous = [o.outputs[0].text for o in outputs]
            stage_outputs.append([
                dict(group=group, brief=selected[i], prompt=conversations[i],
                     text=o.outputs[0].text, tokens=list(o.outputs[0].token_ids),
                     prompt_tokens=len(o.prompt_token_ids),
                     finish_reason=o.outputs[0].finish_reason,
                     stop_reason=o.outputs[0].stop_reason)
                for i, o in enumerate(outputs)
            ])
            phases.append(dict(group=group, stage=stage, seconds=time.perf_counter()-stage_start,
                               min_p_calls=count-calls_before,
                               generated_tokens=sum(len(o.outputs[0].token_ids) for o in outputs),
                               input_tokens=sum(len(o.prompt_token_ids) for o in outputs)))
    io_start = time.perf_counter()
    with (artifact_root / f'{label}.json').open('w', encoding='utf-8') as f:
        json.dump(dict(briefs=briefs, settings=settings, stages=stage_outputs), f, ensure_ascii=False, indent=2)
        f.flush()
        os.fsync(f.fileno())
    io_seconds = time.perf_counter()-io_start
    torch.cuda.synchronize()
    elapsed = time.perf_counter()-start
    measured_gc = sum(gc_durations)
    gc_count = len(gc_durations)
    signature = json.dumps(stage_outputs, sort_keys=True, ensure_ascii=False)
    row = dict(variant=variant, batch=batch, min_p=min_p, label=label,
               seconds=elapsed, min_p_calls=count, file_write_seconds=io_seconds,
               gc_seconds=measured_gc, gc_count=gc_count, stages=phases,
               generated_tokens=sum(p['generated_tokens'] for p in phases),
               finish_reasons=[r['finish_reason'] for stage in stage_outputs for r in stage],
               output_sha256=hashlib.sha256(signature.encode()).hexdigest())
    print(json.dumps(row), flush=True)
    return row, signature


try:
    for case in metadata['matrix']:
        batch, min_p, pairs = case['batch'], case['min_p'], case['pairs']
        expected = None
        for variant in variants:
            _, signature = run_task(variant, batch, min_p, f'warm-{batch}-{min_p}-{variant}')
            if expected is None:
                expected = signature
            assert signature == expected, ('warmup mismatch', batch, min_p, variant)
        for repeat in range(pairs):
            names = tuple(variants)
            rotated = names[repeat % len(names):] + names[:repeat % len(names)]
            order = rotated if (repeat // len(names)) % 2 == 0 else tuple(reversed(rotated))
            for variant in order:
                row, signature = run_task(variant, batch, min_p,
                                          f'b{batch}-p{min_p}-r{repeat}-{variant}')
                row['repeat'] = repeat
                row['outputs_equal'] = signature == expected
                records.append(row)
                save_progress()
                assert row['outputs_equal'], ('output mismatch', batch, min_p, repeat, variant)
    metadata['completed'] = True
    save_progress()
finally:
    states.apply_min_p = variants['baseline']
    llm.llm_engine.engine_core.shutdown()
