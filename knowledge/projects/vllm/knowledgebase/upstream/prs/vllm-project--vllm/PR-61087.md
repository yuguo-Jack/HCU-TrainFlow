# [Perf][LoRA] Avoid empty split-K programs in shrink

[Upstream PR](https://github.com/vllm-project/vllm/pull/61087) · state: open · merged: False

## Original PR description



## Overview

Cap LoRA shrink's configured split-K at the number of K blocks to avoid launching empty reduction programs. Fixes #61030; the proposed cap and original H100 evidence are credited to @ToukoUrsin.

## Claims

- With the default small-batch config and K=4096, reduce split-K from 64 to 16, removing 48 empty programs per M/N tile and adapter.
- Preserve useful K-block assignments and the fixed batch-invariant partition. Configs already at or below the cap retain their split count.
- Cover single-split stores, masked tails, non-power-of-two caps, and unchanged configurations with real GPU correctness and launch-grid assertions.

## Details

The default config for M<128 requests 64 splits with BLOCK_K=256. When the configured split count exceeds ceil(K/BLOCK_K), only the first ceil(K/BLOCK_K) splits have valid K data; each has one K iteration. The cap removes empty splits without changing useful tile assignments. `EVEN_K` is recomputed afterward, and the existing batch-invariant override still sets BLOCK_K=256 and SPLIT_K=8.

Ordinary multi-split accumulation continues to use floating-point atomics and is not bitwise deterministic. Reducing to one split switches to direct stores, covered by K=255/256 tests. The change introduces no new device synchronization and does not change operator schemas or default configuration tables.

Duplicate-work checks: no open PR referencing #61030 or implementing this cap was found. #59572 optimizes the M-dimension grid; #50278 forces split-K=1 for determinism. This PR preserves split-K parallelism up to the number of useful K blocks. The issue author proposed the exact one-line cap and is credited above and in the production commit.

**gpu_ab.py**

```python
"""Local A/B measurement; uses the real shrink entry point and metadata."""
import inspect
import json
import statistics
from pathlib import Path

import torch
import vllm.lora.ops.triton_ops.lora_shrink_op as op
import vllm.lora.ops.triton_ops.utils as utils
from vllm.lora.ops.triton_ops import LoRAKernelMeta

assert torch.cuda.is_available()
assert not op.is_batch_invariant
assert utils.envs.VLLM_TUNED_CONFIG_FOLDER is None
source = inspect.getsource(op._lora_shrink)
new = 'SPLIT_K = min(kernel_config["split_k"], triton.cdiv(K, BLOCK_K))'
old = 'SPLIT_K = kernel_config["split_k"]'
assert source.count(new) == 1
namespace = dict(op.__dict__)
exec(compile(source.replace(new, old), "<baseline-shrink>", "exec"), namespace)
baseline = namespace["_lora_shrink"]
candidate = op._lora_shrink
results = []
for m, k, num_loras, nslices in [(32,4096,1,1), (32,4096,16,1),
                               (32,4096,32,1), (32,4096,16,3),
                               (32,256,4,1), (128,4096,16,1)]:
    torch.manual_seed(0)
    x = torch.randn((m,k), device="cuda", dtype=torch.float16)
    weights = [torch.randn((num_loras,16,k), device="cuda", dtype=x.dtype)
               for _ in range(nslices)]
    mapping = torch.arange(m,device="cuda",dtype=torch.int32) % num_loras
    mapping[-1] = -1
    meta = LoRAKernelMeta.make(num_loras,m,device="cuda")
    meta.prepare_tensors(mapping)
    args = meta.meta_args(token_nums=m,specialize_active_lora=False)
    outputs = [torch.empty((nslices,m,16),device="cuda",dtype=torch.float32)
               for _ in range(2)]
    utils._LORA_A_PTR_DICT.clear()
    for f,out in zip((baseline,candidate),outputs):
        for _ in range(5):
            f(x,weights,out,*args,0.5)
    torch.cuda.synchronize()
    ref = torch.zeros((nslices,m,16),dtype=torch.float64)
    for s,w in enumerate(weights):
        for lid in range(num_loras):
            mask = mapping.cpu() == lid
            ref[s,mask] = 0.5*x.cpu().double()[mask] @ w.cpu().double()[lid].T
    for out in outputs:
        torch.testing.assert_close(out.cpu().double(),ref,rtol=1e-2,atol=1e-2)
    graphs = []
    for f,out in zip((baseline,candidate),outputs):
        graph = torch.cuda.CUDAGraph()
        with torch.cuda.graph(graph):
            for _ in range(200):
                f(x,weights,out,*args,0.5)
        graphs.append(graph)
    samples = [[],[]]
    for repeat in range(7):
        for which in ((0,1) if repeat % 2 == 0 else (1,0)):
            start,end = (torch.cuda.Event(enable_timing=True) for _ in range(2))
            start.record()
            for _ in range(20):
                graphs[which].replay()
            end.record()
            end.synchronize()
            samples[which].append(start.elapsed_time(end)*1000/4000)
    medians = [statistics.median(times) for times in samples]
    row = dict(m=m,k=k,num_loras=num_loras,nslices=nslices,
               baseline_us=medians[0],candidate_us=medians[1],
               speedup=medians[0]/medians[1],samples_us=samples)
    results.append(row)
    print(json.dumps(row),flush=True)
    del graphs, outputs, weights, x, meta
    utils._LORA_A_PTR_DICT.clear()
    torch.cuda.empty_cache()
report = dict(torch=torch.__version__,cuda=torch.version.cuda,
              gpu=torch.cuda.get_device_name(),results=results,
              method="7 alternating A/B rounds; 200 calls/graph x 20 replays; includes output zeroing; FP16, rank16, scaling0.5")
Path(__file__).with_name("gpu-benchmark.json").write_text(json.dumps(report,indent=2))
```

**serving_repro.py**

```python
import gc
import inspect
import json
import os
import statistics
import time
from pathlib import Path

os.environ["VLLM_ENABLE_V1_MULTIPROCESSING"] = "0"
os.environ["VLLM_DISABLE_COMPILE_CACHE"] = "1"
import torch
from transformers import AutoModelForCausalLM
from peft import LoraConfig, get_peft_model

root = Path('/tmp/vllm-lora-split-k-serving')
root.mkdir(exist_ok=True)
from huggingface_hub import snapshot_download
model_path = snapshot_download('Qwen/Qwen3-0.6B', revision='c1899de289a04d12100db370d81485cdf75e47ca')
adapter = root / 'adapter'
if not adapter.exists():
    torch.manual_seed(0)
    base = AutoModelForCausalLM.from_pretrained(model_path,dtype=torch.float16,local_files_only=True)
    peft = get_peft_model(base,LoraConfig(r=16,lora_alpha=16,
        target_modules=['q_proj','v_proj','o_proj','gate_proj','up_proj','down_proj'],
        task_type='CAUSAL_LM'))
    with torch.no_grad():
        for name,param in peft.named_parameters():
            if 'lora_B' in name:
                param.normal_(std=0.002)
    peft.save_pretrained(adapter)
    del peft,base
    gc.collect()

from vllm import LLM, SamplingParams
from vllm.lora.request import LoRARequest
import vllm.lora.ops.triton_ops.lora_shrink_op as op

candidate_code = op._lora_shrink.__code__
source = inspect.getsource(op._lora_shrink).replace('def _lora_shrink(', 'def _baseline(')
source = source.replace('SPLIT_K = min(kernel_config["split_k"], triton.cdiv(K, BLOCK_K))','SPLIT_K = kernel_config["split_k"]')
exec(compile(source,'<baseline-shrink>','exec'),op.__dict__)
baseline_code = op._baseline.__code__
llm = LLM(model=model_path,enable_lora=True,max_loras=1,max_lora_rank=16,
          dtype='float16',max_model_len=128,max_num_seqs=4,
          max_num_batched_tokens=128,gpu_memory_utilization=0.40,
          kv_cache_memory_bytes=32*1024*1024,enforce_eager=True,
          attention_config={'backend':'TRITON_ATTN'},enable_prefix_caching=False,
          seed=0)
request = LoRARequest('local-test',1,str(adapter))
params = SamplingParams(temperature=0,max_tokens=16,ignore_eos=True,logprobs=1)
prompts = ['The capital of France is','Write a short Python function that adds two numbers.',
           'Explain why the sky is blue in one sentence.','The result of 2 + 2 is']
samples = {'baseline':[],'candidate':[]}
records = {}
call_grids = []
original_kernel = op._lora_shrink_kernel
class RecordKernel:
    def __getitem__(self,grid):
        call_grids.append(grid)
        return original_kernel[grid]
op._lora_shrink_kernel = RecordKernel()
try:
    for label,code in [('baseline',baseline_code),('candidate',candidate_code)]:
        op._lora_shrink.__code__ = code
        outputs = llm.generate(prompts,params,lora_request=request,use_tqdm=False)
        records[label] = [dict(tokens=list(o.outputs[0].token_ids),text=o.outputs[0].text,
            logprobs=[float(p[t].logprob) for t,p in zip(o.outputs[0].token_ids,o.outputs[0].logprobs)])
            for o in outputs]
        print(label,json.dumps(records[label]),flush=True)
    assert [o['tokens'] for o in records['baseline']] == [o['tokens'] for o in records['candidate']]
    max_logprob_diff = max(abs(a-b) for old,new in zip(records['baseline'],records['candidate'])
                          for a,b in zip(old['logprobs'],new['logprobs']))
    recorded_launch_count = len(call_grids)
    op._lora_shrink_kernel = original_kernel
    for repeat in range(5):
        for label,code in ([('baseline',baseline_code),('candidate',candidate_code)]
                           if repeat % 2 == 0 else [('candidate',candidate_code),('baseline',baseline_code)]):
            op._lora_shrink.__code__ = code
            torch.cuda.synchronize()
            start = time.perf_counter()
            llm.generate(prompts,params,lora_request=request,use_tqdm=False)
            torch.cuda.synchronize()
            samples[label].append(time.perf_counter()-start)
    assert call_grids
    report = dict(model='Qwen/Qwen3-0.6B',model_revision=Path(model_path).name,
                  adapter='random nonzero LoRA rank16; seed0; B std0.002',
                  batch=4,output_tokens=16,eager=True,token_ids_equal=True,
                  max_generated_logprob_diff=max_logprob_diff,results=records,
                  samples_seconds=samples,medians_seconds={k:statistics.median(v) for k,v in samples.items()},
                  shrink_launch_count=recorded_launch_count)
    print(json.dumps(report),flush=True)
    Path(__file__).with_name('serving-results.json').write_text(json.dumps(report,indent=2))
finally:
    op._lora_shrink.__code__ = candidate_code
    op._lora_shrink_kernel = original_kernel
```
