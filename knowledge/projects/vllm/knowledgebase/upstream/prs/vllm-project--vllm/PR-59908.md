# [Core][V2] Send new requests' token ids to the workers as int32 arrays

[Upstream PR](https://github.com/vllm-project/vllm/pull/59908) · state: open · merged: False

## Original PR description

## Purpose

With the v2 model runner, every new request's prompt is sent to the workers twice as Python lists (`prompt_token_ids` and its copy `prefill_token_ids`), and each worker writes the prefill ids into the UVA staging buffer one int at a time. With long prompts this puts tens of ms of CPU between the last decode step and a large prefill step.

This PR:
- shares one list between the two fields while the request has no output tokens;
- pickles the token ids as int32 arrays (v2 only; not M-RoPE or pooling) and stages them with numpy;
- packs each prompt once in the EngineCore input thread (skipped for in-process executors).

The token ids themselves are unchanged. No open PR covers this. Related: #51334 (open) keeps `StagedWriteTensor` writes as NumPy chunks, overlapping only the worker staging part here; #45840 (merged) trims the resumed-request `all_token_ids` copy.

```python
# SPDX-License-Identifier: Apache-2.0
"""CPU cost of shipping new requests' token ids to v2 model-runner workers.

Runs one prefill step's worth of new requests through the same code the v2
path uses: the scheduler's NewRequestData.from_request call, the pickling done
by the shm MessageQueue (EngineCore -> workers), and the worker's
RequestState.add_request + apply_staged_writes. Run it unchanged on main and on
the PR branch:

    python bench_token_transport.py [--num-reqs 6] [--prompt-len 116000]

The stage + apply part needs a GPU (the token buffer is UVA); the rest is CPU.
"total" is the scheduling-path work (build + pickle + unpickle + stage). On
the PR branch, the input-thread prepacking is timed and printed separately;
--no-prepack skips it, so the prompts are packed inside pickle instead (the
path of resumed / streaming requests).
"""

import argparse
import pickle
import random
import time
from array import array

import torch

import vllm.v1.core.sched.scheduler as sched_mod
from vllm.sampling_params import SamplingParams
from vllm.v1.core.sched.output import NewRequestData
from vllm.v1.request import Request

PR_PATH = hasattr(sched_mod, "_v2_prefill_token_ids")


def make_requests(num_reqs: int, prompt_len: int) -> list[Request]:
    rng = random.Random(0)
    reqs = []
    for i in range(num_reqs):
        prompt = [rng.randrange(150_000) for _ in range(prompt_len)]
        req = Request(str(i), prompt, SamplingParams(max_tokens=16), None)
        req.num_computed_tokens = prompt_len * 9 // 10  # ~90% prefix-cache hit
        reqs.append(req)
    return reqs


def scheduler_new_reqs(reqs: list[Request], block_size: int):
    """The v2 branch of Scheduler.schedule() building NewRequestData."""
    out = []
    for req in reqs:
        block_ids = (list(range(len(req.prompt_token_ids) // block_size + 1)),)
        if PR_PATH:
            out.append(
                NewRequestData.from_request(
                    req,
                    block_ids,
                    sched_mod._v2_prefill_token_ids(req),
                    uses_mrope=False,
                    pack_token_ids=True,
                )
            )
        else:
            out.append(
                NewRequestData.from_request(
                    req, block_ids, req._all_token_ids, uses_mrope=False
                )
            )
    return out


def ms(t0: float, t1: float) -> float:
    return (t1 - t0) * 1e3


def main() -> None:
    p = argparse.ArgumentParser()
    p.add_argument("--num-reqs", type=int, default=6)
    p.add_argument("--prompt-len", type=int, default=116_000)
    p.add_argument("--repeat", type=int, default=5)
    p.add_argument("--no-prepack", action="store_true")
    args = p.parse_args()
    block_size = 128

    req_states = None
    if torch.cuda.is_available():
        from vllm.v1.worker.gpu.states import RequestState

        req_states = RequestState(
            max_num_reqs=args.num_reqs,
            max_model_len=args.prompt_len + 1024,
            max_num_batched_tokens=8192,
            num_speculative_steps=0,
            vocab_size=150_000,
            device=torch.device("cuda"),
        )

    prepack = PR_PATH and not args.no_prepack
    path = "main (lists)"
    if PR_PATH:
        path = "int32 arrays (this PR)" + ("" if prepack else ", no prepack")
    print(f"code path: {path}")
    print(f"{args.num_reqs} new requests x {args.prompt_len} prompt tokens")
    rows = []
    for _ in range(args.repeat):
        reqs = make_requests(args.num_reqs, args.prompt_len)
        tp = time.perf_counter()
        if prepack:
            # EngineCore.preprocess_add_request (input thread, off the
            # scheduling path) packs each prompt once.
            for req in reqs:
                req.packed_prompt_token_ids = array("i", req.prompt_token_ids)
        t0 = time.perf_counter()
        new_reqs = scheduler_new_reqs(reqs, block_size)
        t1 = time.perf_counter()
        # MessageQueue.enqueue / dequeue pickle at the highest protocol.
        blob = pickle.dumps(new_reqs, protocol=pickle.HIGHEST_PROTOCOL)
        t2 = time.perf_counter()
        got = pickle.loads(blob)
        t3 = time.perf_counter()
        stage = float("nan")
        if req_states is not None:
            # GPUModelRunner.add_requests -> RequestState.add_request, then
            # apply_staged_writes before the forward.
            for d in got:
                req_states.add_request(
                    req_id=d.req_id,
                    prompt_len=d.prompt_len,
                    all_token_ids=d.prefill_token_ids,
                    num_computed_tokens=d.num_computed_tokens,
                    max_tokens=16,
                )
            req_states.apply_staged_writes()
            torch.cuda.synchronize()
            stage = ms(t3, time.perf_counter())
            for d in got:
                idx = req_states.req_id_to_index[d.req_id]
                ref = reqs[int(d.req_id)].prompt_token_ids
                assert req_states.all_token_ids.gpu[idx, : len(ref)].tolist() == ref
                req_states.remove_request(d.req_id)
        rows.append(
            (ms(t0, t1), ms(t1, t2), ms(t2, t3), stage, len(blob), ms(tp, t0))
        )

    def total(r):
        return r[0] + r[1] + r[2] + (r[3] if req_states is not None else 0.0)

    best = min(rows, key=total)
    print(
        f"build {best[0]:.1f} ms | pickle {best[1]:.1f} ms | unpickle {best[2]:.1f} ms"
        f" | stage + apply {best[3]:.1f} ms | {best[4] / 1e6:.1f} MB"
        + ("" if req_states is not None else " (no GPU: stage skipped)")
    )
    print(f"total {total(best):.1f} ms")
    if prepack:
        print(f"input-thread prepack {best[5]:.1f} ms (not in total)")


if __name__ == "__main__":
    main()
```
