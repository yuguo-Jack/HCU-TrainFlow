# [Kernel] Tune small-M block-FP8 Triton configs for NVIDIA L20

[Upstream PR](https://github.com/vllm-project/vllm/pull/59232) · state: open · merged: False

## Original PR description

## Overview

L20 currently falls back to generic block-FP8 Triton parameters for these small-M shapes. Add five device-specific configuration tables and a 20-case numerical regression in the existing kernel suite. The change improves the measured decode kernels and shows a client-visible decode benefit under the qualified graph profile below.

## Claims

- Six changed files: five JSON tables and one existing test file. Kernel implementation, backend selection and serving code are unchanged.
- 165 anchors: 152 nondefault entries confined to M=1..32 and 13 generic-equivalent entries. M>=33 retains generic-equivalent parameters through the existing lookup. Two tables retain defaults at M=17..20.
- Targets block128x128, BF16 output and VLLM_BATCH_INVARIANT=0. The measured serving improvement concerns decode; no TTFT or independent model-quality acceptance is claimed.

"""Compare the frozen L20 FP8 tables with the production generic parameters.

Requires a candidate vLLM environment, FlashInfer and cupti-python. No tuning,
model requests or serving-performance inference. Raw times are milliseconds.
"""

import argparse
import ast
import hashlib
import importlib.metadata
import inspect
import json
import math
import statistics
import time
import warnings
from pathlib import Path

TABLES = [
    (14336, 5120, "8a89853ce3c8344b22a304874be049b3a7a9bd458368fb5f10fad090c9b030be"),
    (16384, 5120, "d39f41ff5c95cf518a83906408481f73550e0136afc17f7e63d6bd71bc867377"),
    (34816, 5120, "bfe6889250acbd8686f63de047a1386bd5935596c5f1496f3ae412204d2e2c23"),
    (5120, 17408, "608f8901b58249e48c5d452aa2f2266b778af3206e6178a7514fa1f15003d874"),
    (5120, 6144, "b300fcc7e1b8711f2c5ed2475c6c4ad934364da79be8a256f0527540cd23159f"),
]
DEFAULT = dict(
    BLOCK_SIZE_M=64,
    BLOCK_SIZE_N=128,
    BLOCK_SIZE_K=128,
    GROUP_SIZE_M=32,
    num_warps=4,
    num_stages=2,
)


def require(value, message):
    if not value:
        raise RuntimeError(message)


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def generic_config(source):
    """Check the comparison baseline against the actual production fallback."""
    public = next(
        n
        for n in ast.parse(source).body
        if isinstance(n, ast.FunctionDef) and n.name == "w8a8_triton_block_scaled_mm"
    )
    assignments = [
        n.value
        for n in ast.walk(public)
        if isinstance(n, ast.Assign)
        and any(isinstance(t, ast.Name) and t.id == "config" for t in n.targets)
        and isinstance(n.value, ast.Dict)
    ]
    require(len(assignments) == 1, "Unrecognized production default")

    def resolve(node):
        if isinstance(node, ast.Constant):
            return ast.literal_eval(node)
        if isinstance(node, ast.Name):
            return {"block_n": 128, "block_k": 128, "block_size": [128, 128]}[node.id]
        if isinstance(node, ast.Subscript):
            return resolve(node.value)[ast.literal_eval(node.slice)]
        raise RuntimeError("Unrecognized production config expression")

    return {
        ast.literal_eval(k): resolve(v)
        for k, v in zip(assignments[0].keys, assignments[0].values)
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--timeout-seconds", type=int, default=500)
    parser.add_argument(
        "--expected-kernel-sha256",
        default=("013df536b819fda78ff0864b7d8d25e419c8e6eb9b58af2d66bf0887bdda0749"),
    )
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    started = time.monotonic()
    result = dict(
        success=False,
        model_requests=0,
        serving_speedup_claim_allowed=False,
        script_sha256=sha(__file__),
        seed=0,
        timing_unit="milliseconds",
    )
    try:
        import cupti  # noqa: F401
        import torch
        from flashinfer.testing import bench_gpu_time_with_cupti

        from vllm import envs
        from vllm.model_executor.layers.quantization.utils import fp8_utils as ops
        from vllm.triton_utils import triton

        require(not envs.VLLM_BATCH_INVARIANT, "Requires VLLM_BATCH_INVARIANT=0")
        require(torch.cuda.device_count() == 1, "Expose exactly one GPU")
        gpu = torch.cuda.get_device_properties(0)
        require(
            gpu.name == "NVIDIA L20" and (gpu.major, gpu.minor) == (8, 9), "L20 only"
        )
        production = Path(inspect.getsourcefile(ops))
        require(
            sha(production) == args.expected_kernel_sha256, "Kernel source SHA changed"
        )
        require(
            generic_config(production.read_text(encoding="utf-8")) == DEFAULT,
            "Default config changed",
        )
        result.update(
            gpu=dict(name=gpu.name, uuid=str(gpu.uuid), capability=[8, 9]),
            torch=torch.__version__,
            triton=triton.__version__,
            versions={
                p: importlib.metadata.version(p)
                for p in ("vllm", "flashinfer-python", "cupti-python")
            },
            kernel_sha256=sha(production),
            default_config=DEFAULT,
            timing=dict(
                method="CUPTI",
                cuda_graph=True,
                cold_l2=True,
                dry_run_iters=20,
                repeat_iters=100,
                rounds=2,
                arm_order=["stock", "candidate", "candidate", "stock"],
            ),
            tables=[],
            correctness_checks=0,
            public_exact_match_checks=0,
            timing_records=0,
        )
        torch.backends.cuda.matmul.allow_tf32 = False
        generator = torch.Generator(device="cuda").manual_seed(0)
        with (args.output / "raw.jsonl").open("x", buffering=1) as raw:

            def emit(kind, **fields):
                raw.write(json.dumps(dict(kind=kind, **fields), allow_nan=False) + "\n")

            def budget():
                require(
                    time.monotonic() - started < args.timeout_seconds,
                    "Time budget reached",
                )

            for n, k, expected in TABLES:
                filename = (
                    f"N={n},K={k},device_name=NVIDIA_L20,dtype=fp8_w8a8,"
                    "block_shape=[128,128].json"
                )
                path = production.parent / "configs" / filename
                require(sha(path) == expected, "Table SHA changed: " + filename)
                table = {
                    int(m): cfg for m, cfg in json.loads(path.read_bytes()).items()
                }
                require(
                    set(table) == set(range(1, 34)) and table[33] == DEFAULT,
                    "Anchors changed",
                )
                timing_ms = (
                    [1, 2, 8, 16, 17, 20, 21, 32, 33, 64]
                    if n == 14336
                    else [1, 8, 16, 17, 20, 32, 33, 64]
                )
                result["tables"].append(
                    dict(N=n, K=k, sha256=expected, timing_Ms=timing_ms)
                )
                b = (torch.randn((n, k), device="cuda", generator=generator) * 0.1).to(
                    torch.float8_e4m3fn
                )
                bs = (
                    torch.rand((n // 128, k // 128), device="cuda", generator=generator)
                    * 0.1
                )
                bd = b.float() * bs.repeat_interleave(128, 0).repeat_interleave(128, 1)
                for m in list(range(1, 34)) + [64]:
                    budget()
                    a = (
                        torch.randn((m, k), device="cuda", generator=generator) * 0.1
                    ).to(torch.float8_e4m3fn)
                    scales = (
                        torch.rand((m, k // 128), device="cuda", generator=generator)
                        * 0.1
                    )
                    ref = ((a.float() * scales.repeat_interleave(128, 1)) @ bd.T).to(
                        torch.bfloat16
                    )
                    config = dict(
                        stock=DEFAULT,
                        candidate=table[min(table, key=lambda x: abs(x - m))],
                    )
                    outputs = {
                        arm: torch.empty((m, n), device="cuda", dtype=torch.bfloat16)
                        for arm in config
                    }

                    def launch(
                        arm,
                        outputs=outputs,
                        config=config,
                        m=m,
                        n=n,
                        k=k,
                        a=a,
                        b=b,
                        scales=scales,
                        bs=bs,
                    ):
                        c = outputs[arm]
                        cfg = config[arm]
                        grid = (
                            triton.cdiv(m, cfg["BLOCK_SIZE_M"])
                            * triton.cdiv(n, cfg["BLOCK_SIZE_N"]),
                        )
                        ops._w8a8_triton_block_scaled_mm[grid](
                            a,
                            b,
                            c,
                            scales,
                            bs,
                            m,
                            n,
                            k,
                            128,
                            128,
                            a.stride(0),
                            a.stride(1),
                            b.stride(1),
                            b.stride(0),
                            c.stride(0),
                            c.stride(1),
                            scales.stride(0),
                            scales.stride(1),
                            bs.stride(1),
                            bs.stride(0),
                            **cfg,
                        )

                    for arm in config:
                        launch(arm)
                        torch.cuda.synchronize()
                        observed = outputs[arm].float()
                        require(
                            bool(torch.isfinite(observed).all()), "Nonfinite output"
                        )
                        error = float(
                            (observed - ref.float()).abs().mean()
                            / ref.float().abs().mean()
                        )
                        require(
                            math.isfinite(error) and error < 0.001,
                            "Numerical tolerance",
                        )
                        emit(
                            "correctness",
                            N=n,
                            K=k,
                            M=m,
                            arm=arm,
                            normalized_error=error,
                        )
                        result["correctness_checks"] += 1
                        if m in timing_ms:
                            for _ in range(20):
                                launch(arm)
                    public = ops.w8a8_triton_block_scaled_mm(
                        a, b, scales, bs, [128, 128], torch.bfloat16
                    )
                    require(
                        bool(torch.equal(public, outputs["candidate"])),
                        "Public operation mismatch",
                    )
                    result["public_exact_match_checks"] += 1
                    emit("public_lookup", N=n, K=k, M=m, exact_match=True)
                    if m in timing_ms:
                        for round_id in range(2):
                            for position, arm in enumerate(
                                ("stock", "candidate", "candidate", "stock")
                            ):
                                budget()
                                with warnings.catch_warnings():
                                    warnings.filterwarnings(
                                        "error", message=".*[Ff]alling back.*"
                                    )
                                    samples = [
                                        float(t)
                                        for t in bench_gpu_time_with_cupti(
                                            lambda arm=arm, launch=launch: launch(arm),
                                            dry_run_iters=20,
                                            repeat_iters=100,
                                            use_cuda_graph=True,
                                            cold_l2_cache=True,
                                        )
                                    ]
                                require(
                                    len(samples) == 100
                                    and all(
                                        math.isfinite(t) and t > 0 for t in samples
                                    ),
                                    "CUPTI samples",
                                )
                                median = statistics.median(samples)
                                byte_min = (
                                    m * k
                                    + n * k
                                    + 4 * m * (k // 128)
                                    + 4 * (n // 128) * (k // 128)
                                    + 2 * m * n
                                )
                                emit(
                                    "timing",
                                    N=n,
                                    K=k,
                                    M=m,
                                    round=round_id,
                                    position=position,
                                    arm=arm,
                                    config=config[arm],
                                    input_pointers=[
                                        x.data_ptr() for x in (a, b, scales, bs)
                                    ],
                                    samples_ms=samples,
                                    median_us=median * 1000,
                                    dense_math_tflops=2 * m * n * k / (median * 1e9),
                                    minimum_traffic_gbps=byte_min / (median * 1e6),
                                )
                                result["timing_records"] += 1
                    del a, scales, ref, outputs, public, observed
                del b, bs, bd, launch
                print(
                    json.dumps(
                        dict(
                            shape_complete=[n, k],
                            timing_records=result["timing_records"],
                        )
                    ),
                    flush=True,
                )
        require(
            (
                result["correctness_checks"],
                result["public_exact_match_checks"],
                result["timing_records"],
            )
            == (340, 170, 336),
            "Incomplete coverage",
        )
        result["success"] = True
    except Exception as error:
        result["error"] = type(error).__name__ + ": " + str(error)
        raise
    finally:
        result["elapsed_seconds"] = time.monotonic() - started
        (args.output / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
