# [Bugfix][Multimodal] Fix DeepSeek-OCR zero-sum patch guard in _encode…

[Upstream PR](https://github.com/vllm-project/vllm/pull/64091) · state: closed · merged: True

## Original PR description

Fixes #64075.

_encode_local_features in deepseek_ocr.py and deepseek_ocr2.py used torch.sum(patches).item() == 0 to detect "no crop patches", so a tiled image whose normalized pixels cancel to zero (e.g. half-white/half-black 1536x768) wrongly returned None, dropped local features, and crashed EngineCore with a placeholder/embedding shape mismatch. Replace the guard with patches.numel() == 0; #59417 fixed the equivalent check in _parse_and_validate_image_input but left this one as "not engine-fatal", which #64075 disproves.

New test class TestDeepseekOCREncodeLocalFeaturesGuard fails on the base commit (2 failed) and passes with the fix (13 passed in tests/models/multimodal/processing/test_deepseek_ocr.py); ruff check and ruff format are clean.

```text
import os

import pytest
import torch
from PIL import Image

from vllm.transformers_utils.processors.deepseek_ocr import (
    ImageTransform,
    dynamic_preprocess,
)


@pytest.mark.core_model
@pytest.mark.parametrize(
    ("model_id", "image_size", "model_env", "load_format"),
    [
        pytest.param(
            "deepseek-ai/DeepSeek-OCR",
            640,
            "DEEPSEEK_OCR_TEST_MODEL",
            "dummy",
            id="ocr1",
        ),
        pytest.param(
            "deepseek-ai/DeepSeek-OCR-2",
            768,
            "DEEPSEEK_OCR2_TEST_MODEL",
            "auto",
            id="ocr2",
        ),
    ],
)
def test_deepseek_ocr_balanced_tiled_image(
    vllm_runner, model_id, image_size, model_env, load_format
):
    image = Image.new("RGB", (1536, 768), "white")
    image.paste("black", (768, 0, 1536, 768))

    crops, crop_ratio = dynamic_preprocess(image, image_size=image_size)
    patches = torch.stack([ImageTransform()(crop) for crop in crops])
    assert crop_ratio == (2, 1)
    assert patches.numel() > 0
    assert patches.sum().item() == 0

    model = os.environ.get(model_env, model_id)
    with vllm_runner(
        model,
        max_model_len=2048,
        gpu_memory_utilization=0.8,
        load_format=load_format,
    ) as vllm_model:
        outputs = vllm_model.generate_greedy(
            ["<image>\nFree OCR."], max_tokens=8, images=[[image]]
        )

    assert len(outputs) == 1
    assert len(outputs[0][0]) > 0
```
