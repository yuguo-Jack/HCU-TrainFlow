# [ROCm][Spec Decode] Rank unified attention above ROCM_ATTN for spec decode and hybrid models

[Upstream PR](https://github.com/vllm-project/vllm/pull/61433) · state: open · merged: False

## Original PR description

## Overview

On ROCm, auto-selection now ranks `ROCM_ATTN` below the unified attention backends (`ROCM_AITER_UNIFIED_ATTN`, `TRITON_ATTN`) when speculative decoding is on or the model is hybrid. In both cases `ROCM_ATTN` falls back to its Triton prefix-prefill kernel on every step.

## Claims

- Qwen3.8-27B + `z-lab/Qwen3.8-27B-DFlash2` (7 draft tokens) on MI355X: output throughput is +11% / +31% / +54% at concurrency 1 / 16 / 64.
- The same hybrid model without spec decode gains +5% / +10% / +10%.
- `ROCM_ATTN` still works when selected explicitly with `--attention-backend ROCM_ATTN`, and its priority is unchanged for other models.

## Validation

Setup: MRV2, ShareGPT, 512 output tokens, temperature 0, one run per point, output tok/s.

```
  [--speculative-config '{"method": "dflash", "model": "z-lab/Qwen3.8-27B-DFlash2", "num_speculative_tokens": 7}']
vllm bench serve --dataset-name sharegpt --sharegpt-output-len 512 --temperature 0 \
  --num-prompts max(64, 8*C) --max-concurrency C
```

## Details

**Root cause.** `ROCM_ATTN` uses its HIP paged-attention kernel only for single-token decode with a power-of-two block size. Otherwise it runs `context_attention_fwd`, the Triton prefix-prefill kernel:
- Speculative decoding verifies several tokens per request, so every decode step takes that fallback.
- Hybrid models enlarge the attention block size to match the mamba page size (832 for Qwen3.8-27B), which disables the HIP kernel even for single-token decode.

The unified backends handle multi-token queries and any block size in one kernel.

**Change.** `_rocm_attn_falls_back_to_triton(vllm_config)` detects these two cases. When it holds, `get_valid_backends` moves `ROCM_ATTN` to just after `TRITON_ATTN`. `ROCM_ATTN` stays valid, so explicit selection still works.

**Trade-off.** A spec decode or hybrid deployment that relied on auto-selecting `ROCM_ATTN` will now get a unified backend. I validated the speedup on one hybrid model with DFlash2 and without spec decode, but not on dense models with EAGLE/MTP.

No host synchronization is added; the decision is made once at configuration time.

```python
"""Build fixed-length chat prompts from ShareGPT text for the ISL/OSL sweep.

Usage: python3 gen_prompts.py <tokenizer> <out_dir> [sharegpt.json]

Writes <out_dir>/isl{ISL}_c{C}.jsonl (and _warm) with prompts that are ISL-8..ISL
tokens after the chat template. Every prompt uses a disjoint slice of the
corpus, so prefix caching never hits across requests or runs.
"""

import json
import os
import sys

from transformers import AutoTokenizer

# ISL -> concurrencies; prompts per run = max(MIN_PROMPTS[isl], PER_C[isl] * C)
PLAN = {
    1024: [1, 16, 64],
    8192: [1, 16, 64],
    65536: [1, 8, 16],
    130048: [1, 4, 8],
}
PER_C = {1024: 4, 8192: 4, 65536: 2, 130048: 2}
MIN_PROMPTS = {1024: 8, 8192: 8, 65536: 4, 130048: 4}
WARM = 2

HEAD = "Here are some past conversations between users and an assistant:\n\n"
TAIL = "\n\nSummarize the conversations above in detail."


def num_prompts(isl, c):
    return max(MIN_PROMPTS[isl], PER_C[isl] * c)


def main():
    tok_path, out = sys.argv[1], sys.argv[2]
    os.makedirs(out, exist_ok=True)
    tok = AutoTokenizer.from_pretrained(tok_path)
    if len(sys.argv) > 3:
        ds = sys.argv[3]
    else:
        from huggingface_hub import hf_hub_download

        ds = hf_hub_download(
            "anon8231489123/ShareGPT_Vicuna_unfiltered",
            "ShareGPT_V3_unfiltered_cleaned_split.json",
            repo_type="dataset",
        )
    convs = json.load(open(ds))

    def stream():
        for conv in convs:
            text = "\n".join(
                f"{t['from']}: {t['value']}" for t in conv.get("conversations", [])
            )
            if text:
                yield tok.encode(text + "\n\n", add_special_tokens=False)

    it = stream()
    buf: list[int] = []

    def take(n):
        nonlocal buf
        while len(buf) < n:
            buf += next(it)
        ids, buf = buf[:n], buf[n:]
        return ids

    def wrap(body):
        msgs = [{"role": "user", "content": HEAD + body + TAIL}]
        return tok.apply_chat_template(msgs, tokenize=False, add_generation_prompt=True)

    overhead = len(tok.encode(wrap(""), add_special_tokens=False))
    for isl, concs in PLAN.items():
        runs = [("warm", WARM)] + [(f"c{c}", num_prompts(isl, c)) for c in concs]
        for name, n in runs:
            with open(f"{out}/isl{isl}_{name}.jsonl", "w") as f:
                lens = []
                for _ in range(n):
                    target = isl - overhead
                    for _ in range(5):
                        prompt = wrap(tok.decode(take(target)))
                        got = len(tok.encode(prompt, add_special_tokens=False))
                        # Never over ISL, so ISL + OSL fits max-model-len 131072.
                        if 0 <= isl - got <= 8:
                            break
                        target -= got - isl + (got > isl)
                    lens.append(got)
                    f.write(json.dumps({"prompt": prompt}) + "\n")
            print(f"isl{isl}_{name}: {n} prompts, len {min(lens)}..{max(lens)}")


if __name__ == "__main__":
    main()
```

```python
"""Print main-vs-PR markdown tables from results/*/summary.txt."""

import glob
import re
import sys
from collections import defaultdict

root = sys.argv[1] if len(sys.argv) > 1 else "results"
rows = defaultdict(dict)  # (cfg, isl, c) -> variant -> metrics
for f in glob.glob(f"{root}/*/summary.txt"):
    for line in open(f):
        m = re.match(r"(\w+)__(\w+) isl=(\d+) c=(\d+) (.*)", line)
        if not m or " n=0 " in line:
            continue
        cfg, var, isl, c, rest = m.groups()
        if rest == "FAILED":  # no result json: the 90 min bench timeout hit
            rows[(cfg, int(isl), int(c))][var] = None
            continue
        kv = dict(x.split("=") for x in rest.split())
        rows[(cfg, int(isl), int(c))][var] = {k: float(v) for k, v in kv.items()}

ISL = {1024: "1k", 8192: "8k", 65536: "64k", 130048: "127k"}
for cfg in sorted({k[0] for k in rows}):
    keys = sorted(k for k in rows if k[0] == cfg)
    other = "ua" if any("ua" in rows[k] for k in keys) else "pr"
    print(f"\n### {cfg} (main vs {other})\n")
    print("| ISL/OSL | conc | out tok/s main | out tok/s " + other + " | speedup "
          "| median TPOT ms main | " + other + " | median TTFT ms main | " + other
          + " | accept len main | " + other + " |")
    print("|---|---|---|---|---|---|---|---|---|---|---|")
    for k in keys:
        a, b = rows[k].get("main", {}), rows[k].get(other, {})
        def g(x, key, fmt):
            if x is None:
                return ">90 min" if key == "out_tok_s" else "-"
            return format(x[key], fmt) if x else "-"
        sp = f"{b['out_tok_s'] / a['out_tok_s']:.2f}x" if a and b else "-"
        print(f"| {ISL[k[1]]}/1k | {k[2]} | {g(a, 'out_tok_s', '.1f')} | "
              f"{g(b, 'out_tok_s', '.1f')} | {sp} | {g(a, 'med_tpot_ms', '.1f')} | "
              f"{g(b, 'med_tpot_ms', '.1f')} | {g(a, 'med_ttft_ms', '.0f')} | "
              f"{g(b, 'med_ttft_ms', '.0f')} | {g(a, 'accept_len', '.2f')} | "
              f"{g(b, 'accept_len', '.2f')} |")
```
