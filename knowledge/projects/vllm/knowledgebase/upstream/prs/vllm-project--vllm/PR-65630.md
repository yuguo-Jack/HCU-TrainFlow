# [Perf][Quantization] Batch the Marlin MoE scale and zero-point conversion over experts

[Upstream PR](https://github.com/vllm-project/vllm/pull/65630) · state: open · merged: False

## Original PR description

## Overview

At load, Marlin converts MoE scales and zero points one expert at a time, with a numpy round trip on the host for each expert's zero points. I batch it over experts on the device.

## Claims

- A 4-layer Qwen3-30B-A3B shaped AWQ model converts its MoE weights in 0.11 to 0.27 s, not 1.14 to 1.51 s.
- Bit-identical output, load time only.

## Validation

RTX 3070 Ti, WSL. The real checkpoint exceeds 8 GiB, so I loaded a random 4-layer one with its shapes, old and new alternating. Converted tensor hashes, greedy tokens and logprobs match.

| Run | Per layer (s) | 4 layers (s) |
|---|---|---|
| old | 0.37, 0.36, 0.34, 0.45 | 1.51 |
| new | 0.14, 0.10, 0.014, 0.014 | 0.27 |
| old | 0.25, 0.27, 0.31, 0.31 | 1.14 |
| new | 0.032, 0.022, 0.018, 0.040 | 0.11 |

One w13, median of 5:

| Tensor | old | new |
|---|---|---|
| scales | 18.1 ms | 0.19 ms |
| AWQ zero points | 104.4 ms | 0.86 ms |
| compressed-tensors zero points | 91.2 ms | 0.73 ms |

## Details

All experts at once took 672 MiB extra on a DeepSeek-V3 w13 layer, so zero points go in 4M-element chunks (74 MiB peak, loops 31 MiB). No open PR changes these converters, and #64182's weight cache excludes MoE.

```python
"""Old per-expert loops (copied from main) against the batched functions.

Usage: python bench_postprocess.py [cpu|cuda] [repeats]
"""
import os
import statistics
import sys
import time

import torch

from vllm.model_executor.layers.quantization.utils import marlin_utils as mu
from vllm.model_executor.layers.quantization.utils.quant_utils import unpack_cols

dev = sys.argv[1] if len(sys.argv) > 1 else "cuda"
reps = int(sys.argv[2]) if len(sys.argv) > 2 else 5


def old_scales(s, size_k, size_n, group_size, is_a_8bit=False):
    out = torch.empty_like(s)
    for e in range(s.shape[0]):
        out[e] = mu.marlin_permute_scales(s[e], size_k, size_n, group_size, is_a_8bit)
    return out


def old_awq_zp(q, size_k, size_n, num_bits, is_a_8bit=False):
    out = torch.empty_like(q)
    for e in range(q.shape[0]):
        out[e] = mu.awq_to_marlin_zero_points(q[e], size_k, size_n, num_bits, is_a_8bit)
    return out


def old_packed_zp(q, size_k, size_n, num_bits, is_a_8bit=False):
    out = torch.empty_like(q)
    for e in range(q.shape[0]):
        zp = unpack_cols(q[e], num_bits, size_k, size_n)
        out[e] = mu.marlin_zero_points(zp, size_k, size_n, num_bits, is_a_8bit)
    return out


def timeit(f):
    f()
    if dev == "cuda":
        torch.cuda.synchronize()
    ts = []
    for _ in range(reps):
        t0 = time.perf_counter()
        f()
        if dev == "cuda":
            torch.cuda.synchronize()
        ts.append((time.perf_counter() - t0) * 1000)
    return statistics.median(ts), min(ts)


print(f"load {os.getloadavg()} device {dev} reps {reps} torch {torch.__version__}")
g = 128
cases = {
    "Qwen3-30B-A3B E128 tp1": (128, 2048, 768),
    "Qwen3-30B-A3B E128 tp2": (128, 2048, 384),
    "E256 K2048 N512 tp1": (256, 2048, 512),
}
for name, (E, K, N) in cases.items():
    for tag, sk, sn in (("w13", K, 2 * N), ("w2", N, K)):
        ng = sk // g
        s = torch.randn(E, ng, sn, device=dev).half()
        zp = torch.randint(-(2**31), 2**31 - 1, (E, ng, sn // 8), device=dev, dtype=torch.int32)
        rows = [
            ("scales", lambda: old_scales(s, sk, sn, g), lambda: mu.marlin_moe_permute_scales(s, sk, sn, g)),
            ("awq_zp", lambda: old_awq_zp(zp, ng, sn, 4), lambda: mu.moe_awq_to_marlin_zero_points(zp, ng, sn, 4)),
            ("ct_zp", lambda: old_packed_zp(zp, ng, sn, 4), lambda: mu.moe_packed_to_marlin_zero_points(zp, ng, sn, 4)),
        ]
        for what, fo, fn in rows:
            eq = torch.equal(fo(), fn())
            mo, lo = timeit(fo)
            mn, ln = timeit(fn)
            print(f"{name} {tag} {what}: old {mo:.1f} ms (min {lo:.1f}) | new {mn:.2f} ms (min {ln:.2f}) | equal {eq}")
```
