# [Bugfix] Prevent stale collective RPC response reuse

[Upstream PR](https://github.com/vllm-project/vllm/pull/57846) · state: open · merged: False

## Original PR description

## Purpose

A caller-caught control RPC error or timeout can leave untagged worker replies queued, so the next RPC returns stale results. Mark the executor failed and notify EngineCore when replies become unsafe; preserve recovery when all expected replies were consumed.

```python
from collections import deque
from types import SimpleNamespace

from vllm.v1.executor.multiproc_executor import MultiprocExecutor, WorkerProc


class BroadcastMQ:
    def __init__(self):
        self.sent = []

    def enqueue(self, item):
        self.sent.append(item)


class ResponseMQ:
    def __init__(self, *responses):
        self.responses = deque(responses)
        self.reads = 0

    def dequeue(self, timeout=None):
        self.reads += 1
        return self.responses.popleft()


success = WorkerProc.ResponseStatus.SUCCESS
failure = WorkerProc.ResponseStatus.FAILURE
executor = MultiprocExecutor.__new__(MultiprocExecutor)
executor.rpc_broadcast_mq = BroadcastMQ()
executor.response_mqs = [
    ResponseMQ((failure, "rank0 boom"), (success, "new-rank0")),
    ResponseMQ((success, "stale-rank1"), (success, "new-rank1")),
]
executor.futures_queue = deque()
executor.is_failed = False
executor.failure_callback = None
executor.vllm_config = SimpleNamespace(
    parallel_config=SimpleNamespace(enable_fault_tolerance=False)
)

try:
    executor.collective_rpc("first")
except RuntimeError as exc:
    print(f"first_error={exc}")

print(f"failed_after_first={executor.is_failed}")
print(f"reads_after_first={[q.reads for q in executor.response_mqs]}")

try:
    print(f"second_result={executor.collective_rpc('second')}")
except RuntimeError as exc:
    print(f"second_error={exc}")

print(f"reads_after_second={[q.reads for q in executor.response_mqs]}")
print(f"broadcast_count={len(executor.rpc_broadcast_mq.sent)}")
```

```python
import argparse

from vllm.engine.arg_utils import EngineArgs
from vllm.v1.executor.multiproc_executor import MultiprocExecutor


def fail_on_rank_zero(worker):
    if worker.global_rank == 0:
        raise RuntimeError("intentional rank-0 failure")
    return f"first-rank-{worker.global_rank}"


def identify_rank(worker):
    return f"second-rank-{worker.global_rank}"


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--expect", choices=("main", "branch"), required=True)
    args = parser.parse_args()

    config = EngineArgs(
        model="facebook/opt-125m",
        tensor_parallel_size=2,
        distributed_executor_backend="mp",
        max_model_len=256,
        gpu_memory_utilization=0.2,
        enforce_eager=True,
    ).create_engine_config()

    executor = MultiprocExecutor(config)
    try:
        print(
            "COLLECTIVE_RPC_SOURCE="
            f"{executor.collective_rpc.__func__.__code__.co_filename}",
            flush=True,
        )

        try:
            executor.collective_rpc(fail_on_rank_zero, timeout=30)
        except RuntimeError as error:
            first_error = str(error)
        else:
            raise AssertionError("the first RPC unexpectedly succeeded")

        print(f"FIRST_RPC_ERROR={first_error}", flush=True)
        print(f"EXECUTOR_FAILED_AFTER_FIRST={executor.is_failed}", flush=True)

        try:
            second_result = executor.collective_rpc(identify_rank, timeout=30)
        except RuntimeError as error:
            second_outcome = f"RuntimeError: {error}"
        else:
            second_outcome = repr(second_result)

        print(f"SECOND_RPC_OUTCOME={second_outcome}", flush=True)

        assert "intentional rank-0 failure" in first_error
        if args.expect == "main":
            assert not executor.is_failed
            assert second_outcome == "['second-rank-0', 'first-rank-1']"
        else:
            assert executor.is_failed
            assert second_outcome == "RuntimeError: Executor failed."

        print(f"TP2_REPRO_{args.expect.upper()}=PASS", flush=True)
    finally:
        executor.shutdown()

if __name__ == "__main__":
    main()
```
