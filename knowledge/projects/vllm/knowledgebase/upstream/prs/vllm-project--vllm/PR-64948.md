# [Spec Decode] Avoid redundant EAGLE3 auxiliary concatenation with fc_norm

[Upstream PR](https://github.com/vllm-project/vllm/pull/64948) · state: open · merged: False

## Original PR description



## Overview

Avoid the raw auxiliary-state concatenation before per-chunk `fc_norm` in the V2 Llama-family EAGLE3 path. Current code concatenates, splits for normalization, then concatenates again.

## Claims

- Removes one raw auxiliary allocation while preserving the incumbent norm and FC operations. Three BF16 states of width 2,560 across 32,768 rows account for 480 MiB.
- Preserves the Tensor API, promotion order, full-input normalization, default configurations and subclass overrides. Model eligibility resolves once at load.

## Validation

H100, Torch 2.14.0+cu130; public base `7b77ed581d92fa14e34ab43344837bb1ca1b4257`.

```bash
.venv/bin/python -m pytest tests/v1/cudagraph/test_cudagraph_manager.py -v
```

Base: 13 passed; both allocation regressions fail. Candidate: 15 passed; all 30 neighboring tests pass. Applicable pre-commit hooks pass. The regressions construct real heads, compare exact output/input bytes under C/native norm priorities and assert profiler-recorded concatenation allocation.

Trained validation uses `datalab-to/chandra-ocr-2@af93b47dba1b47b6640c86ccf487ed2260ab9a09` and `jbarrow/joecr-chandra-2-eagle3.1@fb5d00f141267198ca19e6bdcf890c367394cc10`. Eight fixed synthetic images contain 24 printed fields. Matched base/candidate token IDs and per-page draft/acceptance counters agree exactly in eager TP1, compiled/FULL TP1 and eager TP2; all 24 fields match. Each baseline also repeats exactly. This is a small synthetic OCR parity check. The trained checkpoint stays outside CI; the tiny real-module regression runs in the existing spec-decode job.

## Details

A cached optional combiner accepts the original list and normalizes eligible inputs before one concatenation. Other models/configurations and unsupported input layouts retain concatenation first. V1 remains unchanged. Related #43482/#50658/#64701 affect other paths; bounded current searches found no matching PR for this Llama/V2 AR allocation.

```python
"""Pinned trained fc_norm serving and drafter-sensitive OCR parity evidence."""

import argparse
import base64
import hashlib
import io
import json
import os
import re
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont
import torch
import vllm
from vllm import LLM, SamplingParams
from tests.v1.e2e.spec_decode.utils import compute_acceptance_rate
from eagle_aux_prompts import OCR_LAYOUT_PROMPT


def counters(metrics):
    return {
        m.name: getattr(m, "values", getattr(m, "value", None))
        for m in metrics
        if m.name.startswith("vllm:spec_decode_")
    }


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--mode", choices=["eager", "full"], required=True)
    parser.add_argument("--tp", type=int, default=1)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    if args.mode == "full":
        # The default breakable graph path deliberately disables torch.compile.
        os.environ["VLLM_USE_BREAKABLE_CUDAGRAPH"] = "0"
    config = dict(
        model="datalab-to/chandra-ocr-2",
        revision="af93b47dba1b47b6640c86ccf487ed2260ab9a09",
        dtype="bfloat16",
        max_model_len=4096,
        max_num_seqs=1,
        max_num_batched_tokens=2048,
        gpu_memory_utilization=0.6,
        enable_chunked_prefill=True,
        enable_prefix_caching=False,
        enforce_eager=args.mode == "eager",
        tensor_parallel_size=args.tp,
        async_scheduling=False,
        seed=0,
        disable_log_stats=False,
        limit_mm_per_prompt={"image": 1},
        compilation_config={
            "mode": 0 if args.mode == "eager" else 3,
            "cudagraph_mode": "FULL",
            "cudagraph_capture_sizes": [4, 8, 16, 32],
        },
        speculative_config=dict(
            method="eagle3",
            model="jbarrow/joecr-chandra-2-eagle3.1",
            revision="fb5d00f141267198ca19e6bdcf890c367394cc10",
            num_speculative_tokens=3,
        ),
    )
    print(
        "PROVENANCE", vllm.__file__, torch.__version__, json.dumps(config), flush=True
    )
    llm = LLM(**config)
    effective_config = llm.llm_engine.vllm_config
    effective = dict(
        compilation_mode=int(effective_config.compilation_config.mode),
        cudagraph_mode=str(effective_config.compilation_config.cudagraph_mode),
        async_scheduling=effective_config.scheduler_config.async_scheduling,
        breakable_cudagraph=os.environ.get("VLLM_USE_BREAKABLE_CUDAGRAPH"),
    )
    assert effective["compilation_mode"] == (0 if args.mode == "eager" else 3), (
        effective
    )
    print("EFFECTIVE", json.dumps(effective), flush=True)
    results = []
    for i in range(8):
        fields = [
            f"Invoice {12345 + i}",
            f"Date 2026-10-{20 + i:02d}",
            f"Total {42 + i}.50",
        ]
        image = Image.new("RGB", (256, 192), "white")
        draw = ImageDraw.Draw(image)
        for line, text in enumerate(fields):
            draw.text(
                (16, 16 + line * 30), text, fill="black", font=ImageFont.load_default()
            )
        image = image.resize((1024, 768), Image.Resampling.NEAREST)
        buffer = io.BytesIO()
        image.save(buffer, format="PNG")
        data = buffer.getvalue()
        messages = [
            {
                "role": "user",
                "content": [
                    {
                        "type": "image_url",
                        "image_url": {
                            "url": "data:image/png;base64,"
                            + base64.b64encode(data).decode()
                        },
                    },
                    {"type": "text", "text": OCR_LAYOUT_PROMPT},
                ],
            }
        ]
        outputs = llm.chat(
            messages,
            SamplingParams(temperature=0, max_tokens=512, seed=0),
            use_tqdm=False,
        )
        assert len(outputs) == 1
        request = outputs[0]
        output = request.outputs[0]
        text_content = re.sub("<[^>]*>", " ", output.text)
        results.append(
            dict(
                id=i,
                fields=fields,
                fields_found=[f in text_content for f in fields],
                image_sha256=hashlib.sha256(data).hexdigest(),
                text=output.text,
                token_ids=output.token_ids,
                finish_reason=output.finish_reason,
                finished=request.finished,
                counters=counters(llm.get_metrics()),
            )
        )
        print("PAGE", json.dumps(results[-1]), flush=True)
    summary = dict(
        config=config,
        source=vllm.__file__,
        torch=torch.__version__,
        cuda=torch.version.cuda,
        effective=effective,
        prompt_sha256=hashlib.sha256(OCR_LAYOUT_PROMPT.encode()).hexdigest(),
        pages=len(results),
        fields=sum(sum(r["fields_found"]) for r in results),
        reference_fields=24,
        empty=sum(not r["token_ids"] for r in results),
        incomplete=sum(not r["finished"] for r in results),
        length_limited=sum(r["finish_reason"] == "length" for r in results),
        acceptance_rate=compute_acceptance_rate(llm.get_metrics()),
        counters=counters(llm.get_metrics()),
    )
    (args.output / "summary.json").write_text(json.dumps(summary, indent=2) + "\n")
    (args.output / "outputs.jsonl").write_text(
        "".join(json.dumps(r) + "\n" for r in results)
    )
    print("RESULT", json.dumps(summary), flush=True)
    llm.llm_engine.engine_core.shutdown()
    assert summary["pages"] == 8 and summary["fields"] == 24
    assert summary["empty"] == summary["incomplete"] == summary["length_limited"] == 0
    assert summary["acceptance_rate"] > 0


if __name__ == "__main__":
    main()
```
