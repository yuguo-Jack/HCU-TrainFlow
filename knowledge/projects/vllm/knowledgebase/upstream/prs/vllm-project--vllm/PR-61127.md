# [Bugfix][CPU] Reject -1 placeholder drafts in the C++ rejection sampler

[Upstream PR](https://github.com/vllm-project/vllm/pull/61127) · state: open · merged: False

## Original PR description

<!-- markdownlint-disable -->

## Overview

The CPU C++ override of `rejection_random_sample_kernel` accepts `-1` placeholder drafts. Like the Triton kernel, it should reject them. Fixes #61117.

No other open PR changes this loop: #54952 edits `sample_recovered_tokens_impl` in the same file, but not the rejection loop.

## Claims

- On CPU builds without Triton, which use the C++ overrides, a padded (`-1`) draft is now always rejected. Before, the kernel read `target_probs[row * stride - 1]`, which is the previous row's last entry, or memory before the buffer for the first row. It could accept the placeholder and take the next token from a padding row.
- Builds with Triton, including the x86 CPU wheel with triton-cpu, and GPU are unchanged. They already use the Triton kernel, which has `if draft_token_id < 0: accepted = False`.

import argparse
import re
import tempfile

import torch
from transformers import AutoTokenizer, LlamaConfig, LlamaForCausalLM

import vllm.utils.cpu_triton_utils as cpu_tl
from vllm import LLM, SamplingParams
from vllm.sampling_params import StructuredOutputsParams

EOS, HIDDEN = 50256, 64
BIAS = {  # per-token logit bias of the toy target and draft
    "target": {"a": 15.0, "b": 14.0, "c": 13.5, "ab": 13.0, "bc": 12.5,
               "x": 12.0, "z": 12.0, "<eos>": 14.0},
    "draft": {"a": 13.0, "b": 15.0, "c": 12.0, "ab": 12.0, "bc": 14.0,
              "x": 14.5, "z": 13.0, "<eos>": 12.0},
}


def build(kind, tok, out):
    torch.manual_seed({"target": 1, "draft": 2}[kind])
    cfg = LlamaConfig(vocab_size=len(tok), hidden_size=HIDDEN, intermediate_size=128,
                      num_hidden_layers=2, num_attention_heads=1, num_key_value_heads=1,
                      max_position_embeddings=128, tie_word_embeddings=False,
                      bos_token_id=EOS, eos_token_id=EOS, torch_dtype="float32")
    model = LlamaForCausalLM(cfg).float()
    with torch.no_grad():
        model.model.embed_tokens.weight[:, 0] = 1.0
        head = torch.zeros_like(model.lm_head.weight)
        g = torch.Generator().manual_seed(100 + {"target": 1, "draft": 2}[kind])
        for s, b in BIAS[kind].items():
            tid = EOS if s == "<eos>" else tok.encode(s)[0]
            head[tid, 0] = b / HIDDEN**0.5
            head[tid, 1:] = 3.0 * torch.randn(HIDDEN - 1, generator=g) / HIDDEN**0.5
        model.lm_head.weight.copy_(head)
    model.save_pretrained(out)
    tok.save_pretrained(out)


# Workaround for #55005 (CPU expand shim never copies its output back).
def expand(output, input_val, cu_num_tokens, replace_from, replace_to,
           MAX_NUM_TOKENS=None):
    cu = cu_num_tokens.to(torch.int64)
    vals = torch.where(input_val == replace_from,
                       torch.as_tensor(replace_to, dtype=input_val.dtype), input_val)
    out = torch.repeat_interleave(vals, torch.diff(cu, prepend=cu.new_zeros(1)))
    output[: out.numel()].copy_(out.to(output.dtype))


# Workaround for #61131 (slot-mapping fallback signature after #57169).
def slot_mapping(num_tokens, max_num_tokens, query_start_loc_ptr, positions_ptr,
                 block_table_ptr, block_table_stride, slot_mapping_ptr,
                 KV_CACHE_BLOCK_SIZE, TOTAL_CP_WORLD_SIZE, TOTAL_CP_RANK,
                 CP_KV_CACHE_INTERLEAVE_SIZE, PAD_ID, BLOCK_SIZE):
    torch.ops._C.compute_slot_mapping_kernel_impl(
        query_start_loc_ptr, positions_ptr, block_table_ptr, slot_mapping_ptr,
        KV_CACHE_BLOCK_SIZE)


# Count -1 placeholder drafts that the C++ rejection kernel accepts.
accepted_placeholders = 0
orig_rrs = cpu_tl._rejection_random_sample_kernel_impl


def counted_rrs(out, cu, draft_ids, *args, **kwargs):
    global accepted_placeholders
    res = orig_rrs(out, cu, draft_ids, *args, **kwargs)
    ends, d, o = cu.tolist(), draft_ids.tolist(), out.tolist()
    for r, end in enumerate(ends):
        start = ends[r - 1] if r else 0
        for pos, did in enumerate(d[start:end]):
            if did != o[r][pos]:
                break
            accepted_placeholders += did == -1
    return res


counted_rrs.__signature__ = __import__("inspect").signature(orig_rrs)
cpu_tl._expand_kernel_impl = expand
cpu_tl._compute_slot_mapping_kernel_impl = slot_mapping
cpu_tl._rejection_random_sample_kernel_impl = counted_rrs

ap = argparse.ArgumentParser()
ap.add_argument("--n", type=int, default=600)
ap.add_argument("--draft-sample-method", default="probabilistic")
a = ap.parse_args()
root = tempfile.mkdtemp()
tok = AutoTokenizer.from_pretrained("openai-community/gpt2")
for kind in ("target", "draft"):
    build(kind, tok, f"{root}/{kind}")
llm = LLM(f"{root}/target", dtype="float32", seed=0, max_model_len=64,
          enforce_eager=True, max_num_seqs=256, gpu_memory_utilization=0.2,
          speculative_config={"method": "draft_model", "model": f"{root}/draft",
                              "num_speculative_tokens": 2,
                              "draft_sample_method": a.draft_sample_method})
regex = "[abc]{3}"
sps = [SamplingParams(temperature=1.0, max_tokens=8, seed=i,
                      structured_outputs=StructuredOutputsParams(regex=regex))
       for i in range(a.n)]
# One generate() call: requests that start while others decode get -1 drafts.
outs = llm.generate([{"prompt_token_ids": [EOS]}] * a.n, sps, use_tqdm=False)
bad = sum(o.outputs[0].finish_reason == "error"
          or not re.fullmatch(regex, o.outputs[0].text) for o in outs)
print(f"RESULT n={a.n} accepted_placeholders={accepted_placeholders} "
      f"grammar_invalid_or_error={bad}")

## Details

**Root cause.** The scheduler pads new requests' first-step drafts with `-1`. With structured output, only the first row of such a request is grammar-masked; the rows after the `-1` are not. `rejection_random_sample_kernel_impl` used the draft id as an index without checking it, so a placeholder could be accepted, and the following token would then be sampled from an unmasked row. The greedy C++ kernel compares ids and is not affected. Neither is `sample_recovered_tokens`, which only uses the id for an equality check.

**Fix.** The C++ loop now skips the probability lookup for `draft_id < 0` and rejects. That is the same rule as the Triton kernel.

**MRV1 note.** The C++ overrides are only registered when Triton is unavailable, and without Triton the V1 model runner is used. So this path is MRV1-only, but it is the default for CPU builds without Triton, such as macOS. MRV2 has its own sampler, and #54442 already makes it fail closed on placeholder rows.
