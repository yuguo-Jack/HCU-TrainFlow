# [Bugfix] Keep legacy WNA16 MoE dispatch on Triton under batch invariance

[Upstream PR](https://github.com/vllm-project/vllm/pull/61038) · state: open · merged: False

## Original PR description



## Purpose

`should_moe_wna16_use_cuda` selects the CUDA WNA16 MoE kernel (`moe_wna16_gemm`) when there are at most six tokens per expert and the Triton `fused_moe_kernel_gptq_awq` otherwise. For the legacy `fused_experts` path with `use_int4_w4a16` under `VLLM_BATCH_INVARIANT=1` this means a request's experts are computed by the CUDA kernel when it is processed alone and by the Triton kernel when it is processed in a large enough batch, which defeats batch invariance. Two more things show up at the kernel level: the CUDA kernel is not deterministic run to run (same inputs, different output), and with the fixed batch-invariant config (`BLOCK_SIZE_K=32`) it rejects group sizes above 32 (`BLOCK_SIZE_K must divisible by group_size`). The Triton kernel with the fixed config gave bitwise identical rows for every batch size tried (M = 1, 2, 8, 32, 128, 300).

This PR makes `should_moe_wna16_use_cuda` return `False` in batch-invariant mode, so the Triton kernel with the fixed config is always used. Nothing changes when the flag is unset.

Note: the modular MoE path (`TritonWNA16Experts`, used by the compressed-tensors and GPTQ/AWQ MoE methods when serving) already calls the Triton kernel directly, so this only affects direct `fused_experts` users and the `get_default_config` path; it is a hardening of the public API rather than a serving fix.

```python
"""Kernel-level batch-invariance check of the W4A16 Triton/CUDA MoE path.

Builds a random int4 (group 128, symmetric) MoE, runs fused_moe on a batch of
M rows and on row 0 alone, and compares row 0 bitwise. Runs with the CUDA
small-batch kernel allowed (main behaviour) and with it disabled (the fix).
"""
import os, sys
os.environ.setdefault("VLLM_BATCH_INVARIANT", "1")
import torch
import vllm.envs as envs
import vllm.model_executor.layers.fused_moe.fused_moe as fm
from vllm.config import VllmConfig, set_current_vllm_config
from tests.kernels.moe.utils import fused_moe
from vllm.model_executor.layers.fused_moe.config import int4_w4a16_moe_quant_config
from vllm.model_executor.layers.quantization.utils.quant_utils import quantize_weights
from vllm.scalar_type import scalar_types

torch.manual_seed(0)
dev = "cuda"
dtype = torch.bfloat16
e, n, k, topk = 128, 768, 2048, 8   # Qwen3-30B-A3B-like shapes
group_size = int(sys.argv[1]) if len(sys.argv) > 1 else 32
print("group_size", group_size)
print("VLLM_BATCH_INVARIANT =", envs.VLLM_BATCH_INVARIANT, "| should_moe_wna16_use_cuda(8*topk...) =", fm.should_moe_wna16_use_cuda(8 * topk, group_size, e, 4))

w1 = torch.randn((e, 2 * n, k), device=dev, dtype=dtype) / 10
w2 = torch.randn((e, k, n), device=dev, dtype=dtype) / 10
w1_q = torch.empty((e, 2 * n, k // 2), device=dev, dtype=torch.uint8)
w2_q = torch.empty((e, k, n // 2), device=dev, dtype=torch.uint8)
w1_s = torch.empty((e, 2 * n, k // group_size), device=dev, dtype=dtype)
w2_s = torch.empty((e, k, n // group_size), device=dev, dtype=dtype)
for i in range(e):
    for w, wq, ws in ((w1, w1_q, w1_s), (w2, w2_q, w2_s)):
        _, qweight, scales, _ = quantize_weights(w[i].T, scalar_types.uint4b8, group_size, False, False)
        qweight = qweight.T.contiguous().to(torch.uint8)
        wq[i] = qweight[:, 1::2] * 16 + qweight[:, ::2]
        ws[i] = scales.T
qc = int4_w4a16_moe_quant_config(w1_scale=w1_s, w2_scale=w2_s, w1_zp=None, w2_zp=None, block_shape=[0, group_size])

def run(a, score):
    with set_current_vllm_config(VllmConfig()):
        return fused_moe(a, w1_q, w2_q, score, topk, renormalize=False, global_num_experts=e, quant_config=qc)

Ms = [1, 2, 8, 32, 128, 300]
a_all = torch.randn((max(Ms), k), device=dev, dtype=dtype) / 10
s_all = torch.randn((max(Ms), e), device=dev, dtype=dtype)
for allow_cuda in (True, False):
    orig = fm.should_moe_wna16_use_cuda
    if not allow_cuda:
        fm.should_moe_wna16_use_cuda = lambda *a_, **k_: False
    res = []
    try:
        ref = run(a_all[:1].clone(), s_all[:1].clone())[0]
        again = run(a_all[:1].clone(), s_all[:1].clone())[0]
        res.append(f"M=1 repeated:{'same' if torch.equal(again, ref) else 'DIFF(maxabs=%.3g)' % (again.float() - ref.float()).abs().max().item()}")
        for M in Ms:
            out = run(a_all[:M].clone(), s_all[:M].clone())[0]
            res.append(f"M={M}:{'same' if torch.equal(out, ref) else 'DIFF(maxabs=%.3g)' % (out.float() - ref.float()).abs().max().item()}")
    except Exception as exc:  # noqa: BLE001
        res.append(f"ERROR {type(exc).__name__}: {str(exc).splitlines()[0][:160]}")
    print(f"cuda_small_batch_kernel_allowed={allow_cuda}: row0 vs M=1 ->", ", ".join(res))
    fm.should_moe_wna16_use_cuda = orig
```
