# [Bugfix][Model] DeepSeek-V4.1: keep load permutations and Engram peer handles out of the sleep weights pool

[Upstream PR](https://github.com/vllm-project/vllm/pull/64922) · state: open · merged: False

## Original PR description

## Overview

Level-2 sleep discards GPU tensors in the sleep-mode weights pool that are not parameters or buffers. Two DeepSeek-V4.1 objects held such tensors:

- `permute_on_load` (mega-attention fused layouts): the weight-loader closure held the permutation as a GPU tensor. After level-2 sleep, reload permuted `wq_b` / `wo_a` with garbage. The loader now keeps its own host copy.
- `EngramPeerExchange`: `symm_mem.rendezvous` allocates the handles' device pointer tables, which `barrier()` reads, from the current pool. Model init runs in the weights pool, so level-2 sleep zeroed them, and the first forward after wake-up hit an illegal memory access. `rendezvous` now runs in a private `torch.cuda.MemPool`.

Split from #62365 by @aoshen02.

## Changes

- `vllm/models/deepseek_v41/common/ops/fused_layout.py`: `permute_on_load` copies `perm` to the host once (`perm.detach().to("cpu", copy=True)`).
- `vllm/models/deepseek_v41/nvidia/ops/engram_peer.py`: both `rendezvous` calls run under `torch.cuda.use_mem_pool(self._handle_pool)`.
- `tests/kernels/test_dsv41_mega_attn_layouts.py`: `test_permute_on_load_owns_its_permutation` (CPU) zeroes the caller's `perm` after registering the loader, then loads.

```python
"""Level-2 sleep with the real CuMemAllocator for vllm DSv4.1 sleep-pool fixes.

MODE=permute: build a mega-attention load permutation the way model init does
(on the GPU, inside the "weights" pool), sleep level 2, wake, then run the
weight loader and compare with the expected permuted checkpoint.
MODE=engram: build a real EngramPeerExchange (1-rank Engram DP group, stub
Engram tables) inside the "weights" pool, sleep level 2, wake, then call both
handles' barrier() and run exchange(). Run with PYTHONPATH=<tree>.
"""
import os, sys
from types import SimpleNamespace
import torch
from vllm.device_allocator.cumem import CuMemAllocator

mode = os.environ["MODE"]
alloc = CuMemAllocator.get_instance()

if mode == "permute":
    torch.cuda.set_device(0)
    from vllm.model_executor.model_loader.weight_utils import default_weight_loader
    from vllm.models.deepseek_v41.common.ops.fused_layout import permute_on_load, q_fused_permutation
    with alloc.use_memory_pool("weights"), torch.device("cuda"):
        perm = q_fused_permutation(64, 512)  # 64 local heads, head_dim 512
        param = torch.nn.Parameter(torch.empty(perm.numel(), 128, dtype=torch.bfloat16), requires_grad=False)
        param.weight_loader = default_weight_loader
        permute_on_load(param, perm, 0)
    ckpt = torch.randn(perm.numel(), 128, dtype=torch.bfloat16)
    expected = ckpt[q_fused_permutation(64, 512)]
    param.weight_loader(param, ckpt.cuda())
    print("cold load matches:", torch.equal(param.data.cpu(), expected), flush=True)
    alloc.sleep(offload_tags=())
    alloc.wake_up(None)
    param.weight_loader(param, ckpt.cuda())  # reload after level-2 sleep
    torch.cuda.synchronize()
    ok = torch.equal(param.data.cpu(), expected)
    print("RESULT reload after level-2 sleep matches cold load:", ok)
    sys.exit(0 if ok else 1)

import torch.distributed as dist
rank = int(os.environ.get("RANK", "0")); world = int(os.environ.get("WORLD_SIZE", "1"))
torch.cuda.set_device(int(os.environ.get("LOCAL_RANK", "0")))
if world == 1:
    dist.init_process_group("nccl", init_method="tcp://127.0.0.1:%s" % os.environ.get("PORT", "29631"),
                            world_size=1, rank=0, device_id=torch.device("cuda", 0))
else:
    dist.init_process_group("nccl", device_id=torch.device("cuda", torch.cuda.current_device()))
import vllm.models.deepseek_v41.nvidia.ops.engram_peer as ep
ep.get_engram_dp_group = lambda: SimpleNamespace(world_size=world, rank_in_group=rank, device_group=dist.group.WORLD)
heads, dim, vocab = 8, 256, 1024
with alloc.use_memory_pool("weights"), torch.device("cuda"):
    weight = torch.zeros(vocab, heads, dim, dtype=torch.uint8)
    scale = torch.zeros(vocab, heads, dim // 32, dtype=torch.uint8)
    table = SimpleNamespace(n_hash_cols=heads, dim=dim, vocab_start_idx=0, vocab_end_idx=vocab,
                            _storage=lambda: (weight, scale))
    layer = SimpleNamespace(staged_rows=torch.empty(1), layer_hash_index=0, embed_tokens=table)
    exch = ep.EngramPeerExchange([layer], max_tokens=16)
torch.cuda.synchronize()
exch.id_handle.barrier(); exch.row_handle.barrier(); torch.cuda.synchronize()
print(f"rank{rank} barrier before sleep OK", flush=True)
alloc.sleep(offload_tags=())
alloc.wake_up(None)
torch.cuda.synchronize()
exch.id_handle.barrier(); exch.row_handle.barrier()
torch.cuda.synchronize()
print(f"RESULT rank{rank} barrier after level-2 sleep OK", flush=True)
dist.destroy_process_group()
```
