# [Bugfix][Spec Decode] Include draft models in DeepGEMM warmup

[Upstream PR](https://github.com/vllm-project/vllm/pull/60995) · state: open · merged: False

## Original PR description

## Overview

DeepGEMM startup warmup now includes the draft model, covering eligible draft-only layer shapes before speculative serving starts.

Related: #60929

## Claims

- An available draft model uses the same warmup token limit as the target model.
- Absent drafts, unsupported DeepGEMM platforms and `VLLM_DEEP_GEMM_WARMUP=skip` retain their existing behavior.
- No latency, memory, model-quality or original-workload OOM improvement is claimed.

These checkpoints have random FP8 decoder weights and a zero LM head to make acceptance deterministic. This checks actual loading, backend selection, draft warmup and speculative generation, **not model quality or realistic acceptance**. The original DeepSeek-V4-Pro + DSpark / 2x GB300 / TP2+EP2 / memory-utilization 0.99 OOM scenario was not reproduced; only the local GPU was authorized.
```python
"""Real CUDA DeepGEMM regression probe; orchestration stubs are listed below."""
import hashlib
import json
import os
import sys
import time
from pathlib import Path
from types import SimpleNamespace
from unittest.mock import Mock
import torch
import vllm
from vllm.model_executor.kernels.linear.scaled_mm.deep_gemm import DeepGemmFp8BlockScaledMMKernel
from vllm.model_executor.warmup import deep_gemm_warmup as dg, kernel_warmup as kw
from vllm.utils.deep_gemm import fp8_gemm_nt, is_deep_gemm_supported, calc_diff
phase = sys.argv[1]
result_path = Path(sys.argv[2])
torch.manual_seed(60995)
assert is_deep_gemm_supported()
def layer(n, k):
    m = torch.nn.Module()
    m.register_buffer('weight', torch.randn(n, k, device='cuda').to(torch.float8_e4m3fn))
    m.register_buffer('weight_scale', torch.ones(n // 128, k // 128, device='cuda'))
    m.weight_block_size = [128, 128]
    # Synthetic provider registration; actual fp8_gemm_nt and warmup are unmodified.
    m.deep_gemm_warmup_provider = object.__new__(DeepGemmFp8BlockScaledMMKernel)
    return m
target, draft = layer(512, 1024), layer(768, 1024)
worker = Mock(use_v2_model_runner=True)
worker.get_model.return_value = target
worker.get_draft_model.return_value = draft
worker.scheduler_config.max_num_batched_tokens = 64
worker.vllm_config.kernel_config.enable_jit_warmup = False
worker.vllm_config.kernel_config.enable_cutedsl_warmup = False
worker.vllm_config.kernel_config.enable_flashinfer_autotune = False
worker.vllm_config.compilation_config.cudagraph_capture_sizes = []
worker.model_runner.attn_groups = []
stubs = ['qwen_triton_warmup', 'qwen_vl_triton_warmup', 'mamba_triton_warmup',
         '_warmup_gemm_rs_ar', '_warmup_ll_bf16_router_gemm',
         'flashinfer_sparse_mla_decode_autotune_warmup',
         'deepseek_v4_sparse_mla_attention_warmup', 'b12x_warmup',
         'minimax_m3_msa_warmup']
for name in stubs:
    setattr(kw, name, Mock())
dg.is_global_first_rank = lambda: False
cache = Path(os.environ['JIT_CACHE_DIR'])
def artifacts():
    return sorted(str(p.relative_to(cache)) for p in cache.rglob('*') if p.is_file())
before = artifacts()
print('RUN real CUDA warmup', phase, flush=True)
start = time.monotonic()
kw.kernel_warmup(worker)
torch.cuda.synchronize()
warm_seconds = time.monotonic() - start
warm_shapes = sorted(list(s) for s in dg.FP8_GEMM_NT_WARMUP_CACHE)
after_warm = artifacts()
assert [512, 1024] in warm_shapes
assert ([768, 1024] in warm_shapes) == (phase == 'patched')
cases = []
for model_name, model in [('target', target), ('draft', draft)]:
    for tokens in [1, 7, 32, 57, 64]:
        a = torch.randn(tokens, 1024, device='cuda').to(torch.float8_e4m3fn)
        scales = torch.ones(tokens, 8, device='cuda')
        out = torch.empty(tokens, model.weight.shape[0], device='cuda', dtype=torch.bfloat16)
        reference = a.float() @ model.weight.float().T
        prior = artifacts()
        fp8_gemm_nt((a, scales), (model.weight, model.weight_scale), out)
        torch.cuda.synchronize()
        diff = float(calc_diff(out, reference))
        assert torch.isfinite(out).all()
        assert diff < 0.001, diff
        added = sorted(set(artifacts()) - set(prior))
        if phase == 'patched':
            assert not added, added
        cases.append(dict(model=model_name, m=tokens, diff=diff, new_jit_files=added))
        print('PASS', model_name, tokens, diff, 'new JIT files', len(added), flush=True)
final_artifacts = artifacts()
if phase == 'baseline':
    assert any(c['new_jit_files'] for c in cases if c['model'] == 'draft')
else:
    prior = artifacts()
    kw.kernel_warmup(worker)
    torch.cuda.synchronize()
    assert artifacts() == prior
```

## Details

The startup dispatcher previously passed only `worker.get_model()` to DeepGEMM warmup. It now uses the existing `Worker.get_draft_model()` API, which returns `None` when no draft model is available and supports both model runners. The existing provider filtering and compilation cache remain responsible for eligible layers and repeated shapes. Startup work may increase for draft shapes not already covered by the target.

Duplicate checks included the issue comments, issue-number PR search and DeepGEMM/warmup/draft/speculator searches. No matching open PR was found. Related PR #57258 changes warmup ordering relative to memory profiling; #60468 adds MegaMoE warmup. Neither supplies this missing draft-model call.
