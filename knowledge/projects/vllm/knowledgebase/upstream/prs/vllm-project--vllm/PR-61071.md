# [Bugfix] Apply LM-head bias in native Moondream3 and PhiMoE models

[Upstream PR](https://github.com/vllm-project/vllm/pull/61071) · state: open · merged: False

## Original PR description

## Overview

Apply the loaded LM-head bias when computing Moondream3 and PhiMoE logits.

## Claims

- Preserve the checkpoint's output projection, including its bias.
- Reuse the logits processor's existing bias argument; no new implementation or inference-time branching.

## Validation

Moondream3: two real images, official HF/main/candidate/revert. Greedy answers are unchanged; common top-5 logprob mean absolute error against HF changes from 0.0904 to 0.0680 and from 0.2080 to 0.1142. Revert exactly reproduces main. This is probability-correction evidence, not benchmark accuracy.

Phi-3.5-MoE: four one-token prompts completed on HF and on main/candidate/revert. All greedy token IDs agree. The bias changes probabilities; the HF comparison does not show a consistent reduction in error. On actual inference hidden states, the maximum absolute difference from `F.linear(hidden, weight, bias)` is 0.125 on main and 0 on the candidate, covering all four prompts. Main/revert selected tokens agree and their maximum logprob difference is 4.76837158203125e-07.

```python
import argparse
import json
from pathlib import Path

import torch
from PIL import Image

BASE = Path('.')
MODEL = './model'
CASES = [('stop_sign', 'What is this sign?'), ('cherry_blossom', 'What season is shown?')]


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--backend', choices=['hf', 'vllm'], required=True)
    parser.add_argument('--output', required=True)
    args = parser.parse_args()
    results = []
    if args.backend == 'hf':
        from transformers import AutoModelForCausalLM
        model = AutoModelForCausalLM.from_pretrained(
            MODEL, trust_remote_code=True, dtype=torch.bfloat16,
        ).eval().cuda()
        logits = []
        def capture(module, inputs, output):
            logits.append(output.detach().reshape(-1, output.shape[-1])[-1].float().cpu())
        handle = model.model.text.lm_head.register_forward_hook(capture)
        for name, question in CASES:
            logits.clear()
            with torch.inference_mode():
                result = model.query(Image.open(BASE / 'inputs' / f'{name}.jpg').convert('RGB'), question,
                                     reasoning=False, settings={'max_tokens': 8, 'temperature': 0})
            assert logits, 'reference did not call its LM head'
            values, indices = logits[0].log_softmax(-1).topk(5)
            results.append({'image': name, 'text': result['answer'],
                            'top5': dict(zip(map(str, indices.tolist()), values.tolist()))})
        handle.remove()
    else:
        from vllm import LLM, SamplingParams
        model = LLM(model=MODEL, tokenizer='moondream/starmie-v1', trust_remote_code=True, dtype='bfloat16',
                    enforce_eager=True, max_model_len=2048, max_num_seqs=1,
                    gpu_memory_utilization=0.9)
        for name, question in CASES:
            prompt = '<|endoftext|><image><|md_reserved_0|>query<|md_reserved_1|>' + question + '<|md_reserved_2|>'
            output = model.generate({'prompt': prompt, 'multi_modal_data': {
                'image': Image.open(BASE / 'inputs' / f'{name}.jpg').convert('RGB')}},
                SamplingParams(temperature=0, max_tokens=8, logprobs=5), use_tqdm=False)[0].outputs[0]
            results.append({'image': name, 'text': output.text, 'token_ids': list(output.token_ids),
                            'top5': {str(k): v.logprob for k, v in output.logprobs[0].items()}})
    Path(args.output).write_text(json.dumps(results, indent=2))
    print(json.dumps(results), flush=True)
    print('MOONDREAM_PROBE_COMPLETED', flush=True)


if __name__ == '__main__':
    main()
```

```python
import argparse,json,pathlib,os
import torch
MODEL='./model'
PROMPTS=['The capital of France is','The result of two plus two is','Water freezes at','A triangle has']
def install_projection_oracle(model):
 def check(module, inputs, output):
  head, hidden = inputs[:2]
  reference=torch.nn.functional.linear(hidden,head.weight.to(hidden.device),head.bias.to(hidden.device))[:,:output.shape[-1]]
  difference=(output-reference).abs().max().item()
  supplied=len(inputs)>2 and inputs[2] is not None
  if supplied:
   torch.testing.assert_close(output,reference,rtol=0,atol=0)
  else:
   assert difference>0, 'missing bias must be observable'
  print('PROJECTION_ORACLE',json.dumps({'bias_supplied':supplied,'rows':hidden.shape[0],'max_abs_error':difference}),flush=True)
 model.logits_processor.register_forward_hook(check)
 return 'PROJECTION_ORACLE_INSTALLED'
def main():
 p=argparse.ArgumentParser();p.add_argument('--backend',choices=['hf','vllm'],required=True);p.add_argument('--output',required=True);a=p.parse_args()
 from transformers import AutoTokenizer
 tokenizer=AutoTokenizer.from_pretrained(MODEL,trust_remote_code=True)
 tokens=[tokenizer.encode(x) for x in PROMPTS]
 if a.backend=='hf':
  from transformers import AutoModelForCausalLM
  model=AutoModelForCausalLM.from_pretrained(MODEL,trust_remote_code=False,dtype=torch.bfloat16,device_map='auto',max_memory={0:'26GiB',1:'26GiB','cpu':'110GiB'},attn_implementation='eager').eval()
  results=[]
  for ids in tokens:
   with torch.inference_mode():logits=model(input_ids=torch.tensor([ids],device=model.get_input_embeddings().weight.device),use_cache=False).logits[0,-1].float()
   logp=logits.log_softmax(-1);vals,idx=logp.topk(5)
   results.append({'token_ids':[int(idx[0])],'top5':dict(zip(map(str,idx.tolist()),vals.tolist()))})
 else:
  from vllm import LLM,SamplingParams
  model=LLM(model=MODEL,trust_remote_code=True,dtype='bfloat16',tensor_parallel_size=1,cpu_offload_gb=64,enforce_eager=True,max_model_len=256,gpu_memory_utilization=0.8,max_num_seqs=4)
  if os.environ.get('PHI_PROJECTION_ORACLE')=='1':
   print(model.apply_model(install_projection_oracle),flush=True)
  outputs=model.generate([{'prompt_token_ids':x} for x in tokens],SamplingParams(temperature=0,max_tokens=1,logprobs=5),use_tqdm=False)
  results=[{'token_ids':r.outputs[0].token_ids,'top5':{str(k):v.logprob for k,v in r.outputs[0].logprobs[0].items()}} for r in outputs]
 result={'backend':a.backend,'prompts':PROMPTS,'prompt_token_ids':tokens,'results':results}
 pathlib.Path(a.output).write_text(json.dumps(result,indent=2));print(json.dumps(result),flush=True)
if __name__=='__main__':main()
```

## Details

Both native models create and load a biased LM head but omit the bias argument at the logits processor call. Other biased model implementations already pass it there. Keep the fix at these two callers.

The change is limited to the native Moondream3 and PhiMoE implementations. It does not change the Transformers backend. Full upstream GPU CI, tensor-parallel projection validation, quantized heads, and accuracy/performance benchmarks were not run.
