# [Perf][DSpark] Stack DeepSeek V41 context WKV projections

[Upstream PR](https://github.com/vllm-project/vllm/pull/63063) · state: closed · merged: True

## Original PR description



## Overview

Replace DeepSeek-V4.1 DSpark’s per-layer context Q+KV projections with one stacked KV-only projection. The existing context path computes Q values that are immediately discarde

refer to https://github.com/vllm-project/vllm/pull/54674

## Claims

- Compute context KV for all draft layers with one MergedColumnParallelLinear.
- Reuse the existing V4 weight-duplication helper to load each layer’s WKV weights and scales into the stacked projection.

# SPDX-License-Identifier: Apache-2.0
# SPDX-FileCopyrightText: Copyright contributors to the vLLM project
"""Compare V4.1 DSpark context projection, normalization and cache insertion.

Uses checkpoint WQA/WKV/RMSNorm weights and synthetic BF16 hidden states.
The baseline source must expose the original DSparkDeepseekV4Model method.
Humming uses its production heuristics in both arms; no tuning overrides.
"""

import argparse
import hashlib
import importlib.util
import json
import statistics
from functools import partial
from pathlib import Path
from types import SimpleNamespace
from unittest.mock import patch

import torch
from flashinfer.testing import bench_gpu_time_with_cupti
from safetensors import safe_open

from vllm.config import VllmConfig, set_current_vllm_config
from vllm.config.kernel import IrOpPriorityConfig
from vllm.model_executor.kernels.linear.mxfp8.humming import HummingMxfp8LinearKernel
from vllm.model_executor.kernels.linear.mxfp8.Mxfp8LinearKernel import (
    Mxfp8LinearLayerConfig,
)
from vllm.model_executor.layers.layernorm import RMSNorm
from vllm.models.deepseek_v41.nvidia import dspark

K, Q, V, LAYERS = 5120, 1280, 512, 3


class Projection(torch.nn.Module):
    def __init__(self, weight, scale, parts):
        super().__init__()
        self.input_size = K
        self.output_partition_sizes = parts
        self.params_dtype = torch.bfloat16
        self.has_bias = False
        self.weight = torch.nn.Parameter(weight.cuda(), requires_grad=False)
        self.weight_scale = torch.nn.Parameter(
            scale.view(torch.uint8)
            .repeat_interleave(32, dim=0)
            .cuda()
            .view(torch.float8_e8m0fnu),
            requires_grad=False,
        )
        self.kernel = HummingMxfp8LinearKernel(
            Mxfp8LinearLayerConfig(weight_shape=tuple(weight.shape))
        )
        self.kernel.process_weights_after_loading(self)

    def forward(self, x):
        return self.kernel.apply_weights(self, x)


def benchmark(args):
    torch.manual_seed(42)
    index = json.loads((args.checkpoint / "model.safetensors.index.json").read_text())[
        "weight_map"
    ]

    def read(name):
        with safe_open(args.checkpoint / index[name], framework="pt") as f:
            return f.get_tensor(name)

    old = []
    kv_weights, kv_scales, norms = [], [], []
    for i in range(LAYERS):
        prefix = f"mtp.{i}.attn."
        qw, qs = read(prefix + "wq_a.weight"), read(prefix + "wq_a.scale")
        vw, vs = read(prefix + "wkv.weight"), read(prefix + "wkv.scale")
        old.append(Projection(torch.cat([qw, vw]), torch.cat([qs, vs]), [Q, V]))
        kv_weights.append(vw)
        kv_scales.append(vs)
        norm = RMSNorm(V, eps=1e-20, dtype=torch.bfloat16).cuda()
        norm.weight.data.copy_(read(prefix + "kv_norm.weight"))
        norms.append(norm)
    merged = Projection(torch.cat(kv_weights), torch.cat(kv_scales), [V] * LAYERS)
    spec = importlib.util.spec_from_file_location(
        "vllm.models.deepseek_v41.nvidia.dspark_baseline", args.baseline_source
    )
    baseline = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(baseline)
    report = {
        "gpu": torch.cuda.get_device_name(),
        "torch": torch.__version__,
        "checkpoint": str(args.checkpoint),
        "source_sha256": {
            "A": hashlib.sha256(args.baseline_source.read_bytes()).hexdigest(),
            "B": hashlib.sha256(Path(dspark.__file__).read_bytes()).hexdigest(),
        },
        "boundary": "3-layer context projection + RMSNorm + RoPE/quant/cache insert",
        "timing": "CUPTI, CUDA graph, cold L2, ABBA, median of round medians",
        "input": "Synthetic BF16 hidden states; actual checkpoint weights/scales",
        "compute_config": merged.kernel.compute_config,
        "tuning": "Both arms warmed; unchanged production Humming heuristics",
        "correctness": "Normalized KV atol=0.02, rtol=0.02; cache bytes may differ",
        "rows": [],
    }
    for m in args.tokens:
        x = torch.randn(m, K, device="cuda", dtype=torch.bfloat16)
        positions = torch.arange(m, device="cuda")
        rope = torch.zeros(m, 64, device="cuda")
        rope[:, :32] = 1
        models = []
        for _ in range(2):
            layers = []
            for i in range(LAYERS):
                attn = SimpleNamespace(
                    fused_wqa_wkv=lambda x, i=i: (old[i](x), None),
                    q_lora_rank=Q,
                    kv_norm=norms[i],
                    kv_mxfp8=True,
                    rotary_emb=SimpleNamespace(cos_sin_cache=rope),
                    swa_cache_layer=SimpleNamespace(
                        block_size=64,
                        kv_cache=torch.zeros(
                            (m + 63) // 64, 64, 528, device="cuda", dtype=torch.uint8
                        ),
                    ),
                )
                layers.append(SimpleNamespace(attn=attn))
            models.append(
                SimpleNamespace(
                    layers=layers,
                    context_wkv_proj=merged,
                    num_dspark_layers=LAYERS,
                    config=SimpleNamespace(head_dim=V),
                )
            )
        modules = [baseline, dspark]
        functions = [
            partial(
                module.DSparkDeepseekV4Model.precompute_and_store_context_kv,
                model,
                x,
                positions,
                [positions] * LAYERS,
            )
            for module, model in zip(modules, models)
        ]
        for _ in range(20):
            for function in functions:
                function()
        torch.accelerator.synchronize()

        normalized = []
        for module, function in zip(modules, functions):
            outputs = []
            with patch.object(
                module,
                "_insert_context_kv",
                lambda attn, kv, p, s, out=outputs: out.append(kv.clone()),
            ):
                function()
            normalized.append(torch.stack(outputs, dim=1))
        torch.testing.assert_close(normalized[1], normalized[0], rtol=0.02, atol=0.02)
        max_abs = (normalized[1].float() - normalized[0].float()).abs().max().item()
        samples = {"A": [], "B": []}
        for arm in "ABBA":
            times = bench_gpu_time_with_cupti(
                functions[arm == "B"], use_cuda_graph=True, cold_l2_cache=True
            )
            samples[arm].append(statistics.median(times) * 1000)
        a, b = [statistics.median(samples[arm]) for arm in "AB"]
        row = {
            "tokens": m,
            "A_us": a,
            "B_us": b,
            "latency_reduction_pct": (1 - b / a) * 100,
            "speedup": a / b,
            "normalized_kv_max_abs": max_abs,
            "round_median_us": samples,
            "useful_kv_projection_tflops_A": 2 * m * K * LAYERS * V / (a * 1e6),
            "useful_kv_projection_tflops_B": 2 * m * K * LAYERS * V / (b * 1e6),
        }
        report["rows"].append(row)
        print(json.dumps(row), flush=True)
        args.output.write_text(json.dumps(report, indent=2) + "\n")
    report["status"] = "completed"
    args.output.write_text(json.dumps(report, indent=2) + "\n")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--checkpoint", type=Path, required=True)
    parser.add_argument("--baseline-source", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--tokens", type=int, nargs="+", default=[8, 48, 1024, 16384])
    args = parser.parse_args()
    priority = IrOpPriorityConfig.with_default(
        ["native"], rms_norm=["vllm_c", "native"]
    )
    with (
        torch.inference_mode(),
        set_current_vllm_config(VllmConfig()),
        priority.set_priority(),
    ):
        benchmark(args)

## Details

The stacked projection uses disable_tp=True and contains one WKV shard per draft layer. Checkpoint WKV weights and scales also continue loading into each layer’s original fused Q+KV projection for draft-query computation.

Context outputs are split by layer before the existing normalization and cache insertion. This adds a separate copy of the WKV weights and scales per TP rank.
