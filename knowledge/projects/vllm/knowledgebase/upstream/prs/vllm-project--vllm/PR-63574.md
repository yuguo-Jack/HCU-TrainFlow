## Overview

Raise patched dependency floors for file locking, telemetry, GraphQL and test/documentation tools, then regenerate all four platform test locks and the documentation lock. This follows #62835 and targets 126 alert instances in the audited dependency snapshot; individual advisories appear in multiple requirement files.

## Claims

| Dependency | Minimum / locked version |
| --- | --- |
| filelock | 4.2.4 |
| virtualenv | 21.14.13 |
| graphql-core | >=3.2.12,<3.3; locked to 3.2.13 |
| OpenTelemetry API, SDK and OTLP exporters | 1.45.1 |
| nltk | 3.10.4 |
| black | 26.10.1 |
| GitPython | 3.2.1 |

- Update source floors so regeneration retains the patched versions. Keep GraphQL within the existing 3.2 release line and align the telemetry package family.
- Preserve unrelated pins. Supporting changes are limited to virtualenv dependencies, telemetry dependencies and removal of packages no longer required by telemetry.

## Validation

- All five locks regenerated successfully with uv 0.11.1 and their checked-in platform/backend/constraint/exclusion arguments. All applicable pre-commit hooks and `git diff --check` passed.
- Existing vLLM tracing suite: **3 passed** with OpenTelemetry 1.45.1 and a local collector.
- Standalone local compatibility checks: **9 passed** for model-cache locking, virtualenv creation, NLTK model persistence, GraphQL parsing/execution, GitPython repository operations and Black formatting. The script is included below for reproduction and is not added to the repository.
- The three NLTK outside-root read/write and GraphQL comment-token-budget checks failed on the old nltk 3.10.3 / graphql-core 3.2.6 versions (`DID NOT RAISE`) and passed with the upgraded versions.

<summary>Standalone local compatibility checks</summary>

Save as `/tmp/test_dependency_compatibility.py` and run the command above in an environment with the upgraded packages. The before-change reproduction used a separate package directory containing only nltk 3.10.3 and graphql-core 3.2.6, prepended to `PYTHONPATH`, and selected `-k 'rejects_outside_model or comment_token_budget'`: **3 failed, 6 deselected**. With the upgraded packages the full script reported **9 passed**. Use a shared `/tmp` location for `--basetemp`: NLTK explicitly trusts private per-user temporary directories on macOS.

```python
import json
import subprocess
from pathlib import Path

import pytest


@pytest.fixture
def nltk_root(tmp_path, monkeypatch):
    import nltk
    import nltk.pathsec

    allowed = tmp_path / 'allowed'
    allowed.mkdir(mode=0o700)
    old_paths = list(nltk.data.path)
    nltk.data.path[:] = [str(allowed)]
    monkeypatch.setenv('NLTK_DATA', str(allowed))
    assert nltk.pathsec.ENFORCE
    try:
        yield allowed, tmp_path
    finally:
        nltk.data.path[:] = old_paths


def test_nltk_rejects_outside_model_write(nltk_root):
    from nltk.tag.perceptron import AveragedPerceptron

    _, root = nltk_root
    target = root / 'outside.json'
    with pytest.raises(PermissionError):
        AveragedPerceptron().save(str(target))
    assert not target.exists()


def test_nltk_rejects_outside_model_read(nltk_root):
    from nltk.tag.perceptron import AveragedPerceptron

    _, root = nltk_root
    target = root / 'outside.json'
    target.write_text(json.dumps({'feature': {'NN': 1.0}}))
    with pytest.raises(PermissionError):
        AveragedPerceptron().load(str(target))


def test_nltk_allowed_model_roundtrip(nltk_root):
    from nltk.tag.perceptron import AveragedPerceptron
    from nltk.stem import PorterStemmer

    allowed, _ = nltk_root
    target = allowed / 'weights.json'
    model = AveragedPerceptron()
    model.weights = {'feature': {'NN': 1.0}}
    model.save(str(target))
    loaded = AveragedPerceptron()
    loaded.load(str(target))
    assert loaded.weights == model.weights
    assert PorterStemmer().stem('running') == 'run'


def test_graphql_comment_token_budget():
    from graphql import GraphQLError, parse

    with pytest.raises(GraphQLError, match='more than 10 tokens'):
        parse('{\n' + '# comment\n' * 1000 + 'field\n}', max_tokens=10)


def test_graphql_normal_query():
    from graphql import build_schema, graphql_sync

    schema = build_schema('type Query { greeting: String! }')
    result = graphql_sync(schema, '{ greeting }', root_value={'greeting': 'hello'})
    assert result.errors is None
    assert result.data == {'greeting': 'hello'}


def test_vllm_model_cache_lock_contention(tmp_path):
    from filelock import Timeout
    from vllm.model_executor.model_loader.weight_utils import get_lock

    first = get_lock('test/model', cache_dir=str(tmp_path))
    second = get_lock('test/model', cache_dir=str(tmp_path))
    assert first.lock_file == second.lock_file
    assert first.mode == 0o666
    with first:
        with pytest.raises(Timeout):
            second.acquire(timeout=0.05)
    with second.acquire(timeout=1):
        assert second.is_locked


def test_virtualenv_creation_without_seeding(tmp_path):
    from virtualenv import cli_run

    target = tmp_path / '.venv'
    cli_run(['--no-seed', str(target)], setup_logging=False)
    result = subprocess.run([str(target / 'bin/python'), '-c', 'import sys; print(sys.prefix)'], check=True, capture_output=True, text=True)
    assert Path(result.stdout.strip()).resolve() == target.resolve()
    assert (target / 'pyvenv.cfg').exists()


def test_gitpython_local_repository(tmp_path):
    from git import Actor, Repo

    repo = Repo.init(tmp_path / 'repo')
    content = Path(repo.working_tree_dir) / 'example.txt'
    content.write_text('dependency check\n')
    repo.index.add(['example.txt'])
    actor = Actor('Dependency Check', 'dependency-check@example.invalid')
    commit = repo.index.commit('Local validation', author=actor, committer=actor)
    assert repo.head.commit.hexsha == commit.hexsha
    assert repo.git.show('HEAD:example.txt') == 'dependency check'
    assert not repo.is_dirty()


def test_black_basic_formatting():
    import black

    assert black.format_str('value=1\n', mode=black.Mode()) == 'value = 1\n'
```
