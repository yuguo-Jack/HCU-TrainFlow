# [Bugfix][Multimodal] Normalize direct PIL image inputs

[Upstream PR](https://github.com/vllm-project/vllm/pull/47566) · state: closed · merged: True

## Original PR description


## Purpose

vLLM normalizes image bytes from URL/base64/file inputs before they reach the model: JPEG EXIF orientation is applied to pixels and images are converted to the default RGB mode with transparency composited against the configured background.

The supported direct `PIL.Image` path used by offline `LLM.generate` and `LLM.chat` skipped that preprocessing and passed the raw PIL object into `ImageProcessorItems`. The same image could therefore reach the model with different pixels depending on whether the user passed an image URL/base64/file or `PIL.Image.open(...)`.

This PR applies the existing EXIF normalization and default RGB conversion to direct PIL image items in the parser. `MediaWithBytes`, tensor/ndarray inputs, embeddings, video and audio are unchanged.

## Test Plan

Run an end-to-end offline `LLM.generate` A/B on `HuggingFaceTB/SmolVLM-256M-Instruct`. The harness feeds a direct `PIL.Image` EXIF JPEG and a direct `PIL.Image` RGBA PNG through the real generation path, records the image item passed by `MultiModalDataParser` to the processor, and compares it with the existing `ImageMediaIO.load_bytes` bytes-loader result.

Run scoped ruff checks on `vllm/multimodal/parse.py`.

```python
from __future__ import annotations

import io
import json
import os
import subprocess
import sys
from typing import Any

from PIL import Image

os.environ.setdefault("VLLM_USE_FLASHINFER_SAMPLER", "0")
os.environ.setdefault("VLLM_ENABLE_V1_MULTIPROCESSING", "0")
os.environ.setdefault("VLLM_LOGGING_LEVEL", "ERROR")

import vllm
from vllm import LLM, SamplingParams
from vllm.multimodal.media.image import ImageMediaIO
from vllm.multimodal.parse import MultiModalDataParser

MODEL = os.environ.get("MODEL", "HuggingFaceTB/SmolVLM-256M-Instruct")
PROMPT = (
    "<|begin_of_text|>User:<image>Describe the image in one short sentence."
    "<end_of_utterance>\nAssistant:"
)
ORIENTATION = 0x0112


def to_jsonable(value: Any) -> Any:
    if isinstance(value, tuple):
        return list(value)
    return value


def image_summary(image: Image.Image) -> dict[str, Any]:
    return {
        "size": list(image.size),
        "mode": image.mode,
        "exif_orientation": image.getexif().get(ORIENTATION),
        "pixel_0_0": to_jsonable(image.getpixel((0, 0))),
        "pixel_1_0": to_jsonable(image.getpixel((min(1, image.width - 1), 0))),
    }


def make_exif_jpeg() -> bytes:
    image = Image.new("RGB", (40, 20), (255, 255, 255))
    for x in range(image.width):
        for y in range(image.height):
            image.putpixel((x, y), (255, 0, 0) if x < 10 else (0, 0, 255))

    exif = image.getexif()
    exif[ORIENTATION] = 6

    buffer = io.BytesIO()
    image.save(buffer, format="JPEG", exif=exif.tobytes())
    return buffer.getvalue()


def make_rgba_png() -> bytes:
    image = Image.new("RGBA", (2, 1), (255, 0, 0, 0))
    image.putpixel((1, 0), (0, 128, 0, 255))

    buffer = io.BytesIO()
    image.save(buffer, format="PNG")
    return buffer.getvalue()


def load_direct(data: bytes) -> Image.Image:
    image = Image.open(io.BytesIO(data))
    image.load()
    return image


captured: list[dict[str, Any]] = []
original_parse_image_data = MultiModalDataParser._parse_image_data


def patched_parse_image_data(self: MultiModalDataParser, data: Any) -> Any:
    result = original_parse_image_data(self, data)
    if result is not None:
        input_items: list[dict[str, Any]] = []
        if isinstance(data, Image.Image):
            input_items.append(image_summary(data))
        elif isinstance(data, (list, tuple)):
            input_items.extend(
                image_summary(item)
                for item in data
                if isinstance(item, Image.Image)
            )

        if input_items:
            parsed_items = [image_summary(result.get(i)) for i in range(len(result))]
            captured.append({"input": input_items, "parsed": parsed_items})
    return result


MultiModalDataParser._parse_image_data = patched_parse_image_data


def run_case(llm: LLM, name: str, data: bytes) -> bool:
    loader_image = ImageMediaIO().load_bytes(data).media
    direct_image = load_direct(data)

    captured.clear()
    outputs = llm.generate(
        {"prompt": PROMPT, "multi_modal_data": {"image": direct_image}},
        SamplingParams(temperature=0.0, max_tokens=4),
        use_tqdm=False,
    )
    generated_text = outputs[0].outputs[0].text

    if len(captured) != 1 or len(captured[0]["parsed"]) != 1:
        print(f"{name}_CAPTURE_ERROR", json.dumps(captured, sort_keys=True))
        return False

    parsed = captured[0]["parsed"][0]
    loader = image_summary(loader_image)

    matches_loader = (
        parsed["size"] == loader["size"]
        and parsed["mode"] == loader["mode"]
        and parsed["exif_orientation"] == loader["exif_orientation"]
        and parsed["pixel_0_0"] == loader["pixel_0_0"]
        and parsed["pixel_1_0"] == loader["pixel_1_0"]
    )

    print(f"{name}_GENERATION_TEXT", repr(generated_text))
    print(f"{name}_LOADER", json.dumps(loader, sort_keys=True))
    print(f"{name}_DIRECT_INPUT", json.dumps(captured[0]["input"][0], sort_keys=True))
    print(f"{name}_PARSED", json.dumps(parsed, sort_keys=True))
    print(f"{name}_MATCHES_BYTES_LOADER", matches_loader)
    return matches_loader


def main() -> int:
    commit = subprocess.check_output(
        ["git", "rev-parse", "HEAD"], text=True
    ).strip()
    print("commit:", commit)
    print("python:", sys.executable)
    print("vllm.__file__:", vllm.__file__)
    print("vllm.__version__:", getattr(vllm, "__version__", "<unknown>"))
    print("model:", MODEL)

    llm = LLM(
        model=MODEL,
        model_impl="transformers",
        trust_remote_code=True,
        max_model_len=8192,
        max_num_seqs=1,
        enforce_eager=True,
        gpu_memory_utilization=0.35,
        limit_mm_per_prompt={"image": 1},
    )

    exif_ok = run_case(llm, "EXIF", make_exif_jpeg())
    rgba_ok = run_case(llm, "RGBA", make_rgba_png())
    all_ok = exif_ok and rgba_ok
    print("DIRECT_PIL_MATCHES_BYTES_LOADER", all_ok)
    return 0 if all_ok else 2


if __name__ == "__main__":
    raise SystemExit(main())
```
