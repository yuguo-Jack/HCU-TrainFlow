``` python

#!/usr/bin/env python
"""Concurrent traffic driver to force adaptive draft-budget trimming."""
import json
import sys
import time
import urllib.request
from concurrent.futures import ThreadPoolExecutor

PORT = sys.argv[1] if len(sys.argv) > 1 else "8002"
N = int(sys.argv[2]) if len(sys.argv) > 2 else 16
MAXTOK = int(sys.argv[3]) if len(sys.argv) > 3 else 8

URL = f"http://localhost:{PORT}/v1/completions"
PROMPTS = [
    "Explain the theory of relativity in simple terms.",
    "Write a poem about a rainy day in the city.",
    "Summarize the history of the Roman Empire.",
    "Describe how a neural network learns.",
    "List five facts about the human brain.",
    "Tell a short story about a lost dog.",
    "Explain photosynthesis to a child.",
    "What are the causes of climate change?",
    "Describe the water cycle step by step.",
    "Write about the importance of sleep.",
    "Explain how the internet works.",
    "Give a recipe for chocolate chip cookies.",
    "Describe the life cycle of a butterfly.",
    "Explain the stock market in simple words.",
    "Write about the benefits of exercise.",
    "Describe the process of making coffee.",
]


def one(i):
    body = json.dumps({
        "model": "deepseek-ai/DeepSeek-V4.1-Flash",
        "prompt": PROMPTS[i % len(PROMPTS)],
        "max_tokens": MAXTOK,
        "temperature": 0.7,
        "seed": i,
    }).encode()
    req = urllib.request.Request(URL, data=body, headers={"Content-Type": "application/json"})
    t0 = time.time()
    try:
        with urllib.request.urlopen(req, timeout=1800) as r:
            d = json.load(r)
        return i, d.get("usage", {}).get("completion_tokens"), time.time() - t0
    except Exception as e:
        return i, f"ERR {type(e).__name__}: {e}", time.time() - t0


with ThreadPoolExecutor(max_workers=N) as ex:
    for i, ntok, dt in ex.map(one, range(N)):
        print(f"req {i}: tokens={ntok} in {dt:.1f}s", flush=True)
print("DONE")
```
