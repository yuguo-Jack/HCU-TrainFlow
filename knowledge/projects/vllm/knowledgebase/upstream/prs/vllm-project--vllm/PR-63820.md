### Perf

```bash
vllm serve zai-org/GLM-5.3 \
    --trust-remote-code \
    -tp 8 \
    --load-format fastsafetensors \
    --gpu-memory-utilization 0.88 \
    --enable-prefix-caching \
    --host 0.0.0.0 \
    --port 9256 \
    --max-model-len auto \
    --max-num-seqs 48 \
    --speculative-config '{"method":"mtp","num_speculative_tokens":5}'  <!-- pp 2 --> 

guidellm run \
  --backend '{"kind":"openai_http","target":"http://127.0.0.1:9256","model":"zai-org/GLM-5.3","request_format":"/v1/chat/completions","timeout":100000,"extras":{"body":{"temperature":1.0,"top_p":0.95}}}' \
  --tokenizer '{"kind":"huggingface_auto","model":"zai-org/GLM-5.3","load_kwargs":{"trust_remote_code":true}}' \
  --data 'kind=synthetic_text,prompt_tokens=8000,output_tokens=1000' \
  --profile 'kind=concurrent,warmup=0.1,cooldown=0.1' \
  --override profile.streams '1,4,16' \
  --constraint 'kind=max_duration,seconds=150' \
  --constraint 'kind=max_errors,count=10' \
  --seed 'kind=static,value=42' \
  --metrics 'kind=generative,sample_size=20' \
  --output 'kind=json,path=/home/yewentao256/guidellm-results/output_vllm_glm53_8k1k_1024.json'
```
