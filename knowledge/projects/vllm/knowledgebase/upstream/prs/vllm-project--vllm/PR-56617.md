# [Bugfix][Pooling] Release cached queries when Flash scoring is cancelled

[Upstream PR](https://github.com/vllm-project/vllm/pull/56617) · state: open · merged: False

## Original PR description

## Overview

Repeated cancellation during Flash late-interaction cleanup can leave requests pending or query embeddings cached. This finishes producer shutdown and cache cleanup before propagating cancellation, building on the basic release implementation in #58975.

## Claims

- Cancel and join query/document producers before aborting requests and releasing cached queries.
- Finish cleanup even if the handler is cancelled again during abort or cache release.
- Preserve request errors through the existing `RequestError.to_exception()` conversion.

```python
"""CPU Ray compiled-DAG reproducer; run against main and the PR checkout."""
import inspect
import threading
import time
from collections import deque
from functools import partial
from types import SimpleNamespace

import ray
import torch
from ray.dag import InputNode, MultiOutputNode

from vllm.pooling_params import LateInteractionParams, PoolingParams
from vllm.v1.engine.core import EngineCore
from vllm.v1.executor.ray_executor import RayDistributedExecutor
from vllm.v1.outputs import ModelRunnerOutput
from vllm.v1.pool.late_interaction_runner import LateInteractionRunner


@ray.remote(num_cpus=1)
class Worker:
    def __init__(self):
        self.runner = LateInteractionRunner(enable_flash=False)
        self.gate = threading.Event()
        self.started = False
        self.events = []

    def execute_model_ray(self, inputs):
        self.started = True
        self.events.append("batch started")
        assert self.gate.wait(30)
        self.runner.postprocess_pooler_output(
            [torch.ones((2, 4))],
            [PoolingParams(late_interaction_params=LateInteractionParams(
                mode="cache_query", query_key="A", query_uses=1))],
            ["A"], [True],
        )
        self.events.append("cache written")
        return ModelRunnerOutput(req_ids=["A"], req_id_to_index={"A": 0})

    def execute_method(self, method, *args, **kwargs):
        return getattr(self, method)(*args, **kwargs)

    def release(self, keys):
        self.events.append("cleanup")
        self.runner.release_queries(keys)

    def open_gate(self):
        self.gate.set()

    def state(self):
        return self.started, self.events, sorted(self.runner._query_cache)


def main():
    ray.init(num_cpus=2, include_dashboard=False, log_to_driver=False,
             object_store_memory=100 * 1024 * 1024)
    worker = Worker.remote()
    with InputNode() as inputs:
        graph = MultiOutputNode([worker.execute_model_ray.bind(inputs)])
    dag = graph.experimental_compile()
    timer = None
    try:
        executor = SimpleNamespace(forward_dag=dag, has_connector=False,
                                   workers=[worker])
        executor.collective_rpc = partial(
            RayDistributedExecutor.collective_rpc, executor)
        scheduled = SimpleNamespace()
        future = RayDistributedExecutor._execute_dag(
            executor, scheduled, None, non_block=True)
        deadline = time.monotonic() + 30
        while not ray.get(worker.state.remote())[0]:
            assert time.monotonic() < deadline
            time.sleep(0.01)
        core = SimpleNamespace(batch_queue=deque([(future, scheduled, future)]),
                               model_executor=executor)
        wait = "wait_for_inflight_batches" in inspect.signature(
            EngineCore.collective_rpc).parameters
        timer = threading.Timer(1, lambda: ray.get(worker.open_gate.remote()))
        timer.start()  # Independent of cleanup returning, including on branch.
        EngineCore.collective_rpc(
            core, "release", args=(["A"],),
            **({"wait_for_inflight_batches": True} if wait else {}))
        queued = core.batch_queue[0][0]
        output = queued.result(timeout=30)
        timer.join()
        _, events, cache = ray.get(worker.state.remote())
        assert output.req_ids == ["A"]
        assert cache == ([] if wait else ["A"])
        if wait:
            assert queued.result() is output  # No second CompiledDAGRef read.
        print("ref:", type(future.ref_or_refs).__name__)
        print("wait:", wait, "events:", " -> ".join(events))
        print("cache:", cache, "output:", output.req_ids)
        print("queued future replaced:", queued is not future)
    finally:
        if timer:
            timer.join(15)
        dag.teardown()
        ray.kill(worker)
        ray.shutdown()


if __name__ == "__main__":
    main()
```
