# [Perf][GLM-5.3] Fuse MXFP4 MoE input quantization with route. 20% TTFT improvement

[Upstream PR](https://github.com/vllm-project/vllm/pull/65420) · state: open · merged: False

## Original PR description

## Overview

Fuse CUTLASS MXFP4 MoE input preparation by removing the separate BF16 row-shuffle launch: the gather path reads routed token rows directly during quantization, while the large-batch fan-out path quantizes each token once and writes its FP4 values and E8M0 scales to expert routes. Both paths preserve the expert-ordered layout expected by CUTLASS GEMM1.

## Claims

- Fuse BF16 row gathering with MXFP4 conversion, eliminating the intermediate shuffled BF16 tensor and its separate launch.
- Add a quantize-once fan-out kernel for larger batches, writing each token’s quantized values and scales to its valid expert routes.
- Select between gather and fan-out using the measured thresholds.

"""Sweep the original, gather, and quantize-once MXFP4 input paths."""

from __future__ import annotations

import argparse
import math
import random
import statistics
from collections.abc import Callable

import torch

from vllm import _custom_ops as ops

HIDDEN_SIZE = 6144
NUM_EXPERTS = 256
TOPK = 8
DEFAULT_TOKENS = (
    1,
    2,
    4,
    8,
    16,
    24,
    32,
    48,
    64,
    96,
    128,
    192,
    256,
    512,
    1024,
    2048,
    4096,
    16384,
)


def parse_tokens(raw: str) -> tuple[int, ...]:
    tokens = tuple(int(value) for value in raw.split(",") if value.strip())
    if not tokens or any(value < 1 for value in tokens):
        raise argparse.ArgumentTypeError("token counts must be positive")
    return tokens


def parse_ints(raw: str) -> tuple[int, ...]:
    values = tuple(int(value) for value in raw.split(",") if value.strip())
    if not values:
        raise argparse.ArgumentTypeError("provide at least one value")
    return values


def route_metadata(
    num_tokens: int,
    num_experts: int,
    topk: int,
    route_pattern: str,
    seed: int,
    hidden_size: int,
) -> tuple[
    torch.Tensor,
    torch.Tensor,
    torch.Tensor,
    torch.Tensor,
    torch.Tensor,
    torch.Tensor,
    int,
]:
    generator = torch.Generator(device="cuda").manual_seed(seed)
    if route_pattern == "regular":
        route_ids = (
            torch.arange(num_tokens * topk, dtype=torch.int32, device="cuda")
            % num_experts
        ).view(num_tokens, topk)
    else:
        logits = torch.randn(
            num_tokens,
            num_experts,
            dtype=torch.float32,
            device="cuda",
            generator=generator,
        )
        if route_pattern == "skewed":
            logits[:, : min(32, num_experts)] += 3.0
        route_ids = logits.topk(topk, dim=1).indices.to(torch.int32).contiguous()
    expert_offsets = torch.empty(num_experts + 1, dtype=torch.int32, device="cuda")
    blockscale_offsets = torch.empty_like(expert_offsets)
    problem_sizes1 = torch.empty(num_experts, 3, dtype=torch.int32, device="cuda")
    problem_sizes2 = torch.empty_like(problem_sizes1)
    input_permutation = torch.empty(
        num_tokens * topk, dtype=torch.int32, device="cuda"
    )
    output_permutation = torch.empty_like(input_permutation)
    sorted_expert_ids = torch.empty_like(input_permutation)
    ops.get_cutlass_moe_mm_data(
        route_ids,
        expert_offsets,
        problem_sizes1,
        problem_sizes2,
        input_permutation,
        output_permutation,
        num_experts,
        128,
        hidden_size,
        blockscale_offsets,
        is_gated=True,
        sorted_expert_ids=sorted_expert_ids,
    )
    torch.cuda.synchronize()
    scale_rows = int(blockscale_offsets[-1].item())
    return (
        route_ids,
        expert_offsets,
        blockscale_offsets,
        input_permutation,
        output_permutation,
        sorted_expert_ids,
        scale_rows,
    )


def benchmark_graph(
    invoke: Callable[[], None], graph_nodes: int, warmup: int
) -> torch.cuda.CUDAGraph:
    warmup_stream = torch.cuda.Stream()
    warmup_stream.wait_stream(torch.cuda.current_stream())
    with torch.cuda.stream(warmup_stream):
        for _ in range(warmup):
            invoke()
    torch.cuda.current_stream().wait_stream(warmup_stream)

    graph = torch.cuda.CUDAGraph()
    with torch.cuda.graph(graph):
        for _ in range(graph_nodes):
            invoke()
    graph.replay()
    torch.cuda.synchronize()
    return graph


def graph_samples(
    graphs: dict[str, torch.cuda.CUDAGraph],
    samples: int,
    rng: random.Random,
    graph_nodes: int,
) -> dict[str, list[float]]:
    elapsed = {name: [] for name in graphs}
    start = torch.cuda.Event(enable_timing=True)
    end = torch.cuda.Event(enable_timing=True)

    for _ in range(samples):
        order = list(graphs)
        rng.shuffle(order)
        for name in order:
            start.record()
            graphs[name].replay()
            end.record()
            end.synchronize()
            elapsed[name].append(start.elapsed_time(end) * 1_000 / graph_nodes)
    return elapsed


def benchmark_shape(
    num_tokens: int,
    num_experts: int,
    topk: int,
    hidden_size: int,
    dtype: torch.dtype,
    route_pattern: str,
    cache_mode: str,
    graph_nodes: int,
    warmup: int,
    samples: int,
    rng: random.Random,
    seed: int,
    gather_block_sizes: tuple[int, ...],
) -> dict[str, list[float]]:
    (
        route_ids,
        expert_offsets,
        blockscale_offsets,
        input_permutation,
        output_permutation,
        sorted_expert_ids,
        scale_rows,
    ) = route_metadata(
        num_tokens, num_experts, topk, route_pattern, seed, hidden_size
    )
    num_routes = num_tokens * topk
    activation_seed = torch.Generator(device="cuda").manual_seed(seed + 1)
    base_activations = torch.randn(
        num_tokens,
        hidden_size,
        dtype=dtype,
        device="cuda",
        generator=activation_seed,
    )
    scale_cols = (hidden_size + 127) // 128
    scale_buffer_rows = num_routes + 127 * min(num_experts, num_routes)
    route_counts = torch.bincount(route_ids.flatten(), minlength=num_experts)
    cache_slots = 1
    if cache_mode == "cold":
        per_call_bytes = (
            base_activations.numel() * base_activations.element_size()
            + num_routes * (hidden_size // 2)
            + scale_rows
            * scale_cols
            * torch.tensor([], dtype=torch.int32).element_size()
        )
        l2_cache_size = getattr(
            torch.cuda.get_device_properties(), "L2_cache_size", 126 * 1024**2
        )
        cache_slots = max(1, math.ceil(2 * l2_cache_size / per_call_bytes))
    graph_nodes = max(graph_nodes, cache_slots)

    def activation_ring() -> torch.Tensor:
        ring = torch.empty(
            cache_slots,
            num_tokens,
            hidden_size,
            dtype=base_activations.dtype,
            device="cuda",
        )
        ring.copy_(base_activations.unsqueeze(0).expand_as(ring))
        return ring

    def next_slot(state: list[int]) -> int:
        slot = state[0]
        state[0] = (slot + 1) % cache_slots
        return slot

    arms: dict[str, Callable[[], None]] = {}
    arm_outputs: dict[
        str, tuple[torch.Tensor, torch.Tensor, list[int]]
    ] = {}

    def allocate_arm(name: str) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, list[int]]:
        activations = activation_ring()
        output = torch.empty(
            cache_slots,
            num_routes,
            hidden_size // 2,
            dtype=torch.uint8,
            device="cuda",
        )
        scales = torch.full(
            (cache_slots, scale_buffer_rows, scale_cols),
            -1,
            dtype=torch.int32,
            device="cuda",
        )
        state = [0]
        arm_outputs[name] = (output, scales, state)
        return activations, output, scales, state

    name = "baseline_shuffle_quantize"
    activations, output, scales, state = allocate_arm(name)
    shuffled_activations = torch.empty(
        cache_slots,
        num_routes,
        hidden_size,
        dtype=dtype,
        device="cuda",
    )

    def baseline(
        activations=activations,
        output=output,
        scales=scales,
        state=state,
        shuffled_activations=shuffled_activations,
    ) -> None:
        slot = next_slot(state)
        torch.ops._moe_C.shuffle_rows(
            activations[slot], input_permutation, shuffled_activations[slot]
        )
        torch.ops._C.mxfp4_experts_quant(
            output[slot],
            scales[slot],
            shuffled_activations[slot],
            expert_offsets,
            blockscale_offsets,
            num_experts,
            sorted_expert_ids,
        )

    arms[name] = baseline

    for block_size in gather_block_sizes:
        name = "gather_auto" if block_size == 0 else f"gather_{block_size}"
        activations, output, scales, state = allocate_arm(name)

        def gather(
            activations=activations,
            output=output,
            scales=scales,
            state=state,
            block_size=block_size,
        ) -> None:
            slot = next_slot(state)
            torch.ops._C.mxfp4_experts_quant(
                output[slot],
                scales[slot],
                activations[slot],
                expert_offsets,
                blockscale_offsets,
                num_experts,
                sorted_expert_ids,
                input_permutation,
                block_size,
            )

        arms[name] = gather

    name = "fanout_default"
    activations, output, scales, state = allocate_arm(name)

    def fanout(
        activations=activations,
        output=output,
        scales=scales,
        state=state,
    ) -> None:
        slot = next_slot(state)
        torch.ops._C.mxfp4_experts_quantize_once_fanout(
            output[slot],
            scales[slot],
            activations[slot],
            route_ids,
            output_permutation,
            expert_offsets,
            blockscale_offsets,
            num_experts,
        )

    arms[name] = fanout

    for invoke in arms.values():
        invoke()
    torch.cuda.synchronize()
    reference_output, reference_scales, reference_state = arm_outputs["gather_auto"]
    reference_slot = (reference_state[0] - 1) % cache_slots
    for name, (output, scales, state) in arm_outputs.items():
        slot = (state[0] - 1) % cache_slots
        if not torch.equal(reference_output[reference_slot], output[slot]):
            raise AssertionError(f"packed output mismatch for {name} at M={num_tokens}")
        if not torch.equal(reference_scales[reference_slot], scales[slot]):
            raise AssertionError(f"scale output mismatch for {name} at M={num_tokens}")

    graphs = {
        name: benchmark_graph(invoke, graph_nodes, warmup)
        for name, invoke in arms.items()
    }
    print(
        f"M={num_tokens} routes={route_pattern} experts_used="
        f"{int((route_counts > 0).sum().item())}/{num_experts} "
        f"max_routes_per_expert={int(route_counts.max().item())} "
        f"cache={cache_mode} graph_nodes={graph_nodes} ring_slots={cache_slots}"
    )
    return graph_samples(graphs, samples, rng, graph_nodes)


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--tokens", type=parse_tokens, default=DEFAULT_TOKENS)
    parser.add_argument("--experts", type=int, default=NUM_EXPERTS)
    parser.add_argument("--topk", type=int, default=TOPK)
    parser.add_argument("--hidden-size", type=int, default=HIDDEN_SIZE)
    parser.add_argument("--dtype", choices=("bf16", "fp16"), default="bf16")
    parser.add_argument(
        "--routes", choices=("regular", "random", "skewed"), default="regular"
    )
    parser.add_argument("--cache-mode", choices=("warm", "cold"), default="warm")
    parser.add_argument(
        "--gather-block-sizes", type=parse_ints, default=(0,)
    )
    parser.add_argument("--gather-block-size", type=int, choices=(0, 192, 384))
    parser.add_argument("--graph-nodes", type=int, default=64)
    parser.add_argument("--warmup", type=int, default=5)
    parser.add_argument("--samples", type=int, default=30)
    parser.add_argument("--seed", type=int, default=0)
    args = parser.parse_args()
    gather_block_sizes = (
        (args.gather_block_size,)
        if args.gather_block_size is not None
        else args.gather_block_sizes
    )
    if 0 not in gather_block_sizes:
        parser.error("--gather-block-sizes must include 0 for the auto reference")
    if any(value not in (0, 192, 384) for value in gather_block_sizes):
        parser.error("gather block sizes must be 0, 192, or 384")
    if args.graph_nodes < 1 or args.warmup < 0 or args.samples < 1:
        parser.error("graph nodes and samples must be positive; warmup nonnegative")
    if args.hidden_size < 32 or args.hidden_size % 32 != 0:
        parser.error("hidden size must be a positive multiple of 32")
    if args.experts < 1 or not 1 <= args.topk <= args.experts:
        parser.error("require experts >= 1 and topk in [1, experts]")
    if not torch.cuda.is_available():
        parser.error("requires a CUDA GPU")
    capability = torch.cuda.get_device_capability()
    if capability[0] not in (10, 11):
        parser.error(f"requires an SM10x/11x GPU, got SM{capability[0]}{capability[1]}")

    rng = random.Random(args.seed)
    dtype = torch.bfloat16 if args.dtype == "bf16" else torch.float16
    print(f"GPU: {torch.cuda.get_device_name()} (SM{capability[0]}{capability[1]})")
    print(
        f"E={args.experts}, topk={args.topk}, K={args.hidden_size}, "
        f"dtype={args.dtype}; "
        f"routes={args.routes}, cache={args.cache_mode}, "
        f"gather_blocks={gather_block_sizes}, "
        "fanout_geometry=4x8, "
        f"graph nodes={args.graph_nodes}, samples={args.samples}"
    )
    print("E,topk,K,dtype,routes,cache,M,arm,median_us,p95_us")
    for num_tokens in args.tokens:
        elapsed = benchmark_shape(
            num_tokens,
            args.experts,
            args.topk,
            args.hidden_size,

            )

    main()

### Input quantization paths and dispatch

`M` is the token-row count for one MoE invocation; 
- **Gather** kernel reads each routed row from the original tokens and quantizes that route.
- **Fan-out** kernel quantizes each token once, then writes its packed values and scale to each valid expert route.

Top-k 1 always uses gather. Top-k 2 uses fan-out at `M >= 80`; top-k 4 at `M >= 96`; other top-k values at `M >= 58`. For top-k 8 with `19 <= M <= 27` and `K=6144`, gather uses the measured 192-thread geometry; other gather cases use automatic geometry.

## Details

- The gather path reads the routed BF16 rows through a_map while quantizing them, removing the intermediate BF16 shuffle. For larger workloads, the fan-out path quantizes each token once and writes the packed values and scales to its routed experts. Both paths preserve the expert ordering expected by the CUTLASS GEMM.

- Dispatch uses the measured M (number of token rows passed to the MoE layer) and top-k ranges. Top-k 1 stays on gather. The measured fan-out thresholds are M >= 80 for top-k 2 and M >= 96 for top-k 4; describe the generic M >= 58 rule and the top-k 8 gather exception at M=19–27 as implemented. The current evidence is from B200; state which expert counts and K sizes were swept and note any hardware or shapes not validated.

## What I checked and found safe

- **Gather path is equivalent to `shuffle_rows` + quant by construction.** In
  `mxfp4_experts_quant.cu` the only load change is
  `input_row = __ldca(&input_permutation[rowIdx])`, with the *output* row still
  indexed by the expert-ordered `rowIdx` and the SF offset still from
  `rowIdx_in_expert`. Since `input_permutation == a_map` and
  `compute_arg_sorts` sets `input_permutation[start] = i/topk`, reading
  `in[a_map[rowIdx]]` is exactly `shuffle_rows(a, a_map)[rowIdx]` — no reordering
  of the quant math, so bit-identical, matching the `torch.equal` test.
- **Padding rows never deref the `-1` sentinel** (loop bounded to the valid
  prefix `min(numRows, input_offset_by_experts[n_experts])`).
- **Invalid-route / EP (`expert_map`) handling consistent:** still calls
  `_prepare_topk_ids_for_cutlass_fp4_moe`; the store-route guard rejects
